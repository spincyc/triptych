the precision needed to verify the claim they support. Dates are the dates on
which the source was read for this guide.

\subsection*{Patristic and medieval witnesses}

\begin{itemize}[leftmargin=1.3em, itemsep=0.25em]
\item Ambrose, \work{Expositio Evangelii secundum Lucam}, lib.\ VII, §245 (PL
15, col.\ 1764). Latin read in the \work{Patrologia Latina} text at Corpus
Corporum, \url{https://mlat.uzh.ch/}, 2026-07-25. \emph{The steward is not
censured; the allurement reading of \latin{mammona iniquitatis}; angels and
saints as the receivers.}
\item Augustine, \work{Quaestiones Evangeliorum} II.34, as transmitted verbatim
in Aquinas, \work{Catena aurea in Lucam}, cap.\ 16, lect.\ 2 (Corpus
Thomisticum, 2026-07-25). \emph{Fraud excluded from imitation; the \latin{e
contrario} structure; riches only to the wicked; the receivers are the just and
holy, not God's debtors.}
\item Augustine, \work{Sermo} 113 (NPNF: Sermon 63 on the New Testament), §§1--2,
\url{https://www.newadvent.org/fathers/160363.htm}, 2026-07-25. \emph{The saints
and the \latin{minimi} as receivers; mammon as a Hebrew word cognate to Punic;
the refutation of ``plunder and then give.''}
\item Augustine, \work{De spiritu et littera} 32.56,
\url{https://www.newadvent.org/fathers/1502.htm}, 2026-07-25. \emph{``Abba,
Father'' as the two peoples in one cry.}
\item Augustine, \work{Enarrationes in Psalmos} 17.28 and 17.32; 30 (Enarratio
I) §§1 and 3; 33 (sermo 2) §§1 and 11; 47 §§2, 8 and 9; 70 (sermo 1) §§3 and 5.
NPNF 1st ser.\ vol.\ 8, at New Advent, which numbers the expositions by the
Hebrew psalm number (18, 31, 34, 48, 71) and whose inline verse numbers run one
lower than the Vulgate. Read 2026-07-25.
\item John Chrysostom, \work{Homilies on the Epistle to the Romans}, Homily XIV
(opening on Rom 8:12--13), NPNF 1st ser.\ vol.\ 11,
\url{https://www.newadvent.org/fathers/210214.htm}, 2026-07-25. \emph{Debtors to
the Spirit; the spirit of bondage as the letter; ``Abba'' as the initiate's
word; joint-heirship.}
\item John Chrysostom, \work{Homily} I \work{on Hebrews}, NPNF 1st ser.\ vol.\
14, \url{https://www.newadvent.org/fathers/240201.htm}, 2026-07-25; Latin form
as in the \work{Catena aurea}, cap.\ 16, lect.\ 2. \emph{``It is not they who
receive us, but our own work.''}
\item Jerome, \work{Epistola} 121 \work{ad Algasiam}, q.\ 6 (PL 22, coll.\
1018--1021). Latin read in the \work{Patrologia Latina} text at Corpus Corporum,
2026-07-25. \emph{The defrauded master's praise; fraud toward the master and
prudence for himself; \latin{mamona} as Syriac; the parable as an exhortation to
remit debts; the restriction to those able to receive; the report of Theophilus
of Antioch's allegory; the negative result on Origen and Didymus.}
\item Gregory the Great, \work{Moralia in Iob} XXI.19.29, as transmitted in the
\work{Catena aurea}, cap.\ 16, lect.\ 2 (Corpus Thomisticum, 2026-07-25).
\emph{The poor as \latin{patroni}.}
\item Bede, \work{In Lucae Evangelium Expositio}, lib.\ V, on Luke 16 (PL 92,
coll.\ 528--531). Latin read in the \work{Patrologia Latina} text at Corpus
Corporum, 2026-07-25. \emph{Every possessor of money is a steward; the excuses
eschatologised; \latin{prudentiores in generatione sua} limited; ``whose works a
man does, of him he is called a son.''}
\item Basil, as transmitted in the \work{Catena aurea}, cap.\ 16, lect.\ 2, and
cited by Aquinas at \work{ST} II-II q.\ 32 a.\ 7 ad 1. \emph{Inherited
injustice in a patrimony.} Cited at the level of transmission; the underlying
Basilian work was not verified for this guide.
\item ``Origen,'' as transmitted in the \work{Catena aurea}, cap.\ 16, lect.\ 2.
\emph{The commendation as \latin{abusive dictum}.} Attribution unverified and
expressly bounded in the commentary; Jerome reports that he could not find
Origen on this parable.
\item Cyril of Alexandria, \work{Commentary on the Gospel of St Luke}, Sermon
108, tr.\ R.~Payne Smith (Oxford), pp.\ 507--510, read in the Internet Archive
scan \texttt{p2commentaryupon00cyriuoft}, 2026-07-25. \emph{The refusal to
allegorise; the rich man as the steward; ``when it has failed''; the
translator's note on the tense.}
\end{itemize}

\subsection*{Thomas Aquinas}

\begin{itemize}[leftmargin=1.3em, itemsep=0.25em]
\item \work{Super Epistolam ad Romanos}, cap.\ 8, lect.\ 2 (on \latin{ergo
debitores}) and lect.\ 3 (on vv.\ 14--17), Corpus Thomisticum,
\url{https://www.corpusthomisticum.org/cro05.html}, 2026-07-25. \emph{Debtors to
the Holy Spirit; the Spirit's leading and free will with Phil 2:13; \latin{abba}
and \latin{pater} as the two peoples; \latin{clamamus} as intensity of heart;
God himself as the principal good inherited; the objection from the father's
death; \latin{cohaeredes}; the necessity of \latin{si tamen compatimur}.}
\item \work{Summa theologiae} II-II q.\ 32 a.\ 7 (whether alms may be given from
ill-gotten goods), with its \latin{sed contra} from Augustine and its \latin{ad
1} stacking five readings of \latin{mammona iniquitatis}; and III q.\ 23 aa.\
1--4 on adoption, especially a.\ 1 ad 3 citing the Gloss on Rom 8. Corpus
Thomisticum, \url{https://www.corpusthomisticum.org/sth3027.html} and
\url{https://www.corpusthomisticum.org/sth4016.html}, 2026-07-25.
\item \work{Catena aurea in Lucam}, cap.\ 16, lect.\ 1--2, Corpus Thomisticum,
\url{https://www.corpusthomisticum.org/clc14.html}, 2026-07-25. \emph{Used as
the checked Latin transmission of Augustine, Ambrose, Gregory, Chrysostom,
Basil, Theophylact, and ``Origen'' on this pericope; each retained witness is
identified as transmitted, and the underlying work is named where it was
separately verified.}
\end{itemize}
