\section*{The Propers: Interpretive Possibilities}

\begin{studybox}{Exploratory notice}
These are editorial and AI proposals, not sourced historical intent or
attributed teaching. Each remains governed by the appointed texts, the documented
reception, and the stated strongest limit.
\end{studybox}

\begin{proposal}{Comparative preference becomes exclusive allegiance}
\pfield{Anchors}{Introit, one day better than thousands; Gradual, confidence in
the Lord rather than man or princes; Epistle, opposed flesh-and-Spirit regimes
and kingdom exclusion; Gospel, two masters; Communion, promoted
\latin{Primum}.}
\pfield{Mechanism}{Repeated comparisons of better trust and better dwelling
become an account of which good finally orders all others when the Gospel
makes divided mastery impossible.}
\pfield{Fruit}{Created goods can be used in order without being raised into a
rival final end.}
\pfield{What separate treatment misses}{The movement from comparative grammar
in the psalms to exclusive service and first priority in Matthew.}
\pfield{Precedent}{\textbf{Near analogue located}: other Triptych guides order
temporal goods without joining this exact five-part sequence.}
\pfield{Strongest limit}{The sequence may describe reception of the assembled
formulary, but it proves no compiler design, condemns no created good, and does
not turn comparative words into formal stages of one original argument.}
\end{proposal}

\begin{proposal}{Human inability and divine addition}
\pfield{Anchors}{Gospel, a person cannot \latin{adícere} one cubit and God
will \latin{adiciéntur} necessities; Communion, the same divine addition with
\latin{Primum} promoted to the beginning.}
\pfield{Mechanism}{The verb reverses agency: anxious mastery cannot add even
one measure, whereas necessities are added when God's kingdom and justice are
sought first.}
\pfield{Fruit}{Freedom from solicitude becomes receptive dependence, not the
mastery by which one tries to supply or enlarge oneself.}
\pfield{What separate treatment misses}{The Communion answers human inability
to add by repeating divine addition while making the kingdom the ordering
priority.}
\pfield{Precedent}{\textbf{Near analogue located}: another guide treats what
God adds when prayer does not presume, but the complete incapacity-to-addition
hinge is not reproduced there.}
\pfield{Strongest limit}{Repeated vocabulary supplies an exploratory hinge,
not a promise of prosperity, a counsel of passivity, or proof of compositional
intent.}
\end{proposal}

\begin{proposal}{A four-element salvation vocabulary}
\pfield{Anchors}{Collect, \latin{propitiatióne perpétua}; Alleluia,
\latin{Deo salutári nostro}; Secret, \latin{hóstia salutáris} and
\latin{propitiátio potestátis}; Postcommunion, \latin{perpétuæ salvatiónis
efféctum}.}
\pfield{Mechanism}{Repeated salvation and propitiation language crosses
entrance, praise, oblation, and completion.}
\pfield{Fruit}{Divine favor becomes saving worship and sacramental effect,
rather than remaining an abstract result.}
\pfield{What separate treatment misses}{The way cognate language changes
function across prayer, chant, oblation, and post-Communion petition.}
\pfield{Precedent}{\textbf{Near analogue located}: neighboring guides follow a
repeated word through changing ritual roles, but none treats this exact
four-element sequence.}
\pfield{Strongest limit}{The related words establish no single compositional
plan, and no whole-Missal semantic concordance was attempted.}
\end{proposal}

\begin{proposal}{Trust is spoken where neighbouring clauses are withheld}
\pfield{Anchors}{Introit, two appointed cuts before the psalm's final trust
blessing; Offertory, a cut before Psalm~33's hope clause; Gradual, explicit
confidence and hope; Eighth-Sunday Communion, the fuller Psalm~33 sequence as
an external liturgical comparator.}
\pfield{Mechanism}{The chant boundaries withhold explicit trust clauses at two
ritual moments while the Gradual speaks trust directly.}
\pfield{Fruit}{Trust can be heard not only in quoted words but in the proper's
distribution of explicit and implicit speech.}
\pfield{What separate treatment misses}{The coordinated relation between what
the two psalm excerpts stop before and what the Gradual voices.}
\pfield{Precedent}{\textbf{Precedent located}: neighboring guides use
coordinated cut points as mechanisms for reading an appointed sequence.}
\pfield{Strongest limit}{Inherited boundaries may have unrelated causes; no
intention follows from the cuts, and the fuller Eighth-Sunday occurrence is a
comparison, not omitted text restored here.}
\end{proposal}

\begin{proposal}{Mediation is not mastery}
\pfield{Anchors}{Gradual, refusal of final trust in human beings or princes;
Offertory, the encircling Angel; Gospel, exclusive service of one master.}
\pfield{Mechanism}{Created help can mediate the Lord's protection without
becoming the final object of confidence or a rival lord.}
\pfield{Fruit}{Legitimate authority, expertise, and angelic ministry remain
possible beneath undivided allegiance to God.}
\pfield{What separate treatment misses}{The distinction between receiving a
minister and submitting to a master.}
\pfield{Precedent}{\textbf{Precedent located}: another guide joins the Angel
and the princes under the distinction between created aid and final trust.}
\pfield{Strongest limit}{Matthew is an editorial bridge rather than a citation
in the psalm commentators; capitalization does not identify the Angel, and
ordinary human authority remains legitimate.}
\end{proposal}

\begin{proposal}{One desire-root turns toward contrary objects}
\pfield{Anchors}{Introit verse, \latin{concupíscit ánima mea}; Epistle,
\latin{caro concupíscit advérsus spíritum}.}
\pfield{Mechanism}{The same Latin root describes appetite directed toward
God's courts and appetite set against the Spirit.}
\pfield{Fruit}{Conversion appears as the transformation of desire rather than
the extinction of creaturely affect.}
\pfield{What separate treatment misses}{The lexical reversal between the
psalm verse and the Pauline lesson.}
\pfield{Precedent}{\textbf{Precedent located}: another guide proposes this
same turn between the two appointed elements.}
\pfield{Strongest limit}{The shared root may be coincidence;
\latin{cor meum et caro mea} lies beyond the appointed Introit cut, flesh is
not bodily nature, and no compiler intent follows.}
\end{proposal}
