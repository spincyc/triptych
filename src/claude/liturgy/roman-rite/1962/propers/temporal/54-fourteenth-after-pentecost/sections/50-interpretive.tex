\section*{The Propers: Interpretive Possibilities}

\begin{studybox}{What this section is}
Everything below is exploratory editorial and AI proposal. None of it is
sourced historical intent, attributed teaching, or documented reception, and
none of it should be quoted as any of those. Each proposal joins at least two
precisely named appointed elements, and each was selected from the conjunctions
a targeted precedent search actually reached; that search's classification is
carried here unchanged. \textbf{Its three verdicts are bounded and mean only
what they say}, and ``not located in the checked corpus'' is a statement about
a named corpus rather than a claim that a connection is unknown, unprecedented,
first, or authored by the model. The corpus searched is recorded in the
terminal appendix and in \texttt{research/scope.md}. Evidence, doctrine and the
literal sense of the appointed texts govern; where a proposal and a checked
witness conflict, the witness wins.
\end{studybox}

\begin{proposal}{P1. Goods are ordered, not refused}
\pfield{Anchors}{Introit \cue{Int.}, \latin{mélior est dies una in átriis tuis
super mília}; Gradual \cue{Grad.}, \latin{bonum est confídere in Dómino, quam
confídere in hómine}; Gospel \cue{Gosp.}, \latin{nemo potest duóbus dóminis
servíre} and \latin{non potéstis Deo servíre et mammónæ}; Communion
\cue{Comm.}, \latin{Primum}; Collect \cue{Coll.}}
\pfield{Mechanism}{The formulary states one choice in two grammatical
registers. The chants state it \textbf{comparatively} --- one day \emph{above}
thousands, better to trust in the Lord \emph{than} in man --- and the readings
state it \textbf{exclusively}: \latin{hæc enim sibi ínvicem adversántur}, no
man can serve two masters, you cannot serve God and mammon. The comparative
register is what keeps the exclusive one from reading as contempt for created
goods.}
\pfield{Fruit}{A doctrine of ordered preference preached without contempt for
creation, and the reason this Mass can command an absolute exclusion and still
promise that all things shall be added.}
\pfield{What the element-by-element reading misses}{Read one element at a time,
the chants look like decoration around two hard sayings. Read together, the two
registers are doing different grammatical work on the same choice.}
\pfield{Precedent}{\textbf{Near analogue located.} The nearest is a sibling
leaf's proposal joining Collect, Offertory, Secret and Postcommunion on
\latin{sic transeamus per bona temporalia, ut non amittamus aeterna}, stated as
a means-end relay; also a sibling on ``a good becomes dangerous when it becomes
an exemption,'' and two postconciliar analogues. \textbf{Why near and not
exact:} that Sunday's ordering is stated in its own appointed Collect, whereas
this one runs the other way and must be reconstructed across chants the Collect
does not mention; the nearest hit has no Gospel or chant anchor, and no ``added
unto you'' verb.}
\pfield{Controlling limit}{\textbf{A serious defeater, conceded rather than
answered:} the comparative in Ps.\ 117:8--9 is arguably a Hebrew idiom of
\emph{exclusion} rather than of degree, in which case both registers are
exclusive and the proposal collapses into the mammon reading. \textbf{The
proposal is therefore stated as contingent on the comparative reading of the
Latin \latin{quam}.} Secondly, the lexical evidence reaches only part of the
anchor set: the nearest lexically grounded set is Introit, Gradual, Epistle and
Gospel, so \textbf{the Communion and the Collect are thematic members here and
not lexical ones}, and that is said rather than blurred. Thirdly, this must not
be built as ``the chants soften the readings'': it is one unit, not a two-sided
contrast.}
\end{proposal}

\begin{proposal}{P2. What man cannot add}
\pfield{Anchors}{Gospel \cue{Gosp.}, Mt.\ 6:27 \latin{Quis autem vestrum
cógitans potest adícere ad statúram suam cúbitum unum?} and Mt.\ 6:33
\latin{adiciéntur}; Communion \cue{Comm.}, \latin{adiciéntur}; Collect
\cue{Coll.}, \latin{quia sine te lábitur humána mortálitas}; Epistle
\cue{Ep.}, \latin{ut non quæcúmque vultis, illa faciátis}}
\pfield{Mechanism}{One verb carries the exchange. The Gospel denies that any
man can \emph{add} a cubit to his stature and then promises that all things
\emph{shall be added}; the Communion repeats the promise alone; and the Collect
and the Epistle supply the same incapacity in other words. \textbf{What man
cannot add to himself, God adds to him}, and the reversal is in the morphology
--- active of the man who cannot, passive of the God who does.}
\pfield{Fruit}{A doctrine of grace read off a single verb, and the reason the
Mass's petition and its promise are one movement rather than two.}
\pfield{What the element-by-element reading misses}{Both occurrences of the
verb sit inside one pericope, so an element-by-element reading files them as a
feature of Matthew rather than as the axis on which the Collect's diagnosis and
the Communion's promise turn.}
\pfield{Precedent}{\textbf{Not located in the checked corpus.} The nearest
same-identity work is a sibling provider's leaf for this Sunday, whose
mechanism is a three-verb distinction and \textbf{not} the incapacity; outside
this identity, a neighbouring leaf's commentary on \latin{qui sine te esse non
possumus} is commentary prose rather than a proposal, rests on a different
Collect, and has no Epistle or Communion anchor and no ``add'' verb.}
\pfield{Search boundary}{The cubit-and-\latin{adícere} anchor occurs in no
interpretive proposal anywhere in the checked corpus outside the two
Fourteenth-Sunday leaves. \textbf{Not reached:} sermon literature generally ---
and a sermon registered for this very Sunday exists and was read for the
commentary above.}
\pfield{Controlling limit}{The Gospel's two occurrences are Matthew's own, and
\textbf{neither the Collect nor the Epistle uses the verb}, so the four-element
unit rests on a doctrinal and not a lexical join at those two members.
\textbf{That is a weaker join than P4's, and it is stated as such.} A second
edition of the 1962 book spells the root with a doubled \latin{-ii-} in all
three places; the difference does not touch the argument, but that reading was
taken from an optical text layer and \textbf{no page image was opened for it.}}
\end{proposal}

\begin{proposal}{P3. Propitiation and salvation saturate the prayers the people do not sing}
\pfield{Anchors}{Collect \cue{Coll.}, \latin{propitiatióne perpétua};
Secret \cue{Sec.}, \latin{tuæ propitiátio potestátis}; Postcommunion
\cue{Postcomm.}, \latin{perpétuæ \dots\ salvatiónis efféctum}; Alleluia
\cue{All.}, \latin{Deo salutári nostro}}
\pfield{Mechanism}{The three orations and the Alleluia carry vocabulary no
other element does --- \latin{propitiátio} twice, the \latin{salut-} and
\latin{salv-} root four times, \latin{semper} twice, \latin{perpétu-} twice ---
so \textbf{the sung and read elements state the problem in the language of
service, trust and anxiety while the priest's own prayers state it in the
language of expiation and rescue.} The Alleluia's \latin{salutári} is the
single word tying the day's one purely jubilant chant into that thread.}
\pfield{Fruit}{The Mass has two registers of address, and a reader who follows
only the readings hears one of them. It also explains why the Alleluia,
otherwise the least integrated element of the formulary, is not simply
decorative.}
\pfield{What the element-by-element reading misses}{Read in order, the orations
are three short prayers in a conventional idiom. Read together and against the
chants, they are the only place in the Mass where the problem is named as
expiation.}
\pfield{Precedent}{\textbf{Near analogue located.} A neighbouring leaf reads
one repeated Latin noun as a bracket across a formulary's orations --- ``the
day's first prayer and its last share one noun'' --- and sibling leaves treat
propitiation doctrinally at the Secret. \textbf{Why near and not exact:} the
precedented move is the bracket across orations; what is not precedented is
doing it across three orations \emph{plus a chant}, nor with this vocabulary.}
\pfield{Search boundary}{\textbf{Not reached:} any lexical concordance over the
orations of the whole Missal, which the tracked 1862 payload would in principle
support but which was not run.}
\pfield{Controlling limit}{The vocabulary of the three orations is the standard
vocabulary of Roman collects generally, so the saturation may be unremarkable.
\textbf{The counter-test was run and it holds only narrowly:} the substantive
\latin{propitiátio} appears in the appointed Latin of \textbf{no other} of the
fifteen formularies collated here, independently reproduced in a second
provider's collation --- \textbf{but the wider \latin{propiti-} sweep hits
eleven of the fifteen}, so the claim is specifically about the substantive and
\textbf{collapses if restated about the propitiation word-family generally}.
Fifteen collated formularies is the sample, not the Missal. And \textbf{this
proposal is invisible in the guide's own English witness}: the registered 1861
rendering drops \latin{propitiátio} at the Collect, turns it into a verb at the
Secret, and drops both \latin{semper} and the substantive at the
Postcommunion, so \textbf{every measurement supporting it must be conducted on
the Latin} --- and that Collect's English carries only an agreement-between-two-scans
detector, with no printed page and no page image ever seen.}
\end{proposal}

\begin{proposal}{P4. The hope-formula lies past the cut, and is handed on at another Mass}
\pfield{Anchors}{Introit \cue{Int.}, Ps.\ 83:10--11a with vv.~2--3a; Gradual
\cue{Grad.}, Ps.\ 117:8--9; Offertory \cue{Off.}, Ps.\ 33:8--9a}
\pfield{Mechanism}{The formula \latin{beatus \dots\ qui sperat in eo} stands at
the close of Ps.\ 33 (v.~23) \emph{and} at the close of Ps.\ 83 (v.~13). It
lies \textbf{past the Introit antiphon's cut at v.~11a and past its psalm
verse's cut at v.~3a}; it lies \textbf{one clause past the Offertory's cut at
v.~9a}; and \textbf{the Gradual alone sings it outright}. Three appointed
excerpts stop short of the same formula and a fourth sings it. \textbf{The
Introit belongs because the formula lies beyond both its boundaries, not
because it uses hope vocabulary --- which it does not:} the \latin{sper-} and
\latin{confid-} stems occur in the Gradual block only, twice each, and not at
all in the Introit.}
\pfield{Fruit}{The formulary's chief coherence is a shared editorial boundary,
and it is verifiable rather than thematic. It also gives the Gradual, the day's
plainest maxim, a structural role it does not obviously have.}
\pfield{The control the corpus supplies, and which no leaf has yet joined to
it}{\textbf{Six Sundays earlier the same Psalm 33 verse is appointed with the
hope clause this Offertory stops before.} The Eighth Sunday after Pentecost's
Communion prints Ps.\ 33:9 whole --- \latin{Gustate et videte, quoniam suavis
est Dominus: beatus vir, qui sperat in eo} --- and an accent-normalised sweep
over all twenty collated text records establishes that \latin{sperat in eo}
occurs in the appointed Latin of the \textbf{Eighth Sunday only}, and
\latin{gustate} in the Eighth and the Fourteenth only. So the beatitude of the
one who hopes is not absent from the repertory; it is assigned to the other
Mass.}
\pfield{Precedent}{\textbf{Precedent located.} A neighbouring leaf argues that
``three independent editorial boundaries in one formulary all fall at the same
place \dots\ each is something the liturgy \emph{did} to it,'' with its own
source-grounded section stating that the formulary's chief coherence is a
shared boundary and is verifiable. A postconciliar sibling's ``the liturgy
disarms the psalms it gives the assembly'' is a near analogue.}
\pfield{Search boundary}{The searched Missal shows Ps.\ 33:8--9 has exactly two
Roman uses, the second appointed immediately after the Canaanite-woman Gospel
--- \textbf{found only by reading a recorded line range, because a whole-word
search misses it entirely through hyphenation in the optical text.} Any further
``uncut occurrence'' negative over that artifact is weak evidence of absence.}
\pfield{Controlling limit}{\textbf{The strongest limit is that the antiphons'
extents are inherited chant tradition and the cuts were not made to suppress
anything.} The textual research flags explicitly that Ps.\ 33:9b and Ps.\ 83:13
are cut by antiphon boundaries with other explanations, and that \textbf{intent
was not established}; it records the coincidence and declines to read it.
Secondly, \latin{confidere} is a trust word rather than a hope word strictly: a
stricter count of \latin{spes} vocabulary alone finds \latin{sperare} twice in
the Gradual and nothing else, a weaker statement. Thirdly, the Eighth Sunday
reading was taken from a sibling collation and a second agreeing record,
\textbf{not from a page image}. Fourthly, Ambrose quotes the fuller verse
\emph{with} the hope clause in the fourth century, which sharpens the
observation about the Roman cut but says nothing about why it was made.}
\end{proposal}

\begin{proposal}{P5. Two chants, two divisions: the Fathers disagree at the Offertory and at the Gradual, and the formulary resolves neither}
\pfield{Anchors}{Offertory \cue{Off.}, Ps.\ 33:8--9a, \latin{Immíttet Angelus
Dómini \dots\ gustáte et vidéte}; Gradual \cue{Grad.}, Ps.\ 117:8--9,
\latin{quam speráre in princípibus}}
\pfield{Mechanism}{Each of the two chants carries an \textbf{independent}
documented division, and the Mass prints both without adjudicating. At the
\textbf{Offertory}, Augustine argues about which noun governs the verb ---
defending \latin{Immíttet Angelus Dómini} against ``some bad copies'' reading
\latin{Immittet Angelum Dominus} --- and then identifies that Angel as Christ,
the Angel of great counsel; while the same verse is attested by Cyril of
Jerusalem and by Ambrose as the Church's \emph{communion} invitation, and the
1962 book assigns it to the \emph{offertory}. At the \textbf{Gradual},
Augustine escalates the excluded second term from bad men to good men to
\textbf{good angels} --- ``for angels also are called princes'' --- making the
exclusion absolute rather than moral. \textbf{The angel who is the Mass's
rescuer at one chant is the Mass's excluded object of trust at the other, and
the Fathers reached each position separately.}}
\pfield{Fruit}{A cross-chant tension genuinely in the sources rather than
manufactured, which the formulary's own arrangement stages; and a way to treat
the Offertory's angel without either flattening the christological
identification or asserting it.}
\pfield{What the element-by-element reading misses}{Each division is reported
under its own chant as a local disagreement. Set side by side, they are the
same word doing opposite work at two moments of one Mass.}
\pfield{Precedent}{\textbf{Near analogue located.} A neighbouring leaf treats a
documented patristic division as itself the cross-element unit --- ``the
tradition genuinely disagrees at four points, and the formulary resolves none
of them \dots\ they yield different sermons, and the guide prints them as four
positions rather than one consensus.'' The angel-and-princes division itself is
documented only in the sibling provider's edition of this identity, classified
there as a near analogue. \textbf{Why near and not exact:} what is unprecedented
in this collection is doing it where \emph{both} elements of a two-chant
pairing carry an independent division.}
\pfield{Controlling limit}{\textbf{The cross-relation between the Gradual's
absolute exclusion and the Gospel's \latin{Nemo potest duóbus dóminis servíre}
is a lead flagged in this research and not Augustine's claim: he does not cite
Mt.\ 6:24 at that locus}, and the Gospel is accordingly not an anchor here. The
Offertory half carries three further limits: Augustine's text differs from the
missal's at the following verb, \latin{éruet} against \latin{erípiet}, which is
evidence that the missal's text is \textbf{not} simply his; the missal's
capital \latin{Angelus} is orthography and \textbf{does not encode} his
identification; and Cyril's authorship of the \work{Mystagogical Catecheses} is
disputed in modern scholarship and that question was not swept. The Gradual
half rests on one modern Latin transcription that reads the accusative where the
missal prints the ablative, \textbf{and nothing should be built on that variant
without checking \work{Patrologia Latina}~37}. All witnesses are read either in the English of the
\work{Nicene and Post-Nicene Fathers} --- first series for Augustine, second
series vol.~7 for the \work{Mystagogical Catecheses} --- or in a web Latin
transcription; \textbf{none is collated against Migne}.}
\end{proposal}
