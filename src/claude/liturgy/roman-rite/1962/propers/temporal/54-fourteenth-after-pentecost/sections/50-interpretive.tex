\section*{The Propers: Interpretive Possibilities}

\begin{studybox}{Exploratory notice}
These are editorial and AI proposals, not sourced historical intent or
attributed teaching. Each remains governed by the appointed texts, the checked
reception, and the stated strongest limit.
\end{studybox}

\begin{proposal}{Comparative preference becomes governing allegiance}
\pfield{Anchors}{Introit, one day better than thousands; Gradual, confidence in
the Lord rather than man or princes; Epistle, opposed flesh-and-Spirit regimes
and kingdom exclusion; Gospel, two masters; Communion, promoted
\latin{Primum}.}
\pfield{Mechanism}{Repeated comparisons of better trust and better dwelling
become an account of which good finally orders all others when the Gospel
makes divided mastery impossible.}
\pfield{Fruit}{Created goods can be used in order without being raised into a
rival final end.}
\pfield{What separate treatment misses}{The movement from comparative grammar
in the psalms to exclusive service and first priority in Matthew.}
\pfield{Precedent}{\textbf{Near analogue located}: other Triptych guides order
temporal goods without joining this exact five-part sequence.}
\pfield{Strongest limit}{The sequence may describe reception of the assembled
formulary, but it proves no compiler design, condemns no created good, and does
not turn comparative words into formal stages of one original argument.}
\end{proposal}

\begin{proposal}{Temporal addition beneath saving direction}
\pfield{Anchors}{Gospel, \latin{adícere} and \latin{adiciéntur}; Collect,
\latin{auxíliis} with \latin{dirigátur}; Postcommunion, \latin{ducant ad
perpétuæ salvatiónis efféctum}.}
\pfield{Mechanism}{The necessities promised by \latin{adiciéntur} remain
subordinate to the divine aid joined to \latin{dirigátur} and to the
sacraments' \latin{ducant} toward everlasting salvation.}
\pfield{Fruit}{Trust in provision remains possible without making provision
the goal of prayer or sacrament.}
\pfield{What separate treatment misses}{No disciple can \latin{adícere};
necessities \latin{adiciéntur}; the Church is the subject of
\latin{dirigátur} through God's \latin{auxíliis}; and the sacraments are the
subject of \latin{ducant}.}
\pfield{Precedent}{\textbf{Precedent located}: the sibling provider's guide
states this relation, and Claude proper~51 provides an analogue for what God
adds.}
\pfield{Strongest limit}{The orations do not use Matthew's lexical root;
grammar proves no common authorship and supports neither passivity nor a
promise of prosperity.}
\end{proposal}

\begin{proposal}{Saving language changes ritual role}
\pfield{Anchors}{Collect, \latin{salutária}; Alleluia, \latin{Deo salutári};
Secret, \latin{hóstia salutáris} and \latin{propitiátio}; Postcommunion,
\latin{salvatiónis efféctum}.}
\pfield{Mechanism}{A saving word-family moves from direction, to the God
praised, to the offered Host, to the sacramental effect toward which recipients
are led.}
\pfield{Fruit}{Salvation appears as divine identity, moral direction,
sacrificial remedy, and sacramental end rather than as an abstract result.}
\pfield{What separate treatment misses}{The way cognate language changes
subject, object, and function across prayer, chant, oblation, and
post-Communion petition.}
\pfield{Precedent}{\textbf{Near analogue located}: neighboring guides follow a
repeated word through changing ritual roles, but none treats this exact
four-element sequence.}
\pfield{Strongest limit}{Morphology and genres differ; the prayers' early
circulation does not establish design of their later Sunday sequence, and no
whole-Missal concordance was attempted.}
\end{proposal}

\begin{proposal}{One desire-root turns toward opposite objects}
\pfield{Anchors}{Introit verse, \latin{concupíscit}; Epistle,
\latin{concupíscit advérsus spíritum} and \latin{concupiscéntiis}; Gospel and
Communion, kingdom-seeking.}
\pfield{Mechanism}{The same Latin desire-root can name longing directed toward
God and desire opposed to the Spirit, with kingdom-seeking supplying the
ordering end.}
\pfield{Fruit}{Conversion appears as the redirection and healing of desire
rather than the extinction of creaturely affect.}
\pfield{What separate treatment misses}{The lexical reversal between the
psalm verse and the Pauline lesson.}
\pfield{Precedent}{\textbf{Precedent located}: the sibling provider's guide
proposes the same turn; another guide offers a near analogue in the retraining
of appetite.}
\pfield{Strongest limit}{The shared root may be coincidence; the positive
occurrence is in the Introit's psalm verse, flesh is not bodily nature, and no
compiler intent follows.}
\end{proposal}

\begin{proposal}{Mediation is not mastery}
\pfield{Anchors}{Gradual, refusal of final trust in human beings or princes;
Offertory, the encircling Angel; Gospel, exclusive service of one master.}
\pfield{Mechanism}{Created help can mediate the Lord's protection without
becoming the final object of confidence or a rival lord.}
\pfield{Fruit}{Legitimate authority, expertise, and angelic ministry remain
possible beneath undivided allegiance to God.}
\pfield{What separate treatment misses}{The distinction between receiving a
minister and submitting to a master.}
\pfield{Precedent}{\textbf{Precedent located}: the sibling provider's guide
joins the Angel and the princes, while another neighboring guide supplies an
analogue for preserving unresolved mediator identity.}
\pfield{Strongest limit}{Matthew is an editorial bridge rather than a citation
in the psalm commentators; capitalization does not identify the Angel, and
ordinary human authority remains legitimate.}
\end{proposal}

\begin{proposal}{Tasting discloses the master that orders desire}
\pfield{Anchors}{Gospel, two masters and mammon; Offertory,
\latin{gustáte et vidéte}; Secret, the saving Host; Communion, the
kingdom-first return.}
\pfield{Mechanism}{The received Eucharistic interpretation of tasting places
embodied reception inside the question of what governs desire and which
kingdom is first.}
\pfield{Fruit}{Moral allegiance and sacramental participation meet without
making bodily life the enemy.}
\pfield{What separate treatment misses}{The path from exclusive service,
through tasting and oblation, to kingdom-first reception.}
\pfield{Precedent}{\textbf{Precedent located}: the sibling provider's guide
states this proposal, and proper~48 supplies a strong analogue by joining
mammon and mastery with the same Psalm~33 tasting text at Communion.}
\pfield{Strongest limit}{Eucharistic reception does not exhaust Psalm~33;
felt sweetness is not the measure of sacramental fruit, and the relation proves
no compiler intent.}
\end{proposal}
