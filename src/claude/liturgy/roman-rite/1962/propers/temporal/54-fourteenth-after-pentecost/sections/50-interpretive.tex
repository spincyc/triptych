\section*{The Propers: Interpretive Possibilities}

\begin{studybox}{Exploratory notice}
These are editorial and AI proposals, not sourced historical intent or
attributed teaching. Each remains governed by the appointed texts, the checked
reception, and the stated strongest limit.
\end{studybox}

\begin{proposal}{Comparison prepares for exclusive choice}
\pfield{Anchors}{Introit, one day better than thousands; Gradual, confidence in
the Lord rather than man or princes; Epistle, opposed desires; Gospel, two
masters; Communion, \latin{Primum}.}
\pfield{Mechanism}{The chants train comparison among real goods before the
readings expose allegiances that cannot be combined. Preference matures into a
decisive ordering without contempt for creation.}
\pfield{Fruit}{The kingdom becomes the end by which food, clothing, human help,
and every other genuine good receives its measure.}
\pfield{What separate treatment misses}{The grammatical escalation from
``better than'' to mutually exclusive service.}
\pfield{Precedent}{\textbf{Near analogue located}: other Triptych guides order
temporal goods without using this Sunday's comparative-to-exclusive sequence.}
\pfield{Strongest limit}{Comparison does not make man, princes, possessions,
food, or clothing evil, and the sequence proves no compiler design.}
\end{proposal}

\begin{proposal}{Human inability and divine addition reverse the same verb}
\pfield{Anchors}{Gospel, the cubit man cannot add and the necessities that will
be added; Communion, the repeated promise; Collect, divine direction;
Postcommunion, sacramental leading.}
\pfield{Mechanism}{The Gospel's verbal field moves agency from anxious
self-enlargement to received provision. The prayers frame that reversal as
dependence on divine direction rather than passivity.}
\pfield{Fruit}{Freedom from anxiety becomes freedom for faithful action under
the Father's care.}
\pfield{What separate treatment misses}{The internal reversal between what the
disciple cannot add and what the Father adds.}
\pfield{Precedent}{\textbf{Near analogue located}: neighboring guides treat
divine addition or human incapacity, but not the cubit-incapacity hinge joined
to this Communion.}
\pfield{Strongest limit}{The antiphon shortens the Gospel; prudent labor remains
necessary, and the promise does not define every divine gift as a temporal
necessity.}
\end{proposal}

\begin{proposal}{Saving grammar changes role across the liturgical action}
\pfield{Anchors}{Collect, \latin{salutária} and perpetual propitiation;
Alleluia, \latin{Deo salutári nostro}; Secret, \latin{hóstia salutáris} and
propitiation; Postcommunion, the effect of perpetual salvation.}
\pfield{Mechanism}{One semantic field changes grammatical and ritual role:
saving things direct the Church, God is acclaimed as saviour, the offered
victim is saving and propitiatory, and the sacraments lead to salvation's
enduring effect.}
\pfield{Fruit}{Salvation appears as divine identity, moral direction,
sacrificial remedy, and sacramental end rather than as an abstract result.}
\pfield{What separate treatment misses}{Related forms perform different work
across one chant and three orations.}
\pfield{Precedent}{\textbf{Near analogue located}: the Claude Thirteenth
Sunday after Pentecost brackets its Gospel with a repeated oration term, but
does not trace this Sunday's salvation field.}
\pfield{Strongest limit}{No whole-Missal concordance establishes rarity; the
forms are not grammatically interchangeable, and the prayers' early
transmission does not prove design of the later four-element sequence.}
\end{proposal}

\begin{proposal}{Two reception disputes distinguish mediation from mastery}
\pfield{Anchors}{Gradual, ``princes''; Offertory, ``Angel of the Lord'';
Gospel, the two masters as doctrinal control.}
\pfield{Mechanism}{The disagreements over earthly or angelic princes and over
the Angel's identity together distinguish receiving creaturely ministry from
placing ultimate confidence in a creature.}
\pfield{Fruit}{Legitimate mediation can be affirmed without weakening the
exclusive service owed to God.}
\pfield{What separate treatment misses}{Each dispute appears lexical in
isolation; together they pose one question about subordinate help and final
allegiance.}
\pfield{Precedent}{\textbf{Near analogue located}: other guides make a genuine
patristic disagreement a synthesis unit, but do not join these two distinct
disputes.}
\pfield{Strongest limit}{The traditions remain plural. The proposal decides
neither the Angel's identity nor the exact referent of the princes, and it does
not make legitimate human or angelic help a rival master.}
\end{proposal}

\begin{proposal}{One desire-root turns toward opposed objects}
\pfield{Anchors}{Introit, \latin{concupíscit et déficit ánima mea}; Epistle,
\latin{caro concupíscit advérsus spíritum} and \latin{concupiscéntiis}; Gospel,
the first seeking of the kingdom.}
\pfield{Mechanism}{A common Latin root voices holy longing and disordered
opposition. Desire appears as a power whose moral character depends on its
object and order.}
\pfield{Fruit}{Ascetic struggle is directed toward rightly ordered love rather
than toward the destruction of desire.}
\pfield{What separate treatment misses}{The same root sounds first in a psalm
of longing and then in Paul's account of conflict.}
\pfield{Precedent}{\textbf{Precedent located}: the sibling provider's guide for
this identity proposes the same lexical turn; other corpus occurrences treat
concupiscence more generally.}
\pfield{Strongest limit}{Lexical coincidence is possible; the Introit instance
belongs to its psalm verse, and no compiler intention follows.}
\end{proposal}

\begin{proposal}{Idolatrous service and mammon test the governing end}
\pfield{Anchors}{Epistle, \latin{idolórum sérvitus}; Gospel,
\latin{duóbus dóminis}, \latin{servíre}, and mammon; Communion, first seeking
of the kingdom.}
\pfield{Mechanism}{The shared field of service lets the Gospel's allegiance
test illuminate why idolatry belongs among works that exclude from the
kingdom; the Communion supplies the positive governing end.}
\pfield{Fruit}{A vice catalogue, a dominical antithesis, and sacramental
priority become three forms of one question: what receives service?}
\pfield{What separate treatment misses}{The service vocabulary connects the
Pauline list to Matthew's explicit master-language and to the Communion's
positive command.}
\pfield{Precedent}{\textbf{Not located in the checked corpus}: no non-target
Triptych proposal joins these exact anchors.}
\pfield{Strongest limit}{The grammar is uneven, and the traditional bridge
between avarice and idolatry is reception rather than Paul's plain wording.
Mammon must not be equated with every work in the Pauline list.}
\end{proposal}
