% Presentation primitives for the Claude edition of the U.S.-English
% postconciliar proper guides.  The file supplies only tabular environments,
% a compact title block, and two inline cues.  It defines no content, asserts
% no source, and deliberately avoids landscape, multi-column, and picture
% environments so that every leaf using it can remain web-eligible.

% Compact identification block: celebration, cycle/formula, governing books.
\newcommand{\properheading}[3]{%
  \begin{center}
    {\Large\bfseries #1}\par
    \vspace{0.16em}
    {\large #2}\par
    \vspace{0.16em}
    {\small #3}
  \end{center}
  \vspace{0.30em}
}

% Inline cues.  \incipit marks a Latin incipit or short Latin cue; \bid marks
% a stable semantic branch identifier used identically in the appendix and in
% the tracked audit records.
\newcommand{\incipit}[1]{\textit{#1}}
\newcommand{\bid}[1]{{\ttfamily\hyphenchar\font=45\relax #1}}
% \work marks the title of a cited work; \term marks a technical term at first
% use.  Both are typographic only and assert nothing about authority.
\newcommand{\work}[1]{\textit{#1}}
\newcommand{\term}[1]{\textit{#1}}
