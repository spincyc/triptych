\section{Scriptural Date and Location}

\zlabel{triptych:concise:chronology:start}
\begin{dossiertable}
Communion, first option & Ps 119 (118):4--5 & No place named in the
verses & \chronodate{communion-antiphon-a}{\chronologyannotation{communion-antiphon-a}}\\
\cmidrule(lr){1-4}
\dossierprose{The opening strophe of the alphabetical psalm of the law; no
author is named and no traditional date is held. The date is the modern
critical boundary for the whole Psalter, within which no single psalm can be
dated securely.}
\midrule
Responsorial Psalm & Ps 145 (144):2--3, 8--9, 17--18 & No place named &
\chronodate{responsorial-psalm}{\chronologyannotation{responsorial-psalm}}\\
\cmidrule(lr){1-4}
\dossierprose{Headed in the Douay--Rheims ``Praise, for David himself''
(v.~1): an attribution by title, with no traditional date attached. The
response is taken from v.~18a. The date is the Psalter's critical boundary
only.}
\midrule
First Reading & Isa 55:6--9 & Isaias, ``concerning Juda and Jerusalem''
(1:1); to his own people &
\chronodate{first-reading}{\chronologyannotation{first-reading}}\\
\cmidrule(lr){1-4}
\dossierprose{Traditional attribution: the prophet Isaias, whose ministry the
\work{Catholic Encyclopedia} (1910) bounds by the range shown; it assigns no
year to the writing of the book. It reports the contention of most modern
non-Catholic scholars that chapters 54--55, treated under ``Second Isaias'',
come from an author ``living towards the close of the Babylonian Captivity'',
and sets against it the Pontifical Biblical Commission's decision of 28~June
1908. No date is given for that position.}
\midrule
Gospel & Mt 20:1--16a & Spoken to the disciples after Peter's question
(19:27) & \chronodate{gospel}{\chronologyannotation{gospel}}\\
\cmidrule(lr){1-4}
\dossierevent{Narrated event: a parable of Jesus, told within the journey
that ends at Jerusalem, before the third prediction of the Passion
(20:17--19).}
\cmidrule(lr){1-4}
\dossierprose{Traditional attribution: St~Matthew the Apostle. The Date column
keeps as disputed alternatives the figures the \work{Catholic Encyclopedia}
reports (1911, 1912): its article on the Gospel says that ``Ancient
ecclesiastical writers are at variance'' and lists further modern figures,
which the chronology record does not carry; the single year printed last is
said, in the article on the New Testament, of the original text of Matthew,
which its author holds to have been Aramaic.}
\midrule
Communion, second option & Jn 10:14 & To the Pharisees, after the man born
blind is healed (9:40--10:6) &
\chronodate{communion-antiphon-b}{\chronologyannotation{communion-antiphon-b}}\\
\cmidrule(lr){1-4}
\dossierevent{Narrated event: the discourse of the good shepherd.}
\cmidrule(lr){1-4}
\dossierprose{Traditional attribution: St~John the Apostle. The
\work{Catholic Encyclopedia} gives the first range shown for ``the Johannine
writings'' (1912) and the second as ``the general opinion'' for the Gospel,
after conceding that ``we possess no certain historical information'' (1910).
The Missal adapts the verse by adding \latin{dicit Dominus} and the noun
\latin{oves}.}
\midrule
Alleluia verse (its basis) & cf.\ Acts 16:14b, an adaptation & Event:
Philippi in Macedonia, by the riverside on the sabbath (16:12--13) &
\chronodate{gospel-acclamation}{\chronologyannotation{gospel-acclamation}}\\
\cmidrule(lr){1-4}
\dossierevent{Narrated event: the conversion of Lydia during St~Paul's second
missionary journey (Acts 15:36--18:22).}
\cmidrule(lr){1-4}
\dossierprose{Traditional attribution: St~Luke. The event ranges are those the
\work{Catholic Encyclopedia} gives for the second journey in its articles on
St~Paul (1911), which does not name Lydia, and on biblical chronology (1908);
the composition date is the one its article on Acts (1907) calls most
probable. The liturgical verse is a petition and not itself a verse of
Scripture.}
\midrule
Second Reading & Phil 1:20c--24, 27a & Paul in bonds (1:7, 13), at Rome; to
the saints at Philippi (1:1) &
\chronodate{second-reading}{\chronologyannotation{second-reading}}\\
\cmidrule(lr){1-4}
\dossierprose{Traditional attribution: St~Paul, writing from captivity. The
first figure is the year the \work{Catholic Encyclopedia}'s chronology of
St~Paul gives the captivity letters (1911); the second is the range its
article on the letter gives for Paul ``at Rome'' (1911), where critics are
said to disagree whether the letter falls early or late in the Roman sojourn,
its author holding for late.}
\end{dossiertable}
