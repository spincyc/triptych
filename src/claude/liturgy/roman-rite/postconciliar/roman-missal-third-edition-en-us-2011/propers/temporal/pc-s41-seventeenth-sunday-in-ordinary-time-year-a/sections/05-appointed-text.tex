\section*{The Appointed Texts: Rights-Limited Study Sheet}

\begin{studybox}{Publication boundary}
The approved U.S. Missal and Lectionary English is protected and is not
reproduced. This is the complete ordered inventory: lawful Latin incipits,
public-domain biblical loci, exact alternatives, and descriptions identifying
the appointed matter without replacing the liturgical books.
\end{studybox}

\begin{unitstable}
\unitphase{Introductory Rites}
\unitrow{Entrance Antiphon}{\emph{Cf.} Ps 67:6--7, 36 (modern 68);
\incipit{Deus in loco sancto suo}; God provides a dwelling and strength.}
\unitrow{Collect}{\incipit{Protéctor in te sperántium, Deus}; protection and
guidance in using passing goods while adhering to those that endure.}
\unitphase{Liturgy of the Word}
\unitrow{First Reading}{1 Kings 3:5, 7--12; Solomon asks for a hearing heart
able to judge the entrusted people and distinguish good from evil.}
\unitrow{Responsorial Psalm}{Ps 119:57, 72, 76--77, 127--130; response
\emph{cf.} v. 97a; instruction is loved above wealth and asked to give light.}
\unitrow{Second Reading}{Romans 8:28--30; divine purpose amid groaning tends
toward conformity to the firstborn Son.}
\unitrow{Gospel Acclamation}{\emph{Cf.} Matthew 11:25; the Father's revelation
is received by the little ones.}
\unitrow{Gospel}{Long: Matthew 13:44--52, treasure, pearl, net, sorting, and the
instructed scribe. Short: Matthew 13:44--46, treasure and pearl only.}
\unitphase{Liturgy of the Eucharist}
\unitrow{Prayer over the Offerings}{\incipit{Súscipe, qu\'{\ae}sumus, Dómine,
múnera}; bounty first received is offered back, with present sanctification
sought.}
\unitrow{Communion Antiphon A}{Ps 102:2 (modern 103); bless the Lord and do not
forget his benefits.}
\unitrow{Communion Antiphon B}{Matthew 5:7--8; mercy and purity of heart retain
their distinct promises.}
\unitrow{Prayer after Communion}{\incipit{Súmpsimus, Dómine, divínum
sacraméntum}; the perpetual memorial of the Passion is ordered toward saving
fruit.}
\end{unitstable}
