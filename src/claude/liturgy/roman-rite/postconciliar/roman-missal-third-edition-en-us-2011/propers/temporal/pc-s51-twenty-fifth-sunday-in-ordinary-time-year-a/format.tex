% Quiet running matter: the Sunday on the left, the current section on the
% right, the folio centred below.
%
% fancyhdr sets the left and the right head string each in a box the full
% width of the head, so an overlong right string prints over the left one
% without any warning. The right box is therefore cut to what the fixed left
% string and a gutter leave of the head, so that a string too long for it
% wraps inside its own box and trips the \headheight warning a log inspection
% catches. A section whose title is wider than that box declares a short
% running form with \runninghead just before its heading; a starred heading,
% which sets no mark, calls \runninghead just after it.
%
% \runninghead sets the mark at once and hands the same string to the next
% \sectionmark, because \rightmark takes the first mark on a page and a mark
% set only after \section would lose to the heading's own on the page the
% heading opens. Its body holds only \gdef and \markright: the web converter
% carries this definition into the document it hands pandoc, and a \newif
% boolean there stops the conversion. The test is therefore an \ifx inside
% \sectionmark, which LaTeX alone calls.
\usepackage{fancyhdr}
\pagestyle{fancy}

% --- Scriptural Date and Location -----------------------------------------------
% Two-tier dossier table after the postconciliar profile's page-2 standard: a
% four-field summary row, then a full-width explanatory row, with an italic
% event row for a narrated event and \midrule only between complete dossiers.
%
% Measurement note. The profile's starting widths are 0.14 / 0.18 / 0.35 /
% 0.16\linewidth with a 0.92\linewidth explanatory row, to be changed only as
% the complete single-page sheet requires. Two measurements are varied:
%
%   D1. Summary-tier widths 0.13 / 0.16 / 0.29 / 0.36\linewidth. The
%       generated annotation prints six disputed composition labels at the
%       Gospel and an event statement beside them, each as the chronology
%       corpus words it, and a 0.16\linewidth Date cell sets them as a column
%       of fragments.
%   D2. Explanatory and event rows at 0.98\linewidth in \scriptsize, so that
%       the same environment can hold seven dossiers on the single physical
%       page the concise companion's page 2 must occupy.
%
% Kept as the profile gives them: the two-tier hierarchy, the four summary
% fields, \footnotesize for the summary tier, \arraystretch 1.02, \LTpre
% 0.15em, \LTpost 0pt, \cmidrule(lr){1-4} within a dossier and \midrule
% between dossiers.
\newenvironment{dossiertable}
{\footnotesize
 \renewcommand{\arraystretch}{1.02}%
 \setlength{\LTpre}{0.15em}\setlength{\LTpost}{0pt}%
 \begin{longtable}{@{}>{\raggedright\arraybackslash}p{0.13\linewidth}>{\raggedright\arraybackslash}p{0.16\linewidth}>{\raggedright\arraybackslash}p{0.29\linewidth}>{\raggedright\arraybackslash}p{0.36\linewidth}@{}}
 \toprule
 \textbf{Textual unit} & \textbf{Citation} & \textbf{Location} & \textbf{Date}\\
 \midrule
 \endhead}
{\bottomrule\end{longtable}}
