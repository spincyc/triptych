% Claude Rosary reference: leaf-local typesetting primitives. This file owns
% only presentation. Every doctrinal, historical, and evidentiary judgment
% remains in the section files and research records.

\usepackage{endnotes}
% Long archival URLs in the references must break freely rather than
% overflowing the measure.
\usepackage{xurl}

\setcounter{tocdepth}{1}
\setlength{\emergencystretch}{2.2em}

\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000

% Work titles and transliterated or Latin technical terms are set through
% semantic commands so a later style change never requires editing the prose.
\newcommand{\work}[1]{\textit{#1}}
\newcommand{\term}[1]{\textit{#1}}

% The governing thesis is substantive content and may open the body.
\newcommand{\rosarythesis}[1]{%
  \begin{studybox}{Governing thesis}
  #1
  \end{studybox}}

% Received-text block: a vocal prayer reproduced verbatim from one identified
% received witness, named in the frame title. The framed border marks quoted
% source text that is not project-owned expression.
\newenvironment{receivedtext}[1]{%
  \begin{massbox}{Received text --- #1}}{\end{massbox}}

% Dated-discipline block: a mutable claim (currently the indulgence grant)
% kept visually distinct with its authority and as-of date in the title.
\newenvironment{disciplinebox}[1]{%
  \begin{massbox}{Current discipline --- #1}}{\end{massbox}}

% Endnotes carry citations and subordinate evidence detail; the heading joins
% the table of contents as an ordinary numbered section.
\renewcommand{\enoteheading}{\section{Notes}}
\renewcommand{\enoteformat}{%
  \noindent\footnotesize\makebox[1.8em][l]{\theenmark.}\hangindent=1.8em}
