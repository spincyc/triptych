% Claude Fatima apparition study: leaf-local typesetting primitives. This file
% owns only presentation. Every historical, doctrinal, and evidentiary
% judgment remains in the section files and research records.

\usepackage{endnotes}
% Long archival URLs in the references must break freely rather than
% overflowing the measure.
\usepackage{xurl}

\setcounter{tocdepth}{1}
\setlength{\emergencystretch}{2.2em}

\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000

% Work titles and untranslated technical terms are set through semantic
% commands so a later style change never requires editing the prose.
\newcommand{\work}[1]{\textit{#1}}
\newcommand{\term}[1]{\textit{#1}}

% The governing thesis is substantive content and may open the body.
\newcommand{\studythesis}[1]{%
  \begin{studybox}{Governing thesis}
  #1
  \end{studybox}}

% Operative-text block: the exact wording of a competent ecclesiastical act,
% with its authority, date, and witness in the frame title. The framed border
% marks quoted source text that is not project-owned expression.
\newenvironment{operativetext}[1]{%
  \begin{massbox}{Operative text --- #1}}{\end{massbox}}

% Endnotes carry citations and subordinate evidence detail; the heading joins
% the table of contents as an ordinary numbered section.
\renewcommand{\enoteheading}{\section{Notes}}
\renewcommand{\enoteformat}{%
  \noindent\footnotesize\makebox[1.8em][l]{\theenmark.}\hangindent=1.8em}

% Status-dossier card: the controlling-status summary in a titled, framed
% block. massbox is breakable, so a long dossier keeps closed frame edges
% wherever it actually ends.
\newenvironment{statusdossier}[1]{%
  \begin{massbox}{#1}\small}{\end{massbox}}
\newcommand{\dosfield}[2]{\par\textbf{#1:} #2}

% Project-synthesis block. This study distinguishes four voices and marks
% them differently: quoted source text is framed (operativetext); magisterial
% and authorial readings are attributed in running prose to their authors at
% their loci; verified fact carries its witness in the notes; and the
% project's own judgment is set here, behind a left rule and under an
% explicit label, so that a reader can weigh it as the editorial and AI
% proposal it is and never mistake it for received teaching.
\newenvironment{synthesis}[1]{%
  \begin{studybox}{Project synthesis --- #1}}{\end{studybox}}

% Register of the study's own judgments. The judgment text is the wide
% column because it is the content; the confidence qualifier is
% subordinate to it; and the cross-reference needs only a narrow rule.
% Field names are repeated in the head so the form stays readable when
% the table breaks across a page.
\newenvironment{judgmentstable}
{\small\begin{longtable}{>{\raggedright\arraybackslash}p{0.40\linewidth}>{\raggedright\arraybackslash}p{0.40\linewidth}>{\raggedright\arraybackslash}p{0.10\linewidth}}
\toprule
\textbf{Judgment} & \textbf{Confidence} & \textbf{Argued}\\
\midrule
\endhead}
{\bottomrule\end{longtable}}

% Named field inside a synthesis block. The repeated field names (judgment,
% reasons, strongest counterargument, what would change this) are a stable
% semantic form, not decoration: every published judgment in this study is
% required to fill them.
\newcommand{\synthfield}[2]{\par\textbf{#1:} #2}
