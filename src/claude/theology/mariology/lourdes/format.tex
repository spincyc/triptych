% Work titles and untranslated technical terms are set through semantic
% commands so a later style change never requires editing the prose.
\newcommand{\work}[1]{\textit{#1}}
\newcommand{\term}[1]{\textit{#1}}

% The governing thesis is substantive content and may open the body.
\newcommand{\studythesis}[1]{%
  \begin{studybox}{Governing thesis}
  #1
  \end{studybox}}

% Operative-text block: the exact wording of a competent ecclesiastical act,
% with its authority, date, and witness in the frame title. The framed border
% marks quoted source text that is not project-owned expression.
\newenvironment{operativetext}[1]{%
  \begin{massbox}{Operative text --- #1}}{\end{massbox}}

% Endnotes carry citations and subordinate evidence detail; the heading joins
% the table of contents as an ordinary numbered section.
\renewcommand{\enoteheading}{\section{Notes}}
\renewcommand{\enoteformat}{%
  \noindent\footnotesize\makebox[2.2em][l]{\theenmark.}\hangindent=2.2em}

% Status-dossier card: the controlling-status summary in a titled, framed
% block. massbox is breakable, so a long dossier keeps closed frame edges
% wherever it actually ends.
\newenvironment{statusdossier}[1]{%
  \begin{massbox}{#1}\small}{\end{massbox}}
