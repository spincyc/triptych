% Claude historical-accounts collection (biblical translations): shared
% typesetting primitives. This file owns only presentation. Every historical
% and evidentiary judgment remains in each account's own leaf.

\usepackage{endnotes}
% Long archival URLs in the references must break freely rather than
% overflowing the measure.
\usepackage{xurl}

\setcounter{tocdepth}{1}
\setlength{\emergencystretch}{2.2em}

\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000

% Ancient work titles are set consistently through one semantic command so a
% later style change never requires editing the prose.
\newcommand{\work}[1]{\textit{#1}}

% The governing thesis is substantive content and may open the body.
\newcommand{\historythesis}[1]{%
  \begin{studybox}{Governing thesis}
  #1
  \end{studybox}}

% Key-witness record: what the witness is, what it establishes, and the
% claim-local limit that must travel with it.
\newcommand{\witnessrecord}[3]{%
  \begin{studyframe}
  \textbf{Witness:} #1\par
  \textbf{What it establishes:} #2\par
  \textbf{Limit:} #3
  \end{studyframe}}

% Endnotes carry citations and subordinate evidence detail; the heading joins
% the table of contents as an ordinary numbered section.
\renewcommand{\enoteheading}{\section{Notes}}
\renewcommand{\enoteformat}{%
  \noindent\footnotesize\makebox[1.8em][l]{\theenmark.}\hangindent=1.8em}

% Evidence-class key for the terminal apparatus.
\newenvironment{evidencekey}
{\small
 \begin{longtable}{>{\bfseries\raggedright\arraybackslash}p{0.08\linewidth}
                   >{\raggedright\arraybackslash}p{0.85\linewidth}}
\toprule
\textbf{Class} & \textbf{Meaning}\\
\midrule
\endhead}
{\bottomrule\end{longtable}}
\newcommand{\evidenceclass}[2]{#1 & #2\\}

% Dated orientation timeline for the terminal apparatus: each row separates
% the claimed event from the witness that carries it and the account's
% confidence status.
\newenvironment{accounttimeline}
{\footnotesize
 \begin{longtable}{>{\bfseries\raggedright\arraybackslash}p{0.15\linewidth}
                   >{\raggedright\arraybackslash}p{0.35\linewidth}
                   >{\raggedright\arraybackslash}p{0.26\linewidth}
                   >{\raggedright\arraybackslash}p{0.15\linewidth}}
\toprule
\textbf{Date} & \textbf{Event or act} & \textbf{Witness or edition} & \textbf{Status}\\
\midrule
\endhead}
{\bottomrule\end{longtable}}
\newcommand{\timelineevent}[4]{#1 & #2 & #3 & #4\\}
