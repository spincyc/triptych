\subsection*{Patristic witnesses}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item Origen, \emph{Contra Celsum} IV.23, preserving Celsus's frogs-in-the-marsh mockery of particular providence; Frederick Crombie's public-domain Ante-Nicene Fathers translation, \sourceurl{https://www.newadvent.org/fathers/04164.htm}{online witness}.
\item St.~Irenaeus of Lyons, \emph{Adversus haereses} III.22.1--2 (real flesh from Mary) and V, preface (``become what we are''); Alexander Roberts and William Rambaut's public-domain Ante-Nicene Fathers translation, \sourceurl{https://www.newadvent.org/fathers/0103322.htm}{Book III witness} and \sourceurl{https://www.newadvent.org/fathers/0103500.htm}{Book V preface}.
\item St.~Athanasius of Alexandria, \emph{De incarnatione Verbi} 8, 17, 41--43, on the assumed body, the uncircumscribed Word, and the reply to the ``nobler instrument'' objection; Archibald Robertson's public-domain translation (NPNF, 2nd series, vol.~4), \sourceurl{https://www.newadvent.org/fathers/2802.htm}{online witness}; sections 41--43 verified in a \sourceurl{https://www.documentacatholicaomnia.eu/03d/0295-0373,_Athanasius,_De_incarnatione_verbi_\%5BSchaff\%5D,_EN.pdf}{full-text NPNF copy}.
\item St.~Leo the Great, \emph{Sermo} 74 (De ascensione II), 2, on the visible Redeemer passing into sacramental presence; Charles Lett Feltoe's public-domain NPNF translation, \sourceurl{https://www.newadvent.org/fathers/360374.htm}{online witness}.
\end{itemize}}
