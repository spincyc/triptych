\subsection*{Sacred Scripture}

\subsection*{Liturgical Books and Received Prayers}

\begin{itemize}
\item \textit{Missale Romanum ex decreto Sacrosancti Concilii Tridentini restitutum}, Editio typica (Vatican City, 1962). Source for the Ash Wednesday imposition of ashes (\textit{Memento, homo}) and Introit (\textit{Misereris omnium}) and the Easter Sequence (\textit{Victimae paschali laudes}).
\item \textit{Missale Romanum}, Editio quarta juxta typicam (Tours: Alfred Mame, 1922), p.\ 339. Public domain. Source for the Latin Preface for the Dead (\textit{Praefatio pro Defunctis}).
\item \textit{The Little Office of the Blessed Virgin Mary and the Office of the Dead} (Latin and English). New York: Benziger Brothers, 1915. Public domain. Source for the Office of the Dead (\textit{Placebo}, \textit{Dirige}, and Responsory \textit{Credo quod Redemptor meus vivit}).
\item \textit{Manual of Prayers for the Use of the Catholic Laity}. Prepared and published by order of the Third Plenary Council of Baltimore. New York: Catholic Publication Society, 1889. Public domain. Source for Evening Compline (pp.\ 69--75), the \textit{Anima Christi} (p.\ 103), Holy Viaticum (pp.\ 475--480), Extreme Unction (pp.\ 487--490), Recommendation of a Departing Soul (pp.\ 511--533), and Burial of the Dead (pp.\ 542, 581--585).
\end{itemize}

\subsection*{Councils and Magisterium}

\begin{itemize}
\item Fourth Lateran Council (1215). Constitution 1, \textit{Firmiter credimus}. DS 801.
\item Council of Vienne (1311--1312). Constitution \textit{Fidei catholicae}. DS 902.
\item Pope Benedict XII. Apostolic Constitution \textit{Benedictus Deus} (29 January 1336). DS 1000--1002.
\item Council of Florence (1439). Bull of Union \textit{Laetentur caeli} (6 July 1439). DS 1304--1306.
\item Fifth Lateran Council (1513). Session VIII, Bull \textit{Apostolici Regiminis} (19 December 1513). DS 1440.
\item Council of Trent. Session V (17 June 1546), Decree on Original Sin, Canons 1--2 (DS 1511--1512); Session XIV (25 November 1551), Doctrine on the Sacrament of Extreme Unction, Prooemium, Chapters 1--3, and Canons 1--4 (DS 1694--1700, 1716--1719); Session XXII (17 September 1562), Doctrine on the Sacrifice of the Mass, Chapter 2 and Canon 3 (DS 1743, 1753); Session XXV (3--4 December 1563), \textit{Decree on Purgatory} (DS 1820). Cited by canonical session and Denzinger-Sch\"{o}nmetzer numbering.
\item \textit{The Catechism of the Council of Trent} (\textit{Catechismus Romanus}). Translated by J.\ Donovan (1829). Baltimore: Lucas Brothers, undated printing. Public domain. Cited at Part I, Creed Articles IV, V, VII, and XI, and Part II, On the Eucharist and On Extreme Unction.
\item Pope St.\ Pius X. Apostolic Constitution \textit{Divino afflatu} (1 November 1911). Cited for the 1911 Roman Breviary Psalter reform.
\item Second Vatican Council. Pastoral Constitution on the Church in the Modern World \textit{Gaudium et spes} (7 December 1965), \S 18. Cited by canonical section number.
\item Pope St.\ Paul VI. Apostolic Constitution \textit{Laudis canticum} (1 November 1970), promulgating the 1971 \textit{Liturgia Horarum}; Apostolic Constitution \textit{Sacram unctionem infirmorum} (30 November 1972), revising the postconciliar Latin rite of Anointing of the Sick.
\item \textit{Catechism of the Catholic Church}. 2nd ed.\ (1997). Cited at \S\S 1005--1014, 1021--1022.
\item Denzinger, Heinrich, and Adolf Sch\"{o}nmetzer (DS). \textit{Enchiridion symbolorum, definitionum et declarationum de rebus fidei et morum}. Standard numbering reference.
\end{itemize}

\subsection*{Fathers of the Church}

\begin{itemize}
\item St.\ Ignatius of Antioch. \textit{Epistle to the Romans} 4--7 (shorter recension). In \textit{The Ante-Nicene Fathers}, vol.\ 1, ed.\ Alexander Roberts, James Donaldson, and A.\ Cleveland Coxe. Public domain.
\item St.\ Cyprian of Carthage. \textit{On the Mortality} (\textit{De mortalitate}), \S\S 2--3, 8--12, 14--18, 20, 22, 26. Translated by Ernest Wallis. In \textit{The Ante-Nicene Fathers}, vol.\ 5. Public domain.
\item St.\ Athanasius of Alexandria. \textit{On the Incarnation of the Word} (\textit{De Incarnatione Verbi}), \S\S 8--9, 20--21, 27--29. Translated by Archibald Robertson. In \textit{Nicene and Post-Nicene Fathers}, Second Series, vol.\ 4 (New York, 1892). Public domain.
\item St.\ Cyril of Jerusalem. \textit{Catechetical Lectures} (Mystagogical Catechesis V), 23.9--10. In \textit{Nicene and Post-Nicene Fathers}, Second Series, vol.\ 7, ed.\ Philip Schaff and Henry Wace. Public domain.
\item St.\ Ambrose of Milan. \textit{On the Decease of His Brother Satyrus} (\textit{De excessu fratris Satyri}), Books I--II. Translated by H.\ de Romestin. In \textit{Nicene and Post-Nicene Fathers}, Second Series, vol.\ 10 (1896). Public domain.
\item St.\ John Chrysostom. \textit{Homilies on Philippians}, Hom.\ 3; \textit{Homilies on the First Epistle to the Thessalonians}, Hom.\ 6. In \textit{Nicene and Post-Nicene Fathers}, First Series, vol.\ 13, ed.\ Philip Schaff. Public domain.
\item St.\ Augustine of Hippo. \textit{The City of God} (\textit{De civitate Dei}), I.12--13; XIII.1--2, 4--6, 9--11, 16--17, 19. Translated by Marcus Dods (Edinburgh: T.\ \& T.\ Clark, 1871), collated with E.\ Hoffmann's Latin text, CSEL 40.1 (Vienna, 1899). \textit{Confessions}, Book IX, chs.\ 10--13, translated by J.\ G.\ Pilkington (NPNF, First Series, vol.\ 1). Public domain.
\item St.\ Leo the Great. \textit{Sermons} LIV (On the Passion, III) and LXXI (On the Lord's Resurrection, I). Translated by Charles Lett Feltoe. In \textit{Nicene and Post-Nicene Fathers}, Second Series, vol.\ 12. Public domain.
\item St.\ John of Damascus. \textit{Exposition of the Orthodox Faith} (\textit{De fide orthodoxa}), II.4. Translated by S.\ D.\ F.\ Salmond. In \textit{Nicene and Post-Nicene Fathers}, Second Series, vol.\ 9 (Edinburgh: T.\ \& T.\ Clark, 1898). Public domain.
\end{itemize}

Dublin: James Duffy, 1843. Public domain. Received English prayer witness for Section 10 (Fourth Consideration, Third Point, ``Affections and Prayers,'' pp.\ 68--69).
