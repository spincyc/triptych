\section{References}

\subsection*{Sacred Scripture}

\begin{itemize}
\item \textbf{Douay-Rheims Bible (Challoner Revision).} \emph{The Holy Bible, Translated from the Latin Vulgate} (Challoner revision, Project Gutenberg eBook 1581; public domain). Primary English text for all quoted Scripture passages (Exod 19--20; Deut 5--6; Ps 39, 45, 49, 115; Prov 19; Isa 1; Hos 6; Amos 5; Mic 6; Mal 1; Matt 5, 9, 12, 21, 25, 26; Luke 17, 22; John 4, 6; Rom 1, 11--13; 1 Cor 4, 11; Heb 10, 13; Jas 1; 1 Pet 2; 1 John 5).
\item \textbf{Vulgata Clementina.} \emph{Biblia Sacra Vulgatae Editionis} (Sixto-Clementine text, eBible \texttt{latVUC} verse tables; public domain). Checked Latin text for Exodus, Psalms (Ps 49, 115), Matthew, Luke, and Romans.
\end{itemize}

\subsection*{Patristic, Scholastic, and Spiritual Sources}

\begin{itemize}
\item \textbf{Augustine of Hippo, St.} \emph{The City of God} (\emph{De civitate Dei}), Book X, chs.\ 1, 3, 5--6, 19--20. English translation by Marcus Dods, vol.\ 1 (Edinburgh: T.\ \& T.\ Clark, 1871; Project Gutenberg eBook 45304, public domain). Cross-checked in Latin against Emanuel Hoffmann, ed., \emph{Corpus Scriptorum Ecclesiasticorum Latinorum} (CSEL), vol.\ 40/1 (Vienna: Tempsky, 1899; public-domain text via the CC~BY-SA~4.0 University of Leipzig / OpenGreekAndLatin TEI encoding).
\item \textbf{Augustine of Hippo, St.} \emph{Expositions on the Book of Psalms} (\emph{Enarrationes in Psalmos}), on Psalm 115 (numbered Psalm CXVI in the English edition, \S\S 9--13). Translated in \emph{Nicene and Post-Nicene Fathers}, Series~I, vol.\ 8, edited by Philip Schaff (New York, 1888; tracked CCEL public-domain text).
\item \textbf{Benedict of Nursia, St.} \emph{Regula Benedicti} (\emph{The Rule of St.\ Benedict}), chs.\ 49 and 72. Tracked public-domain Latin text (Wikisource transcription); English rendering of ch.\ 49 is an editorial translation of the tracked Latin text.
\item \textbf{Francis de Sales, St.} \emph{An Introduction to a Devout Life}. English translations: New York: Fr.\ Pustet \& Co.\ (undated public-domain edition; Part~I, chs.\ 1--2; Part~III, chs.\ 9, 23) and London: Rivingtons, 1876 (public-domain edition; Part~IV, ch.\ 11).
\item \textbf{Gregory of Nazianzus, St.} \emph{Oration 30} (\emph{Fourth Theological Oration}), \S 12. Translated by Charles Gordon Browne and James Edward Swallow in \emph{Nicene and Post-Nicene Fathers}, Series~II, vol.\ 7 (New York, 1894; tracked CCEL public-domain text).
\item \textbf{Irenaeus of Lyons, St.} \emph{Against Heresies} (\emph{Adversus Haereses}), Book IV, chs.\ 14.1--3, 17.1--5, 18.1--6. Translated by Alexander Roberts and W.\ H.\ Rambaut in \emph{Ante-Nicene Fathers}, vol.\ 1 (Buffalo, 1885; tracked CCEL public-domain text).
\item \textbf{John Chrysostom, St.} \emph{Homilies on the Gospel of St.\ Matthew}, Homilies 50.3--4 and 67.2--3. Translated by George Prevost, revised by M.\ B.\ Riddle, in \emph{Nicene and Post-Nicene Fathers}, Series~I, vol.\ 10 (New York, 1888; tracked CCEL public-domain text).
\item \textbf{Justin Martyr, St.} \emph{First Apology}, chs.\ 13 and 65--67. Translated by Marcus Dods and George Reith in \emph{Ante-Nicene Fathers}, vol.\ 1 (Buffalo, 1885; tracked CCEL public-domain text).
\item \textbf{Thomas Aquinas, St.} \emph{Summa Theologiae}: \emph{Prima Secundae} (qq.\ 90--91, 94, 100, 106--109, 114; Project Gutenberg eBook 17897), \emph{Secunda Secundae} (qq.\ 17, 58, 80--85, 106; Project Gutenberg eBook 18755), and \emph{Tertia Pars} (qq.\ 18, 48, 84--85; Project Gutenberg eBook 19950). Translated by Fathers of the English Dominican Province (Benziger Brothers, New York, 1920; public domain).
\end{itemize}

\subsection*{Conciliar, Magisterial, Catechetical, and Canonical Sources}

\begin{itemize}
\item \textbf{Catholic Church.} \emph{Catechism of the Catholic Church} (2nd ed., Latin typical text 1997). Cited strictly in doctrinal paraphrase at nn.\ 1440--1460, 2006--2011, and 2095--2109 (no verbatim reproduction).
\item \textbf{Catholic Church.} \emph{Code of Canon Law} (\emph{Codex Iuris Canonici}), promulgated by Pope St.\ John Paul~II in the Apostolic Constitution \emph{Sacrae disciplinae leges} (25~January 1983), with Book~VI reformed by Pope Francis in \emph{Pascite gregem Dei} (23~May 2021). Summarized strictly in original plain-English paraphrase at cc.\ 1, 11, 87 \S 1, 1245, 1246--1248, and revised Book~VI (cc.\ 1311, 1314) for the universal Latin Church (no verbatim reproduction).
\item \textbf{Catholic Church.} \emph{Code of Canons of the Eastern Churches} (\emph{Codex Canonum Ecclesiarum Orientalium}), promulgated by Pope St.\ John Paul~II in the Apostolic Constitution \emph{Sacri canones} (18~October 1990). Cited solely to mark the jurisdictional boundary excluding Eastern Catholic Churches \emph{sui iuris} from the 1983 Latin Code.
\item \textbf{Council of Trent.} \emph{Decree on Justification} (Session~VI, 13~January 1547, chs.\ 8, 14, 16), \emph{Doctrine on the Sacrament of Penance} (Session~XIV, 25~November 1551, chs.\ 1, 8, and Canon~2), and \emph{Doctrine on the Sacrifice of the Mass} (Session~XXII, 17~September 1562, chs.\ 1--2). In \emph{Dogmatic Canons and Decrees} (New York: Devin-Adair, 1912; public domain).
\item \textbf{Council of Trent (\emph{Catechismus Romanus}).} \emph{The Catechism of the Council of Trent} (1566). Translated by James Donovan (Baltimore: Lucas Brothers, 1829; public domain). Cited on the Sacrament of Penance, Satisfaction (pp.\ 176--178, 204, 332), and the Third Commandment (pp.\ 263--265) as dated sixteenth-century catechetical teaching.
\item \textbf{First Vatican Council.} Dogmatic Constitution on the Catholic Faith \emph{Dei Filius} (24~April 1870), ch.\ 3 and Canon III.5. Verified in Latin (historical Vatican text, public domain) and in English translation by Cardinal Henry Edward Manning in \emph{Dogmatic Canons and Decrees} (New York: Devin-Adair, 1912; public domain).
\item \textbf{Francis, Pope.} Apostolic Constitution \emph{Pascite gregem Dei} (23~May 2021; effective 8~December 2021), reforming Book~VI of the 1983 \emph{Code of Canon Law}. Cited for the penal-law boundary of revised Book~VI (cataloged Holy See Latin web state; no verbatim reproduction).
\item \textbf{John Paul~II, Pope St.} Apostolic Letter \emph{Dies Domini} (31~May 1998), nn.\ 46--53. Cited strictly in doctrinal paraphrase (bibliographic reference; no verbatim reproduction).
\item \textbf{Second Vatican Council.} Constitution on the Sacred Liturgy \emph{Sacrosanctum Concilium} (4~December 1963; \emph{AAS} 56 [1964]), nn.\ 7, 10, 14, 47--48, 106; Dogmatic Constitution on the Church \emph{Lumen gentium} (21~November 1964; \emph{AAS} 57 [1965]), nn.\ 10--11, 34; Declaration on Religious Freedom \emph{Dignitatis humanae} (7~December 1965; \emph{AAS} 58 [1966]), nn.\ 1--3. Cited strictly in doctrinal paraphrase (no verbatim reproduction).
\item \textbf{Third Council of Constantinople (Sixth Ecumenical Council, 681).} Definition of the Two Natural Wills and Operations in Christ (Actio XVIII). Cited as quoted in St.\ Thomas Aquinas, \emph{Summa Theologiae} III, q.\ 18, aa.\ 1 and 6.
\end{itemize}
