\subsection*{Ecumenical Councils, Papal Teaching, and Catechism}

\begin{itemize}
\item \textbf{First Vatican Council.} Dogmatic Constitution on the Catholic Faith \emph{Dei Filius} (24 April 1870). Historical Latin text (public domain) paired with Cardinal Henry Edward Manning's English translation in \emph{Dogmatic Canons and Decrees} (New York: Devin-Adair, 1912), pp.~213--238 (public domain). Cited at Chapters 1, 2, and 4; Canons I.2 and II.1.
\item \textbf{Leo XIII, Pope.} Encyclical Letter \emph{Aeterni Patris} (4 August 1879). Historical Latin text (public domain) paired with the English translation in \emph{The Great Encyclical Letters of Pope Leo XIII} (New York: Benziger Brothers, 1903), pp.~34--57 (public domain). Cited on Wisdom 13 and the physical sciences (pp.~38, 55).
\item \textbf{Pius X, Pope St.} Encyclical Letter \emph{Pascendi Dominici Gregis} (8 September 1907). Historical Latin text (public domain) paired with the 1908 English translation appended to \emph{The Programme of Modernism} (London: T.\ Fisher Unwin, 1908), pp.~175--290 (public domain). Cited on Modernist agnosticism, scientific atheism, and vital immanence (pp.~181--185).
\item \textbf{Second Vatican Council.} Pastoral Constitution on the Church in the Modern World \emph{Gaudium et spes} (7 December 1965). Cataloged official Vatican edition (rights reserved; cited in doctrinal paraphrase without verbatim quotation at nos.\ 36, 57).
\item \textbf{Catholic Church.} \emph{Catechism of the Catholic Church}. Official Vatican English edition (rights reserved; cited in doctrinal paraphrase without verbatim quotation at nos.\ 31--39).
\item \textbf{John Paul II, Pope St.} Encyclical Letter \emph{Fides et ratio} (14 September 1998). Official Vatican English edition (rights reserved; cited in doctrinal paraphrase without verbatim quotation at nos.\ 55, 67, 83, 88).
\end{itemize}

\subsection*{Philosophical, Metrological, and Scientific Works Engaged Conceptually}

\begin{itemize}
\item \textbf{Ayer, A.\ J.} \emph{Language, Truth and Logic}. London: Victor Gollancz, 1936; 2nd ed., 1946. Engaged conceptually on strong and weak verifiability criteria and the 1946 revision.
\item \textbf{Church, Alonzo.} Review of A.\ J.\ Ayer, \emph{Language, Truth and Logic} (2nd ed., 1946). \emph{Journal of Symbolic Logic} 14 (1949): 52--53. Engaged conceptually for the formal triviality proof against indirect verifiability.
\item \textbf{Comte, Auguste.} \emph{Cours de philosophie positive}, 6 vols. Paris: Bachelier, 1830--1842. Engaged conceptually on the law of three stages (Lesson 1) and stellar chemistry (Lesson 19).
\item \textbf{Duhem, Pierre.} \emph{La th\'eorie physique: son objet, sa structure}. Paris: Chevalier \& Rivi\`ere, 1906 (English trans.\ \emph{The Aim and Structure of Physical Theory}). Engaged conceptually to distinguish philosophy-of-science instrumentalism from instrumental exclusivism.
\item \textbf{Hahn, Hans, Otto Neurath, and Rudolf Carnap.} \emph{Wissenschaftliche Weltauffassung: Der Wiener Kreis}. Vienna: Artur Wolf Verlag, 1929. Engaged conceptually as the programmatic manifesto of logical empiricism.
\item \textbf{Hawking, Stephen, and Leonard Mlodinow.} \emph{The Grand Design}. New York: Bantam Books, 2010. Engaged conceptually on contemporary scientistic announcements of the death of philosophy and model-dependent realism.
\item \textbf{Hempel, Carl G.} ``Problems and Changes in the Empiricist Criterion of Meaning.'' \emph{Revue Internationale de Philosophie} 4 (1950): 41--63. Engaged conceptually on the failure of sentence-level empiricist criteria of cognitive significance.
\item \textbf{Hume, David.} \emph{An Enquiry Concerning Human Understanding}. London, 1748 (originally published as \emph{Philosophical Essays Concerning Human Understanding}; retitled 1758). Engaged conceptually on Hume's Fork (Sections IV and XII).
\item \textbf{Joint Committee for Guides in Metrology (JCGM).} \emph{International Vocabulary of Metrology---Basic and General Concepts and Associated Terms} (VIM, 3rd ed., JCGM 200:2012) and \emph{Evaluation of Measurement Data---Guide to the Expression of Uncertainty in Measurement} (GUM, JCGM 100:2008). Engaged conceptually on the distinction between indication, measurand, calibration chain, and measurement result.
\item \textbf{Krauss, Lawrence M.} \emph{A Universe from Nothing: Why There Is Something Rather Than Nothing}. New York: Free Press, 2012. Engaged conceptually on the equivocation between a physical quantum vacuum and metaphysical \emph{nihil}.
\item \textbf{Popper, Karl R.} \emph{Logik der Forschung} (Vienna, 1934 [imprint 1935]; English trans.\ \emph{The Logic of Scientific Discovery}, London, 1959) and \emph{Conjectures and Refutations} (London, 1963). Engaged conceptually on falsifiability as scientific demarcation rather than a criterion of meaning.
\item \textbf{Quine, W.\ V.\ O.} ``Two Dogmas of Empiricism.'' \emph{Philosophical Review} 60 (1951): 20--43. Engaged conceptually on the critique of the analytic/synthetic distinction and confirmation holism.
\item \textbf{van Fraassen, Bas C.} \emph{The Scientific Image}. Oxford: Clarendon Press, 1980. Engaged conceptually on constructive empiricism and philosophy-of-science anti-realism.
\end{itemize}
