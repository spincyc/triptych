\section{References}
\label{sec:references}

\subsection{Sacred Scripture}
\begin{itemize}
\item \emph{The Holy Bible, Douay-Rheims Version} (Bishop Richard Challoner revision; public domain). Primary English scriptural text for all verbatim biblical quotations across Old and New Testaments (Gen, Exod, Lev, 2~Kings, Tob, Ps, Prov, Is, Ezech, Joel, Mal; Matt, Mark, Luke, John, Acts, Rom, 1--2~Cor, Gal, Eph, Col, 1--2~Tim, Titus, Heb, James, 1--2~Pet, 1~John, Apoc).
\item \emph{Biblia Sacra Vulgata} (Clementine Vulgate; public domain). Latin scriptural control, including Vulgate verse numbering at John~6:51--59, Is~11:2--3, and Acts~14:22.
\end{itemize}

\subsection{Fathers of the Church and Early Christian Writers}
\begin{itemize}
\item St.~Ignatius of Antioch, \emph{Epistle to the Ephesians} (ch.~20), \emph{Epistle to the Trallians} (chs.~2--3), and \emph{Epistle to the Smyrnaeans} (chs.~7--8), Shorter Recension, trans.\ Alexander Roberts and James Donaldson, in \emph{Ante-Nicene Fathers}, vol.~1 (Buffalo, 1887; public domain).
\item St.~Justin Martyr, \emph{First Apology}, chs.~61 and 65--67, trans.\ Marcus Dods and George Reith, in \emph{Ante-Nicene Fathers}, vol.~1 (Buffalo, 1887; public domain).
\item St.~Irenaeus of Lyons, \emph{Adversus haereses} (\emph{Against Heresies}), IV.18.1--6 and V.2.2--3, trans.\ Alexander Roberts and W.~H.~Rambaut, in \emph{Ante-Nicene Fathers}, vol.~1 (Buffalo, 1887; public domain).
\item St.~Cyprian of Carthage, \emph{De lapsis} (\emph{On the Lapsed}), \S\S16, 28--29, and \emph{Epistle~XII} (Oxford ed.~\emph{Ep.~XVIII}), trans.\ Ernest Wallis, in \emph{Ante-Nicene Fathers}, vol.~5 (public domain).
\item St.~Serapion of Thmuis, \emph{Euchologion} (fourth-century liturgical witness to the blessing of oil for the sick).
\item St.~Cyril of Jerusalem, \emph{Procatechesis}, \emph{Catechetical Lecture~III}, and \emph{Mystagogical Catecheses~I--V} (traditionally ascribed to Cyril; modern scholarship also considers his successor John~II of Jerusalem), trans.\ Edwin Hamilton Gifford, in \emph{Nicene and Post-Nicene Fathers}, Series~2, vol.~7 (public domain).
\item St.~Ambrose of Milan, \emph{De mysteriis} (\emph{On the Mysteries}), III--IV, VII--IX, and \emph{De paenitentia} (\emph{Concerning Repentance}), I--II, trans.\ H.~de~Romestin, in \emph{Nicene and Post-Nicene Fathers}, Series~2, vol.~10 (public domain).
\item St.~John Chrysostom, \emph{Homilies on the Gospel of Saint Matthew}, Hom.~82 (\emph{NPNF}, Series~1, vol.~10); \emph{Homilies on First Corinthians}, Hom.~24 (\emph{NPNF}, Series~1, vol.~12); \emph{Homilies on Ephesians}, Hom.~20, and \emph{Homilies on 1--2~Timothy}, Hom.~13 on 1~Tim and Hom.~1 on 2~Tim (\emph{NPNF}, Series~1, vol.~13; public domain); and \emph{De sacerdotio} (\emph{On the Priesthood}), Book~III.
\item St.~Jerome, patristic maxim on penance as ``a second plank after shipwreck'' (transmitted via St.~Thomas Aquinas, \emph{ST}~III, q.~84, a.~6).
\item Pope St.~Innocent~I, \emph{Si instituta} (\emph{Epistle~25} to Decentius of Gubbio, A.D.~416).
\item St.~Augustine of Hippo, \emph{De civitate Dei} (\emph{The City of God}), X.5--6, XIV.21--23, XV.16, trans.\ Marcus Dods (Edinburgh, 1871; public domain) and ed.\ Emanuel Hoffmann, CSEL~40 (Vienna, 1899--1900; CC~BY-SA~4.0); \emph{De doctrina christiana} (\emph{On Christian Doctrine}), I.2 and III.9 (\emph{NPNF}, Series~1, vol.~2); \emph{In Ioannis Evangelium Tractatus} (\emph{Tractates on the Gospel of John}), Tr.~26.11--13, 49.24, 80.3, 120.2, 121.4 (\emph{NPNF}, Series~1, vol.~7); \emph{Sermon~71} (\emph{NPNF}, Series~1, vol.~6, \emph{Sermon~21}); \emph{De Genesi ad litteram}, IX.7.12; and \emph{De bono coniugali} (\emph{On the Good of Marriage}).
\item Pope St.~Leo the Great, \emph{Sermon~74} (Second Sermon on the Ascension, \S2: \latin{in sacramenta transivit}), trans.\ Charles Lett Feltoe, in \emph{Nicene and Post-Nicene Fathers}, Series~2, vol.~12 (public domain).
\item Pseudo-Dionysius the Areopagite, \emph{De ecclesiastica hierarchia} (\emph{The Ecclesiastical Hierarchy}), ch.~5, Migne PG~3 (public domain).
\item St.~Bede the Venerable, \emph{In Marci evangelium expositio}, Book~II.6 (on Mark~6:12--13) and on Mark~10:2--12 (PL~92), with English patristic extracts in St.~Thomas Aquinas, \emph{Catena Aurea in Marcum} (Oxford, 1842; public domain; includes Pseudo-Chrysostom [Victor of Antioch] and Theophylact on Mark~6:13).
\item St.~John of Damascus, \emph{De fide orthodoxa} (\emph{Exposition of the Orthodox Faith}), IV.9 and IV.13, trans.\ S.~D.~F.~Salmond, in \emph{Nicene and Post-Nicene Fathers}, Series~2, vol.~9 (Edinburgh, 1898; public domain).
\end{itemize}

\subsection{Medieval and Scholastic Doctors}
\begin{itemize}
\item Anonymous (eleventh century), \emph{De vera et falsa poenitentia}, ch.~10, n.~25 (medieval treatise formerly circulated under Augustine's name, transmitted via Gratian and Peter Lombard).
\item Gratian, \emph{Decretum}, C.~33, q.~3 (\emph{De poenitentia}), d.~1, c.~88.
\item Peter Lombard, \emph{Sententiae in IV libris distinctae}, Book~IV, dd.~3, 17, 23, 24, 26 (Grottaferrata, 1981; restricted critical edition, summarized without verbatim reproduction).
\item St.~Thomas Aquinas, \emph{Summa theologiae}, Pars~Tertia, qq.~60--90, trans.\ Fathers of the English Dominican Province (1920; public domain); \emph{Scriptum super Sententiis}, Book~IV, and \emph{Supplementum Tertiae Partis}, qq.~1--68 (doctrinal synthesis of Penance completion, Extreme Unction, Orders, and Matrimony).
\end{itemize}
