\section{Sources for Further Reading}
\begingroup\small
\ifdefined\TriptychPrintEdition
\interlinepenalty=10000
\else
\fi
\begin{itemize}[leftmargin=1.25em,itemsep=0.2em]
\item Sacred Scripture: Matthew 5:21--30; 7:13--27; 13:24--50; 25:31--46; Mark 9:40--47; Luke 15:11--24; 16:13--31; John 5:28--29; 1 Timothy 2:1--6. English text: Douay--Rheims Bible, Challoner revision. The sentence from Matthew 25:41 is quoted; other passages are summarized. Mark follows this witness's numbering.
\item \emph{Catechism of the Catholic Church}, 1021, 1030--1038, 1451--1460, 1735, 1854--1863, 2284--2287; especially \href{https://www.vatican.va/archive/ENG0015/__P2O.HTM}{1033--1037} on hell, \href{https://www.vatican.va/archive/ENG0015/__P4D.HTM}{1451--1460} on repentance and confession, and \href{https://www.vatican.va/archive/ENG0015/__P80.HTM}{2284--2287} on scandal.
\item Fourth Lateran Council, \emph{Firmiter credimus} (1215), profession of the resurrection and eternal destinies, DS 801. Historical Latin text checked in the registered transcription; English account here is a paraphrase.
\item \emph{Martyrdom of Polycarp} 9--12, especially 11: the Church of Smyrna's account, in the Roberts--Donaldson translation, \href{https://ccel.org/ccel/schaff/anf01.iv.iv.xi.html}{\emph{Ante-Nicene Fathers} I}, complete CCEL electronic witness.
\item St.~Cyprian, \emph{Ad Demetrianum} 23--25, especially 24--25; Robert Ernest Wallis translation, \href{https://ccel.org/ccel/cyprian/epistles/anf05.iv.v.v.html}{\emph{Ante-Nicene Fathers} V}, complete CCEL electronic witness.
\item St.~Cyril of Jerusalem, \href{https://ccel.org/ccel/schaff/npnf207.ii.xxii.html}{\emph{Catechetical Lecture} 18.18--20}; Edwin Hamilton Gifford's revised translation, \emph{Nicene and Post-Nicene Fathers}, second series, VII, retained CCEL text.
\item St.~John Chrysostom, \emph{Homilies on Matthew} 23.9--12; George Prevost translation, revised by M.~B.~Riddle, \href{https://ccel.org/ccel/s/schaff/npnf110/cache/npnf110.txt}{\emph{Nicene and Post-Nicene Fathers}, first series, X}, retained CCEL text.
\item St.~Augustine, \emph{City of God} XIII.2; XXI.9--10, 23; Marcus Dods translation (Edinburgh: T.~\& T.~Clark, 1871), retained transcriptions of \href{https://www.gutenberg.org/ebooks/45304}{volume I} and \href{https://www.gutenberg.org/ebooks/45305}{volume II}.
\item St.~Augustine, \emph{Confessions} VIII.5.10--12; VIII.8.19--12.30; J.~G.~Pilkington translation, \href{https://ccel.org/ccel/s/schaff/npnf101/cache/npnf101.txt}{\emph{Nicene and Post-Nicene Fathers}, first series, I}, complete retained CCEL electronic witness. The reading distinguishes Augustine's account from the volume's editorial notes.
\item St.~Thomas Aquinas, \emph{Summa theologiae} I--II, \href{https://www.newadvent.org/summa/2087.htm\#article3}{q.~87, a.~3}, body and replies; English Dominican translation, second revised edition (1920), consulted through New Advent.
\item St.~Thomas Aquinas, \emph{Summa theologiae} II--II, q.~19, aa.~1--2, 4, 6--7, 9--10; q.~43, aa.~1--2, 7; English Dominican translation in \href{https://www.gutenberg.org/ebooks/18755}{Project Gutenberg 18755}, complete electronic witness (2006). The transcription misnumbers q.~19, aa.~7--8; canonical article numbers are checked against the corresponding titles in New Advent. These readings concern fear, charity, and scandal.
\item St.~Ignatius of Loyola, \emph{Spiritual Exercises}, no.~44 (General Confession with Communion); First Week, Fifth Exercise, nos.~65--71; Elder Mullan translation (New York: P.~J.~Kenedy \& Sons, 1914), unnumbered general-confession leaf and pp.~44--46, \href{https://archive.org/details/TheSpiritualExercisesOfStIgnatius}{scanned printing} and complete text.
\item St.~Francis de Sales, \emph{Introduction to the Devout Life}, I.15, 19; V.1--3, 15; \emph{A New Translation}, translator unnamed (London, Oxford, and Cambridge: Rivingtons, 1876), pp.~41--42, 51--53, 346--353, 370--372, \href{https://archive.org/details/introductiontod00salegoog}{scanned printing} and complete text.
\item St.~Alphonsus Liguori, \emph{Preparation for Death}, XVI.1--3; XVII.1, including its prayers; XXVI.1, 3; XXVII.1; XXXI.1; translated ``by a Catholic Clergyman'' (Dublin: James Duffy, 1843), pp.~195--209, 314, 323--331, 375--379. The \href{https://archive.org/details/preparation_for_death_1893-st_alphonsus_m_liguori}{archive item} is labeled 1893, but the inspected title page reads 1843. The closing prayer reproduces one complete sentence from p.~209.
\item St.~Teresa of Avila, \emph{Life} 32.1--11, especially 6--11; David Lewis translation, third enlarged edition, edited by Benedict Zimmerman (London: Thomas Baker; New York: Benziger Bros., 1904), pp.~286--290, \href{https://archive.org/details/lifeofteresajesu00tereuoft}{scanned printing} and complete text.
\end{itemize}\endgroup
