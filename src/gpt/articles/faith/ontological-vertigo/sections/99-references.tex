\subsection*{Scripture and authoritative Catholic witnesses}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item Sacred Scripture: Wis 11:24--26; John 1:1--3; Acts 17:24--28; Rom 11:36; Col 1:16--17; Heb 1:3. Short English wording is used as a working rendering; Scripture and translations retain their own status.
\item First Vatican Council, dogmatic constitution \emph{Dei Filius} (24 April 1870), chs.~1 and 3 and canons 1.3--5: the simple God really distinct from the world, free creation from nothing, divine non-need, providence, and the creature's total dependence; \sourceurl{https://www.vatican.va/archive/hist_councils/i-vatican-council/documents/vat-i_const_18700424_dei-filius_la.html}{official Latin}.
\item \emph{Catechism of the Catholic Church} 285, 290--301, 306--308, especially 300--301 on transcendence, intimate presence, and present sustaining action, and 306--308 on real secondary causality; \sourceurl{https://www.vatican.va/content/catechism/en/part_one/section_two/chapter_one/article_1/paragraph_4_the_creator.html}{official English}.
\end{itemize}}

\subsection*{Ancient philosophical antecedents}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item Plato, \emph{Republic} VI, 508e--509b, on the Good as source of intelligibility and being; \emph{Timaeus} 27d--29b, on becoming, cause, and the cosmic craftsman. Benjamin Jowett's public-domain English was used as a working witness at the \sourceurl{https://classics.mit.edu/Plato/republic.7.vi.html}{Internet Classics Archive} and \sourceurl{https://classics.mit.edu/Plato/timaeus.html}{Timaeus text}. The article does not identify these doctrines with Christian creation.
\item Aristotle, \emph{Metaphysics} XII.6--7, 1071b3--1072b30, on actuality, the unmoved mover, and the dependence of heaven and nature; W.~D. Ross's public-domain working English, \sourceurl{https://classics.mit.edu/Aristotle/metaphysics.12.xii.html}{Internet Classics Archive}. See also \emph{Physics} VIII.5--6.
\end{itemize}}

\subsection*{Patristic and late-patristic witnesses}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item St.~Irenaeus of Lyons, \emph{Against Heresies} II.34.2--4, on God alone unbegun, created duration according to divine will, and life by participation; Alexander Roberts and William Rambaut's 1885 ANF working English, \sourceurl{https://www.newadvent.org/fathers/0103234.htm}{online witness}.
\item St.~Athanasius of Alexandria, \emph{On the Incarnation of the Word} 3--5, on creation from nothing through the Word and creaturely mutability, \sourceurl{https://www.newadvent.org/fathers/2802.htm}{NPNF working English}; \emph{Against the Heathen} 40--42, on the Word ordering and preserving the created whole, \sourceurl{https://www.newadvent.org/fathers/2801.htm}{working English}.
\item St.~Augustine of Hippo, \emph{Confessions} I.2.2; III.6.11; VII.1.1--2 and 10.16--11.17; XI.4.6--9.11, on existence through divine presence, inwardness and transcendence, rejection of spatial extension, mutable participation, creation in the Word, and \latin{inhorresco et inardesco}; \sourceurl{https://www.newadvent.org/fathers/110101.htm}{Book I}, \sourceurl{https://www.newadvent.org/fathers/110103.htm}{Book III}, \sourceurl{https://www.newadvent.org/fathers/110107.htm}{Book VII}, \sourceurl{https://www.newadvent.org/fathers/110111.htm}{Book XI}.
\item St.~Gregory Nazianzen, \emph{Oration} 28.4--6, on knowing God's existence from created order while failing to comprehend the divine nature; \sourceurl{https://www.newadvent.org/fathers/310228.htm}{NPNF working English}.
\item Pseudo-Dionysius, \emph{Divine Names} V.1--8, on the superessential cause from whom beings receive and preserve existence; \emph{Mystical Theology} I.1--3, on affirmation, negation, and unknowing. John Parker, trans., \emph{The Works of Dionysius the Areopagite}, vol.~1 (London: James Parker, 1897), \sourceurl{https://ccel.org/ccel/dionysius/works/works.toc.html}{public-domain working edition}.
\end{itemize}}

\subsection*{Thomistic framework}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item St.~Thomas Aquinas, \emph{Summa theologiae} I, q.~3, a.~4 and q.~4, a.~2 (God as his existence and subsistent being); q.~13, a.~7 (asymmetry of Creator--creature relation); q.~44, a.~1, including ad 2 (participated being, universal causality, and necessity from another); q.~45, aa.~1--3, 5 (creation and its real relation in the creature); q.~46, a.~2 (temporal beginning not naturally demonstrated); q.~104, aa.~1--2 (conservation); q.~105, aa.~4--5 (divine action in real secondary causes); I--II, q.~79, a.~2 (God causes the act's actuality, not the defect of sin).
\item Aquinas, \emph{De ente et essentia}, ch.~4, on created essence and existence and the first cause whose essence is existence; \emph{De potentia}, q.~3, a.~7 and q.~5, a.~1; \emph{Summa contra Gentiles} II.15--21 and III.65--70, for creation, conservation, and the operations of secondary causes.
\item Latin loci checked in the \sourceurl{https://www.corpusthomisticum.org/iopera.html}{Corpus Thomisticum} Leonine-text corpus; the \sourceurl{https://www.newadvent.org/summa/}{New Advent Summa} was used as a working public English index. Work, question, article, chapter, and reply control over either site's wording.
\end{itemize}}

\subsection*{Mystical and later witnesses}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item St.~Bernard of Clairvaux, \emph{Sermons on the Song of Songs} 83.4--6, on the creature's unequal yet whole return of love toward its source; Latin checked in \emph{Sancti Bernardi Opera}, vol.~2, 300--302, with a \sourceurl{https://la.wikisource.org/wiki/Sermones_in_Canticum_Canticorum/Sermo_83}{stable online Latin witness}; Samuel J.~Eales's 1895 English used only as a public-domain aid.
\item Julian of Norwich, \emph{Revelations of Divine Love}, long text, ch.~5, on the created whole as made, loved, and kept. Grace Warrack's 1901 public-domain working English, \sourceurl{https://ccel.org/ccel/julian/revelations.all.html}{CCEL edition}; the image is treated as mystical apprehension, not demonstration.
\item St.~John of the Cross, \emph{Ascent of Mount Carmel} II.5.3--4, distinguishing God's sustaining substantial presence in every creature from transformative union by love. David Lewis, trans., Benedict Zimmerman, rev., \emph{The Ascent of Mount Carmel} (London: Thomas Baker, 1906), public-domain working edition, \sourceurl{https://archive.org/details/TheAscentOfMountCarmel}{Internet Archive scan}.
\item Ren\'e Descartes, \emph{Meditations on First Philosophy}, Third Meditation, on present conservation and the conceptual distinction between preservation and creation in the respect argued; public-domain Elizabeth S.~Haldane translation, \sourceurl{https://en.wikisource.org/wiki/Meditations_on_First_Philosophy/Meditation_III}{online witness}.
\end{itemize}}
