\subsection*{Authoritative Catholic witnesses}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item First Vatican Council, dogmatic constitution \emph{Dei Filius} (24 April 1870), chs.~1 and 3 and canons 1.3--5: the simple God really distinct from the world, free creation from nothing, divine non-need, providence, and the creature's total dependence; \sourceurl{https://www.vatican.va/archive/hist_councils/i-vatican-council/documents/vat-i_const_18700424_dei-filius_la.html}{official Latin}.
\item \emph{Catechism of the Catholic Church} 301 and 306--308, on sustaining action and genuine secondary causality; \sourceurl{https://www.vatican.va/content/catechism/en/part_one/section_two/chapter_one/article_1/paragraph_4_the_creator.html}{official English}.
\end{itemize}}

\subsection*{Ancient philosophical antecedents}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item Plato, \emph{Republic} VI, 508e--509b, on the Good as source of intelligibility and being; \emph{Timaeus} 27d--29b, on becoming, cause, and the cosmic craftsman. Benjamin Jowett's public-domain English was used as a working witness at the \sourceurl{https://classics.mit.edu/Plato/republic.7.vi.html}{Internet Classics Archive} and \sourceurl{https://classics.mit.edu/Plato/timaeus.html}{Timaeus text}. The article does not identify these doctrines with Christian creation.
\item Aristotle, \emph{Metaphysics} XII.6--7, 1071b3--1072b30, on actuality, the unmoved mover, and the dependence of heaven and nature; W.~D. Ross's public-domain working English, \sourceurl{https://classics.mit.edu/Aristotle/metaphysics.12.xii.html}{Internet Classics Archive}. See also \emph{Physics} VIII.5--6.
\end{itemize}}

\subsection*{Patristic and late-patristic witnesses}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item St.~Irenaeus of Lyons, \emph{Against Heresies} II.34.2--4, on God alone unbegun, created duration according to divine will, and life by participation; Alexander Roberts and William Rambaut's 1885 ANF working English, \sourceurl{https://www.newadvent.org/fathers/0103234.htm}{online witness}.
\item St.~Athanasius of Alexandria, \emph{On the Incarnation of the Word} 3--5, on creation from nothing through the Word and creaturely mutability, \sourceurl{https://www.newadvent.org/fathers/2802.htm}{NPNF working English}; \emph{Against the Heathen} 40--42, on the Word ordering and preserving the created whole, \sourceurl{https://www.newadvent.org/fathers/2801.htm}{working English}.
\item St.~Augustine of Hippo, \emph{Confessions} I.2.2; III.6.11; VII.1.1--2, 10.16--11.17, 17.23, and 20.26--21.27; XI.9.11, on existence through divine presence, inwardness and transcendence, the changeable mind's judgment by unchangeable truth, seeing without yet possessing the way, and \latin{inhorresco et inardesco}; \emph{On the Trinity} VIII.2.3, on the mind's first flash of Truth and return to familiar images. Public-domain NPNF working English: \sourceurl{https://www.newadvent.org/fathers/110107.htm}{\emph{Confessions} VII}, \sourceurl{https://www.newadvent.org/fathers/110111.htm}{\emph{Confessions} XI}, and \sourceurl{https://www.newadvent.org/fathers/130108.htm}{\emph{On the Trinity} VIII}.
\item St.~Gregory Nazianzen, \emph{Oration} 28.2--6, on longing and fear before the theological cloud, the failed attempt to grasp God directly, and knowledge of God's existence from created order without comprehension of the divine nature; \sourceurl{https://www.newadvent.org/fathers/310228.htm}{NPNF working English}.
\item Pseudo-Dionysius, \emph{Divine Names} V.1--8, on the superessential cause from whom beings receive existence; \emph{Mystical Theology} I.1--3, on affirmation, negation, and unknowing. John Parker, trans., \emph{The Works of Dionysius the Areopagite}, vol.~1 (London: James Parker, 1897), \sourceurl{https://ccel.org/ccel/dionysius/works/works.toc.html}{public-domain working edition}.
\end{itemize}}

\subsection*{Thomistic framework}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item St.~Thomas Aquinas, \emph{Summa theologiae} I, q.~2, a.~3 (the First Way); q.~3, a.~4, q.~4, a.~2, q.~6, aa.~1--2, q.~13, a.~5, q.~14, aa.~1 and 8, and q.~19, aa.~1, 3--4 (subsistent being, goodness, analogy, intellect, free will, and willing creatures without need); q.~20, aa.~1--2 (divine love and its causation of created goodness); q.~21, a.~1, especially ad 2--3, and a.~4, ad 4 (divine justice, derivative created dues, and creation without an antecedent creaturely claim); q.~32, a.~1 (Trinity not demonstrable by natural reason).
\item Aquinas, \emph{Summa theologiae} I, q.~13, a.~7 (Creator--creature relational asymmetry); q.~44, aa.~1 and 4, q.~45, aa.~1--3, 5, and q.~46, a.~2 (participated being, creation, its finality, and the temporal-beginning limit); q.~83, a.~1, ad 3, q.~104, aa.~1--2, and q.~105, aa.~4--5 (freedom, conservation, and secondary causes); I--II, q.~10, a.~4 and q.~79, a.~2 (divine motion without coercion and the defect of sin); II--II, q.~129, a.~3, ad 4, and q.~161, aa.~1--5 (magnanimity and humility).
\item Aquinas, \emph{De ente et essentia}, ch.~4, on created essence and existence and the first cause whose essence is existence; \emph{De potentia}, q.~3, a.~7 and q.~5, a.~1; \emph{Summa contra Gentiles} II.15--21 and III.65--70, for creation, conservation, and secondary causes.
\item Latin loci checked in the \sourceurl{https://www.corpusthomisticum.org/iopera.html}{Corpus Thomisticum} Leonine-text corpus; the \sourceurl{https://www.newadvent.org/summa/}{New Advent Summa} was used as a working public English index. Work, question, article, chapter, and reply take precedence over either site's wording.
\end{itemize}}

\subsection*{Later witnesses}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item Ren\'e Descartes, \emph{Meditations on First Philosophy}, Third Meditation, on present conservation and the conceptual distinction between preservation and creation in the respect argued; public-domain Elizabeth S.~Haldane translation, \sourceurl{https://en.wikisource.org/wiki/Meditations_on_First_Philosophy/Meditation_III}{online witness}.
\item C.~S. Lewis, \emph{Mere Christianity}, III.8, ``The Great Sin,'' on genuine humility as freedom from self-occupation. The point is paraphrased; no Lewis wording is reproduced. The common aphorism about ``thinking of yourself less'' is not attributed to Lewis; see the \sourceurl{https://www.cslewis.org/aboutus/faq/quotes-misattributed/}{C.~S. Lewis Foundation attribution record}.
\end{itemize}}
