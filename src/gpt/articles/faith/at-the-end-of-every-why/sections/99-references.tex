\subsection*{Councils and authoritative synthesis}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item Fourth Lateran Council, \emph{Firmiter credimus} (1215), DS 800--802: one, simple Trinitarian God, Creator of visible and invisible things from nothing; the distinction of Persons does not divide the essence. The compact English citation in CCC 202 is governed by the conciliar text.
\item First Vatican Council, dogmatic constitution \emph{Dei Filius} (1870), ch.~1 and canons 1.1--5, DS 3001--3003, 3021--3025: one true, living, simple God; free creation of all things from nothing, distinct from the divine substance and ordered to divine goodness; \sourceurl{https://www.vatican.va/archive/hist_councils/i-vatican-council/documents/vat-i_const_18700424_dei-filius_la.html}{official Latin text}.
\item \emph{Catechism of the Catholic Church}, nos.~198--231, especially 202--213 and 221 (one God, the revealed name, received being, truth, and love), \sourceurl{https://www.vatican.va/content/catechism/en/part_one/section_two/chapter_one/article_1/paragraph_1_i_believe_in_god.index.html}{official English}; nos.~232--267 (Trinity); 279--301 (creation from nothing, freedom, the Word, and divine goodness).
\item \emph{Catechism of the Catholic Church}, nos.~464--469, especially 468--469 (the one divine Person of the Son as subject of the acts and sufferings of his assumed human nature, without confusion or change), \sourceurl{https://www.vatican.va/content/catechism/en/part_one/section_two/chapter_two/artcile_3.html}{official English}.
\end{itemize}}

\subsection*{Patristic and late-patristic witnesses}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item St.~Athanasius of Alexandria, \emph{On the Incarnation of the Word} 3--5, 20, 42, 54, on creation from nothing through the good God's Word, present dependence, Incarnation, and renewal; \sourceurl{https://www.newadvent.org/fathers/2802.htm}{NPNF working English}. \emph{Orations Against the Arians} I.14, 19, on the eternal, uncreated Word proper to the Father; \sourceurl{https://www.newadvent.org/fathers/28161.htm}{working English}.
\item St.~Gregory Nazianzen, \emph{Oration} 28.4--6, on knowing that God is while failing to comprehend the divine nature, \sourceurl{https://www.newadvent.org/fathers/310228.htm}{working English}; \emph{Oration} 45.3, on God as beyond time and creaturely comprehension, drawing through wonder and desire, \sourceurl{https://www.newadvent.org/fathers/310245.htm}{working English}.
\item St.~Augustine of Hippo, \emph{Confessions} III.6.11; VII.10.16--11.17; XI.5.7--9.11, on divine inwardness and transcendence, the unchangeable Light, creaturely participation, creation in the Word, dread, and desire; \sourceurl{https://www.newadvent.org/fathers/110103.htm}{Book III}, \sourceurl{https://www.newadvent.org/fathers/110107.htm}{Book VII}, \sourceurl{https://www.newadvent.org/fathers/110111.htm}{Book XI}.
\item Augustine, \emph{De Trinitate} VIII.2.3--3.5 and VIII.7.10--10.14, on mutable goods, Good itself, true charity, and lover--beloved--love as a beginning of Trinitarian inquiry, \sourceurl{https://www.newadvent.org/fathers/130108.htm}{working English}; XV.17.27--18.32, on Gift and Love, \sourceurl{https://www.newadvent.org/fathers/130115.htm}{working English}; \emph{Homilies on the First Epistle of John} VII.9--10, on the unseen God and visible works of charity, \sourceurl{https://www.newadvent.org/fathers/170207.htm}{working English}.
\item Pseudo-Dionysius, \emph{Divine Names} IV.1, 7, 10, 13, on the Good as cause and end and on love's unitive motion; \emph{Mystical Theology} I.1--3 and II.1, on affirmation, negation, and luminous unknowing. Checked in John Parker, trans., \emph{The Works of Dionysius the Areopagite}, vol.~1 (London: James Parker, 1897), \sourceurl{https://ccel.org/ccel/dionysius/works/works.toc.html}{public-domain working edition}. The author is identified according to modern convention rather than equated with the convert of Acts 17.
\end{itemize}}

\subsection*{Thomistic framework}

{\fontsize{8.6}{9.6}\selectfont
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0.05em]
\item St.~Thomas Aquinas, \emph{Summa theologiae} I, q.~2, aa.~1--3 (demonstration from effects, act and potency, ordered movers and causes, contingency); q.~3, especially aa.~1--5, 7--8 (pure act, essence and existence, simplicity, and exclusion from a genus); q.~4, aa.~2--3 (subsistent being and creaturely likeness); qq.~5--6 (being and goodness, participated goodness); q.~8, aa.~1, 3 (intimate causal presence); q.~11, aa.~3--4 (divine unity); q.~13, aa.~4--5, 11 (analogical names and ``He Who Is'').
\item Aquinas, ST I, q.~19, aa.~2--4 (God's goodness, free willing of creatures, and non-necessity of creation); q.~20, aa.~1--2 (love in God and divine love as the cause of created goodness); q.~27, aa.~1--4 (processions of Word and Love); q.~29, a.~4 (divine Persons as subsistent relations); q.~32, a.~1 (Trinity not demonstrable by natural reason); q.~34, aa.~1--2 (Word); q.~37, aa.~1--2 (Love as essential and personal name); q.~43, aa.~1--2 (missions and eternal processions); q.~44, aa.~1--4 (universal causality and participation); q.~45, aa.~1--3, 5--7 (creation, nothing, and the Trinity); q.~46, a.~2 (temporal beginning not demonstrated); q.~104, a.~1 (conservation as continued causation of being).
\item Aquinas, \emph{De ente et essentia}, ch.~4, on caused existence and the first cause whose essence is existence; \emph{Summa contra Gentiles} I.13, 16, 22, 37--40, 81, 91, and II.15--17, 21, 23, for fuller parallel arguments.
\item Latin loci checked in the \sourceurl{https://www.corpusthomisticum.org/iopera.html}{Corpus Thomisticum} Leonine-text corpus; \sourceurl{https://www.newadvent.org/summa/}{New Advent} used as a working public English index. Citations by work, question, article, and chapter govern over either site's wording.
\end{itemize}}
