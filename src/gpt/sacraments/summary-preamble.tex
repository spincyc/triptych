% Shared one-page sacrament-summary architecture.  This fragment is loaded by
% the reference treatise and by ritual-Mass appendices so that their summaries
% cannot drift into competing formats.
\newcommand{\sacramentSummaryKicker}{}

\newcommand{\sacramentSummaryStart}[3]{%
  \newpage
  \begingroup
  \footnotesize
  \setlength{\parskip}{0.32em}%
  \begin{center}
    \sacramentSummaryKicker
    {\Large\bfseries #1 at a Glance}\\[0.15em]
    {\small #2}\\[0.15em]
    {\footnotesize\itshape #3}
  \end{center}
  \vspace{-0.35em}
}

\newcommand{\sacramentSummaryEnd}{%
  \par\endgroup
  \newpage
}
