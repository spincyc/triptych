\begin{courseunit}{P1}
\chapter{P1 worksheet family: textual prosody}

\begin{worksheet}{P.1}{A}{P1-A}{Syllables and evidence}{Separate syllabification from quantity and quantity from conjecture.}{Begin here before attempting a metrical scheme.}
\Exercise{Divide \Latin{aqua, caelum, terra, agnus, eundem} into syllables.}{\Latin{a-qua, cae-lum, ter-ra, ag-nus, e-un-dem}. The diphthong \Latin{ae} remains one syllabic nucleus; \Latin{qu} functions as one consonantal unit; doubled or successive consonants normally divide between syllables.}
\Exercise{For each statement write \emph{nature}, \emph{position}, or \emph{neither}: \Latin{ae} is long; a vowel stands before \Latin{rr}; a line ends; a vowel stands before one consonant.}{\Latin{ae}: nature; vowel before \Latin{rr}: position; line ending: neither---the final syllable is metrically anceps; vowel before one consonant: neither without further lexical evidence.}
\Exercise{Explain why a syllable long by position need not contain a naturally long vowel.}{Position is a metrical property of the syllable created by the following consonantal environment. Natural quantity belongs to the vowel or diphthong itself. The two kinds of evidence can yield the same metrical length for different reasons.}
\Exercise{In \Latin{multum ille}, mark the ordinary elision environment and state what is elided.}{Final \Latin{-um} stands before initial \Latin{i-}; the final vowel plus \Latin{m} is available for elision. The written words remain unchanged in a diplomatic transcription.}
\LongExercise{Make a five-column ledger for the preceding words: syllable, proposed quantity, reason, metrical position, uncertainty.}{A sound ledger records only supported quantities, names nature or position as the reason, leaves the metrical-position column open until a scheme is tested, and marks unresolved short vowels rather than guessing them.}
\end{worksheet}

\begin{worksheet}{P.2}{B}{P1-B}{Phaedrus before scansion}{Recover grammar before fitting a line to meter.}{Use the verified Phaedrus opening printed in the packet.}
\PassageExercise{Ad rivum eundem lupus et agnus venerant, siti compulsi. Superior stabat lupus.}{Identify the two finite-verb spines and account for \Latin{eundem, siti compulsi, superior}.}{The spines are \Latin{lupus et agnus venerant} and \Latin{lupus stabat}. \Latin{eundem} is accusative masculine singular agreeing with \Latin{rivum}; \Latin{siti} is ablative singular of cause; \Latin{compulsi} is nominative masculine plural agreeing with both animals; \Latin{superior} is nominative masculine singular predicative of the wolf.}
\Exercise{Put the passage into a neutral prose order without calling the result a variant.}{One project reconstruction is \Latin{Lupus et agnus, siti compulsi, ad eundem rivum venerant. Lupus superior stabat.} It is an analytic aid, not Phaedrus' text.}
\Exercise{What contrast is prepared by \Latin{eundem} and \Latin{superior}?}{Both animals come to the same stream, but the wolf occupies the higher position. The spatial inequality is textual; its moral force belongs to interpretation and must be argued from the continuing fable.}
\Exercise{Why should the printed prologue's \Latin{versibus senariis} lead to a test rather than an assumed scansion?}{The verified prologue identifies the expected medium, but each fable line must still be syllabified and its quantities checked. A source label does not license invented quantities or ignored elision.}
\LongExercise{Annotate the excerpt in three layers: grammar, quantity evidence, interpretation.}{The layers should remain visibly distinct. Grammar identifies forms and relations; quantity records reasons and uncertainty; interpretation may discuss shared need and unequal position but may not be presented as a manuscript reading.}
\end{worksheet}

\begin{worksheet}{P.3}{C}{P1-C}{Elision, anceps, and scheme testing}{Keep textual form distinct from metrical realization.}{Use after P1-A and P1-B when the evidence categories are clear.}
\Exercise{Which phrase presents an ordinary elision environment: \Latin{saevae memorem}, \Latin{multum ille}, \Latin{Troiae qui}, or \Latin{primus ab}? Explain each boundary.}{\Latin{multum ille}: final vowel plus \Latin{m} before a vowel. The others begin the next word with a consonantal sound: \Latin{m-, qu-, ab}'s initial vowel is preceded by final \Latin{s} in \Latin{primus}, so no vowel-to-vowel boundary occurs there.}
\Exercise{Define \emph{anceps} at line end without saying that the vowel's natural quantity changes.}{The final syllable may fill the last metrical position whether naturally long or short. The metrical license concerns the position, not a rewritten lexical quantity.}
\Exercise{A proposed scan works only if a known short vowel is called long by nature. What should happen?}{Reject or revise the scan. Recheck syllable division, positional length, elision, the edition's reading, and the proposed scheme; do not change a known quantity to rescue the pattern.}
\Exercise{A stop plus liquid follows a vowel. What should the ledger say before the line is solved?}{Mark the environment explicitly and test both treatments where the verse permits. It deserves separate examination rather than an automatic two-consonant decision.}
\LongExercise{Write a short protocol for testing an unidentified line.}{Model order: preserve the text; divide syllables; mark secure natural quantities; mark positional environments; note elision; leave uncertainties open; compare plausible schemes; reject any scheme requiring false forms or quantities; record the remaining ambiguity.}
\end{worksheet}

\begin{worksheet}{P.4}{D}{P1-D}{Cumulative textual-prosody record}{Explain every mark and every limit.}{Use a fresh line from Phaedrus after the earlier sheets are dependable.}
\PassageExercise{Longeque inferior agnus. Tunc fauce improba latro incitatus iurgii causam intulit.}{Parse the comparative, apposition, participle, and genitives before marking quantity.}{\Latin{inferior} is nominative masculine singular predicative of \Latin{agnus}; \Latin{latro} is the nominative subject and a characterizing renaming of the wolf; \Latin{incitatus} is a perfect passive participle agreeing with \Latin{latro}; \Latin{iurgii} is genitive with \Latin{causam}; \Latin{fauce improba} is ablative and characterizes the utterance or impulse.}
\Exercise{Identify one place where line order contributes more than neutral prose order.}{The opposed placements \Latin{superior} and \Latin{inferior}, followed by the delayed noun \Latin{agnus}, sharpen the spatial contrast; delayed \Latin{latro} then renames the wolf after his motivation is introduced. Other text-based observations are possible.}
\Exercise{State the difference between a source transcription, a prose reconstruction, and a scansion display.}{The transcription records received wording within declared normalization; the reconstruction is project analysis of syntax; the scansion display maps syllables to a proposed metrical scheme. None may silently replace another.}
\Exercise{List four reasons a mark may remain uncertain.}{Unrecorded natural quantity; ambiguous syllabification or consonantal treatment; possible elision; variant reading; more than one metrical resolution; any four genuinely distinct reasons.}
\LongExercise{Prepare a complete P1 record for one line: transcription, syllables, quantity ledger, proposed scheme, grammar, prose order, translation, and limits.}{A complete record lets another reader reconstruct every decision, labels project additions, and leaves unsupported quantities or interpretations unresolved.}
\end{worksheet}

\begin{courseunit}{P5}
\chapter{P5 worksheet family: Christian poetics}

\begin{worksheet}{P.17}{A}{P5-A}{Five distinct evidence questions}{Do not let witness, authorship, meter, use, and source collapse into one claim.}{Begin the Christian-poetry branch here.}
\Exercise{A 1962 Missal page prints a sequence. What does that witness establish directly?}{It establishes the wording and liturgical placement visible in that identified edition, within the recorded transcription and normalization boundary.}
\Exercise{What does the same Missal page not establish by itself?}{The poem's original author, first date, earlier textual history, complete metrical classification, or dependence on a proposed biblical source. Each needs its own evidence.}
\Exercise{Distinguish scriptural diction from a verified scriptural allusion.}{Scriptural diction is vocabulary or imagery characteristic of biblical Latin; a verified allusion requires a sufficiently specific verbal and contextual relationship to an identified source window.}
\Exercise{Why can a poem combine classical inheritance with a later rhythmic organization?}{Vocabulary, syntax, quantity, genre, line structure, verbal and written recurrence, and liturgical use have different histories. Evidence for one inherited feature does not dictate the whole poem's date or form.}
\LongExercise{Create a claim ledger with columns for claim, evidence, confidence, and limit.}{A sound ledger separates exact witness facts from grammatical analysis and interpretation. Every attribution, date, metrical, source, or liturgical-use statement receives its own evidence and stated limit.}
\end{worksheet}

\begin{worksheet}{P.18}{B}{P5-B}{The Pentecost Sequence as syntax}{Recover apposition, ellipsis, relative clauses, and petition.}{Use the verified sequence excerpt printed here.}
\PassageExercise{Consolator optime, dulcis hospes animæ, dulce refrigerium. In labore requies, in æstu temperies, in fletu solacium.}{Map the appositions and supplied predication.}{The addressee is named as \Latin{consolator optimus}, \Latin{hospes animæ dulcis}, and \Latin{refrigerium dulce}; the next triad names rest in labor, coolness in heat, and solace in weeping. A prose aid may supply \Latin{tu es}, but the supplied words are not part of the received line.}
\Exercise{Parse \Latin{optime, dulcis, dulce, animæ}.}{\Latin{optime} is vocative masculine singular with \Latin{consolator}; \Latin{dulcis} is vocative masculine singular with \Latin{hospes}; \Latin{dulce} is vocative neuter singular with \Latin{refrigerium}; \Latin{animæ} is genitive singular dependent on \Latin{hospes}.}
\PassageExercise{Lava quod est sordidum, riga quod est aridum, sana quod est saucium. Flecte quod est rigidum, fove quod est frigidum, rege quod est devium.}{Identify the repeated construction and the function of each \Latin{quod}.}{Six second-person singular imperatives govern six free relative clauses. Each neuter singular \Latin{quod} means ``that which/what'' and is subject of \Latin{est}; the predicate adjective supplies the condition to be cleansed, watered, healed, bent, warmed, or guided.}
\LongExercise{Turn the two imperative triads into a syntax table.}{The table should list imperative, object clause, predicate adjective, contextual sense, and the developing movement: cleansing and restoration followed by softening, warming, and guidance.}
\end{worksheet}

\begin{worksheet}{P.19}{C}{P5-C}{Verse and prose petition}{Compare exact grammar and lexical fields, not merely topic.}{The prose prayers below are from the same verified Pentecost formulary.}
\PassageExercise{Deus, qui hodierna die corda fidelium Sancti Spiritus illustratione docuisti: da nobis in eodem Spiritu recta sapere; et de eius semper consolatione gaudere.}{Map address, relative ground, and petitions in the Collect.}{\Latin{Deus} is vocative; \Latin{qui ... docuisti} states the ground in a relative clause; \Latin{da} governs the infinitives \Latin{sapere} and \Latin{gaudere}; \Latin{nobis} is dative, \Latin{recta} the neuter plural object of \Latin{sapere}, and \Latin{de consolatione} the phrase with \Latin{gaudere}.}
\PassageExercise{Sancti Spiritus, Domine, corda nostra mundet infusio: et sui roris intima aspersione fecundet.}{Identify the two wishes and their subjects.}{\Latin{mundet} and \Latin{fecundet} are present subjunctives expressing petitions. \Latin{infusio Sancti Spiritus} is subject of \Latin{mundet}; the understood same subject continues with \Latin{fecundet}. \Latin{corda nostra} is the first object, with the interior person understood in the second member; \Latin{intima aspersione sui roris} gives the means.}
\Exercise{Connect three exact lexical fields across sequence and prose.}{Examples: \Latin{consolator/consolatio}; \Latin{cordium/corda}; \Latin{reple/infusio}; cleansing in \Latin{lava/mundet}; interiority in \Latin{cordis intima/intima aspersione}. Each link should name actual forms and grammatical differences.}
\Exercise{What formal contrast is strongest?}{The sequence advances through vocative apposition, balanced nominal phrases, and direct imperative chains; the prose prayers develop a periodic relative ground or jussive petition with explicit syntactic subordination.}
\LongExercise{Write a comparison with two similarities, two differences, and one limit.}{A sound response cites forms, separates shared liturgical day from literary form, and does not turn lexical recurrence into proof of direct borrowing or common authorship.}
\end{worksheet}

\begin{worksheet}{P.20}{D}{P5-D}{Bounded Christian-poetry imitation}{Imitate one visible structure while keeping the composition explicitly project-made.}{Use after the evidence and syntax sheets.}
\Exercise{Imitate the pattern \Latin{in labore requies, in æstu temperies, in fletu solacium} with three ordinary settings.}{Model P: \Latin{in via comes, in nocte lumen, in periculo auxilium}. The nouns are nominative appositions or predications within the imitated ellipsis.}
\Exercise{Write a three-imperative chain with free relative \Latin{quod}.}{Model P: \Latin{Erige quod est lapsum, confirma quod est infirmum, illumina quod est obscurum.}}
\Exercise{Parse every form in the model chain.}{\Latin{erige, confirma, illumina} are singular imperatives; each \Latin{quod} is neuter singular subject of \Latin{est}; \Latin{lapsum, infirmum, obscurum} are neuter singular predicate adjectives.}
\Exercise{What must the source ledger say?}{That the wording is P; that its sixfold or threefold imperative-plus-free-relative structure is modeled on the supplied Pentecost Sequence; and whether any exact received phrase was retained.}
\LongExercise{Compose six balanced P lines, then revise one lexical field and one syntactic imbalance.}{Answers vary. The final text must remain grammatical, preserve a declared structural model, distinguish exact borrowing from imitation, and record why each revision improves sense or balance.}
\end{worksheet}

\begin{courseunit}{P6}
\chapter{P6 worksheet family: rhythmic hymnody and sequences}

\begin{worksheet}{P.21}{A}{P6-A}{Three independent pattern records}{Distinguish quantity, page-based textual pattern, and written line-ending correspondence.}{Begin before making any claim about sound.}
\Exercise{Define quantitative structure.}{An organization of metrically long and short syllables according to a stated scheme, supported by natural quantity, position, and permitted resolutions.}
\Exercise{Define page-based textual pattern for this course.}{A recurring organization of syllable counts, line groups, parallel constructions, repeated words, and written endings. It is recorded separately from quantitative scansion and from any claim about pronunciation.}
\Exercise{Distinguish a visible ending correspondence from rhyme or assonance.}{A visible correspondence records repeated letters or morphemes at line endings. Rhyme and assonance are sound classifications and require phonological evidence; similar spelling alone does not establish them.}
\Exercise{Why can one line support more than one record without the records becoming identical?}{The same written syllables can participate in lexical quantity, recurring count and grammar, and visible ending correspondence. Each observation answers a different structural question.}
\LongExercise{Make parallel ledgers for one supplied sequence triad.}{Ledger one records grammar; ledger two records syllable count, parallel construction, and line grouping; ledger three records quantity and uncertainty; a fourth small column records exact written-ending correspondences. Conclusions should cite the relevant ledger.}
\end{worksheet}

\begin{worksheet}{P.22}{B}{P6-B}{Visible pairings in the sequence}{Test exact and inexact correspondences rather than calling every ending rhyme.}{Use the complete Pentecost Sequence in the packet.}
\Exercise{Compare \Latin{radium/cordium}.}{Both end in the exact written sequence \Latin{-dium}, after different preceding material, and form a strong line-ending correspondence; their grammatical forms and lexical meanings remain different.}
\Exercise{Compare \Latin{refrigerium/solacium}.}{They share the broader neuter singular \Latin{-ium} ending but not a longer identical written sequence. This is weaker than identity of a longer ending and should be described precisely.}
\Exercise{Compare \Latin{saucium/devium}.}{Both have three-syllable adjective forms ending visibly in \Latin{-ium}, but the stems differ; record the correspondence without claiming identical quantity or pronunciation from spelling alone.}
\Exercise{What large-scale structure is visible in \Latin{lava, riga, sana / flecte, fove, rege}?}{Two balanced triads of imperatives, each followed by a free relative clause. The first moves through cleansing, watering, and healing; the second through bending, warming, and guiding.}
\LongExercise{Choose four paired endings from the sequence and rank the visible correspondence.}{A sound answer copies exact spellings, identifies shared material, notes grammatical form, and uses ordered descriptions such as exact longer ending, shared suffix, shared vowel letters, or weak resemblance.}
\end{worksheet}

\begin{worksheet}{P.23}{C}{P6-C}{The imperative architecture}{Recover the poem's full verbal movement from memory.}{Use the sequence only after attempting the chain unaided.}
\Exercise{Restore the full imperative chain.}{\Latin{veni---emitte---veni---veni---veni---reple---lava---riga---sana---flecte---fove---rege---da---da---da---da}. The exact chain has four occurrences each of \Latin{veni} and \Latin{da}; it should then be checked against the printed witness.}
\Exercise{Group the verbs into arrival, illumination, restoration, guidance, and gift.}{Arrival: four \Latin{veni} occurrences; sending/illumination and filling: \Latin{emitte, reple}; restoration: \Latin{lava, riga, sana}; guidance or correction: \Latin{flecte, fove, rege}; gift: four \Latin{da} occurrences. Other group labels are possible if all sixteen forms are accounted for.}
\Exercise{How do nominal lines contribute when no finite verb is printed?}{Vocatives and appositions name the addressee; balanced predicate nouns name the relief sought. Ellipsis carries the address and predication across line boundaries.}
\Exercise{Why does an imperative ledger reveal form before meter is settled?}{It identifies recurring clause spines and large-scale semantic progression directly from morphology. That structure does not depend on assigning every syllable to a quantitative scheme.}
\LongExercise{Reconstruct one omitted stanza from its verbs and lexical field, then compare with the source.}{The first version should be labeled P reconstruction. The comparison records exact words recovered, changed order or inflection, omissions, and what repeated returns made more dependable.}
\end{worksheet}

\begin{worksheet}{P.24}{D}{P6-D}{Patterned imitation and audit}{Build visible recurrence without sacrificing grammar.}{The model and all responses are P.}

\Exercise{What may not be inferred from a successful project pattern?}{Ancient or medieval authorship, liturgical reception, historical meter, or performance practice.}
\Exercise{Write two new parallel petitions with correct objects.}{Model P: \Latin{Corda nostra purifica; gressus nostros dirige.} Both imperative verbs govern masculine or neuter plural accusative objects with agreeing possessives.}
\LongExercise{Compose four to eight P lines with one repeated grammatical pattern and two exact visible ending correspondences.}{The record should include the prose thought, grammar check, line counts, ending comparison, revision list, and a statement of which pattern was deliberately imitated.}

\begin{worksheet}{P.25}{A}{P7-A}{The seven-layer commentary}{Keep textual fact, analysis, and inference in a reproducible order.}{Use as the commentary memory sheet.}
\Exercise{Name the seven layers in order.}{Witness and boundary; text, normalization, and any supplied material variants; meter or visible pattern; syntax and prose order; diction and rhetoric; supported source, allusion, or comparative context; interpretation, limits, and project translation.}
\Exercise{Why does witness identification come before interpretation?}{It fixes which wording, boundary, normalization, and edition are under discussion. Otherwise later claims may silently analyze a different or composite text.}
\Exercise{Where should a prose-order reconstruction appear?}{Within syntax analysis, clearly labeled P. It must not replace the received text or be cited as a variant.}
\Exercise{Where should unresolved quantity appear?}{In the metrical ledger and again among interpretive limits if it affects a claim. Uncertainty should not disappear between scansion and conclusion.}
\LongExercise{Make a one-page reusable commentary template.}{The template should prompt an exact locator and text class, normalization and material-variant fields (including ``none supplied''), scansion or visible-pattern reasons, syntax map, rhetoric, an optional supported context window, translation, thesis, counterevidence, and explicit limit.}
\end{worksheet}

\begin{worksheet}{P.26}{B}{P7-B}{Phaedrus and unequal position}{Build an interpretation from exact forms.}{Use the six-line verified Phaedrus window.}
\PassageExercise{Ad rivum eundem lupus et agnus venerant, siti compulsi. Superior stabat lupus, longeque inferior agnus.}{Separate shared condition from spatial contrast.}{Shared: the same stream, joint pluperfect arrival, and plural \Latin{siti compulsi}. Contrast: singular predicates \Latin{superior} and \Latin{inferior}, intensified by \Latin{longe}, assign unequal position to wolf and lamb.}
\PassageExercise{Tunc fauce improba latro incitatus iurgii causam intulit.}{Explain how the wolf is characterized before direct speech.}{\Latin{latro} renames him as robber; \Latin{fauce improba} characterizes the organ or manner of speech; \Latin{incitatus} presents him as driven; \Latin{iurgii causam intulit} says he introduced a cause for quarrel. The language frames the accusation before its content appears.}
\Exercise{Write a defensible thesis in one sentence.}{Model: Phaedrus turns physical elevation into an image of unequal power before the wolf manufactures a verbal grievance. This is a source-grounded inference, not wording copied from the witness.}
\Exercise{Give one limit to that thesis.}{The excerpt alone does not establish a general biography, political program, or universal theory of fable; even the power reading must remain tied to the exact contrast and narrative sequence.}
\LongExercise{Write a 250-word mini-commentary with all seven layers.}{A sound answer identifies Postgate 1919 and the excerpt boundary, records normalization, gives cautious metrical evidence, maps syntax, supports the thesis from exact diction, supplies a project translation, and states a limit.}
\end{worksheet}

\begin{worksheet}{P.27}{C}{P7-C}{Comparative poetic commentary}{Compare functions while preserving distinct genres and witnesses.}{Choose Virgil, Horace, and the Pentecost Sequence from the packet.}
\Exercise{Compare one kind of syntactic delay in Virgil and Horace.}{Virgil extends the relative subject's journey toward delayed \Latin{venit}; Horace embeds indirect questions and alternatives before concentrated counsel. Both sustain syntax, but their meters, genres, and arguments remain distinct.}
\Exercise{Compare one form of repetition in Horace and the Sequence.}{Horace uses paired \Latin{quem ... quem}, \Latin{seu ... seu}, and a culminating sequence of counsels; the Sequence repeats \Latin{veni}, balanced nominal triads, free relatives, and imperative groups. Exact functions should be cited from context.}
\Exercise{What would be an overclaim from these parallels?}{That one poet copied the other, that all three share a meter, or that similar imperatives prove common authorship or date. Structural comparison alone does not establish dependence.}
\Exercise{How should a project translation be used in comparison?}{As an explicit analytic rendering checked against the Latin, never as an independent ancient witness or substitute for form-level comparison.}
\LongExercise{Write a comparative paragraph with observation, inference, and limit visibly labeled.}{The observation quotes forms; the inference explains a function such as delay, balance, or concentration; the limit rejects unsupported historical or genetic claims.}
\end{worksheet}

\begin{worksheet}{P.28}{D}{P7-D}{Complete commentary dossier}{Join evidence without flattening its classes.}{Use four to twelve lines from one of the verified Virgil, Horace, or Pentecost texts reprinted in this packet.}
\Exercise{What is the minimum source header?}{Stable source ID; author or liturgical work as warranted; exact work and locus; consulted edition and page/scan locator; text class; normalization and excerpt boundary.}
\Exercise{What must a scansion or visible-pattern table show besides marks?}{For quantitative verse: syllable divisions, quantity reasons, foot or line positions, elision, alternative resolutions, and unresolved points. \Exercise{What makes a syntax map checkable?}{Every finite verb, non-finite form, subject, object, governor, clause relation, and significant ellipsis is accounted for from the actual Latin.}
\Exercise{What makes an interpretation proportionate?}{It rests on quoted grammar, diction, order, form, and context; addresses plausible counterevidence; and claims no more than the supplied witness and comparison establish.}
\LongExercise{Produce a 500-word commentary dossier and a one-paragraph self-audit.}{The dossier includes all seven layers. The self-audit names the strongest evidence, weakest inference, unresolved textual or metrical point, and the exact revision another return should test.}
\end{worksheet}

\begin{worksheet}{P.29}{A}{P8-A}{Prose plan before verse}{Fix the intended proposition and declared form.}{Begin every composition here, even after extensive experience.}
\Exercise{Name the four bounded choices offered by the packet.}{Four hexameters; two elegiac couplets; one Horatian line pattern sustained for four lines; or a six- to twelve-line rhythmic hymn stanza.}
\Exercise{What belongs in the prose plan?}{Speaker and addressee, principal proposition, grammatical spine, sequence of thought, essential images or petitions, intended register, and the exact relation between movements.}
\Exercise{Why should the form be declared before drafting?}{Its fixed constraints determine what must be measured and revised, while preventing an accidental sequence of lines from being relabeled after the fact.}
\Exercise{Write a two-sentence prose plan for a hymn stanza about light and perseverance.}{Model P: \Latin{Christus mentes laborantium illuminat. Ei petimus ut nos in via confirmet et ad finem bonum perducat.}}
\LongExercise{Expand the prose plan into a line-by-line thought outline without changing vocabulary.}{The outline should divide complete semantic movements, show carryover where intended, and reveal whether the chosen form has enough or too much material before metrical substitution begins.}

\begin{worksheet}{P.30}{B}{P8-B}{Quantity and syntax ledgers}{No metrical success can purchase a false form or lost relation.}{Use during the first patterned draft.}
\Exercise{List the columns of a quantity ledger.}{Word and syllable; proposed quantity; natural or positional reason; elision or resolution; metrical position; alternative; uncertainty.}
\Exercise{List the columns of a syntax ledger.}{Word or phrase; lemma and morphology; governor; function; clause; relation to the prose plan; any unusual order or ellipsis.}
\Exercise{What warning is raised when the meter works but the syntax ledger has an unattached accusative?}{The line is unsound until the case receives a defensible governor and function. Metrical fit does not create syntax.}
\Exercise{What warning is raised when syntax works but one long syllable has no reason?}{The proposed scan is not established. Recheck lexical quantity, position, elision, scheme, and wording before keeping the line.}
\LongExercise{Audit one line in both ledgers and write a verdict.}{The verdict should be \emph{retain}, \emph{revise}, or \emph{return to prose}, with exact reasons from both ledgers rather than a general impression.}
\end{worksheet}

\begin{worksheet}{P.31}{C}{P8-C}{Revision and source ledgers}{Make every substitution and every influence visible.}{Use after a complete first draft exists.}
\Exercise{List the columns of a revision ledger.}{Draft number; old wording; new wording; reason; grammatical effect; semantic effect; metrical or rhythmic effect; retained uncertainty.}
\Exercise{List the columns of a source ledger.}{Feature or exact phrase; stable source ID and locus; quotation, adaptation, structural model, echo, or general influence; changes made; claim limit.}
\Exercise{Why is a metrically convenient synonym not automatically acceptable?}{It may have the wrong construction, register, semantic range, quantity, or source association. Substitution is a lexical judgment, not an English-gloss exchange.}
\Exercise{Give three separate revision passes.}{Exactness of thought; idiomatic grammar, order, and diction; metrical or rhythmic pattern. A later proof pass should also recheck source labels and transcription.}
\LongExercise{Document three actual revisions, including one rejected attractive phrase.}{A sound record explains what each change fixes and what it costs. The rejected phrase is retained in the ledger with the grammatical, semantic, source, or pattern reason for removal.}
\end{worksheet}

\begin{worksheet}{P.32}{D}{P8-D}{Final poem and commentary}{Present original verse as original, with enough evidence to audit it.}{Use when a bounded composition has survived all ledgers.}
\Exercise{What must appear above the poem?}{A P label, declared form and scheme, short prose statement of intended thought, and the composition's own title or identifier without false attribution.}
\Exercise{What must appear below the poem?}{Scansion or rhythmic-pattern record, project translation, syntax notes, revision summary, source ledger, remaining uncertainty, and prose commentary using the P7 layers.}
\Exercise{Translate and explain the model documentation sentence \Latin{Nihil tamen auctori antiquo attribuo quod ipse composui}.}{``Yet I attribute to no ancient author what I myself composed.'' \Latin{nihil} is object of \Latin{attribuo}; \Latin{auctori antiquo} dative; \Latin{quod} has \Latin{composui} and refers to what was composed; \Latin{ipse} emphasizes the composer.}
\Exercise{When is an unfinished composition still a successful record?}{When it accurately identifies why a sound line or stanza has not yet been achieved, preserves the grammatical prose plan, documents attempted revisions, and makes the next test specific.}
\LongExercise{Prepare the final bounded poem, four ledgers, translation, and self-commentary.}{Answers vary. \begin{worksheet}{P.5}{A}{P2-A}{The six-foot frame}{Memorize the frame before solving variations within it.}{Use as the hexameter construction sheet.}
\Exercise{State the alternatives in feet one through four, the usual fifth foot, and the sixth foot.}{Feet one through four may be dactyls (long--short--short) or spondees (long--long); the fifth is normally a dactyl; the sixth is long plus anceps.}
\Exercise{Why may a word boundary fall inside a foot?}{Feet are metrical units made of syllables; words are lexical units. Their boundaries interact rhetorically but do not have to coincide.}
\Exercise{Define a caesura cautiously.}{A significant word boundary within a foot, often contributing an internal articulation of the line. Its presence and effect must be shown in the particular verse.}
\Exercise{What does enjambment require the reader to do?}{Continue the syntax across the verse boundary while still observing the line's metrical and rhetorical shape. A line end does not automatically complete a clause.}
\LongExercise{From memory draw six empty feet and add a separate row for word boundaries.}{The first row should show four variable feet, a normally dactylic fifth, and long plus anceps sixth. The second row must remain independent so that a word boundary is not mistaken for a foot division.}

\Exercise{Identify the objects of \Latin{cano} and the antecedent of \Latin{qui}.}{\Latin{arma} and \Latin{virum} are coordinated accusative objects; \Latin{qui} refers to \Latin{virum}.}

\begin{worksheet}{P.7}{C}{P2-C}{A sentence across four lines}{Map delayed government before interpreting epic scale.}{Use lines 1--4 as one sentence.}

\Exercise{Parse \Latin{profugus, iactatus, memorem}.}{\Latin{profugus} nominative masculine singular agreeing with the man; \Latin{iactatus} nominative masculine singular perfect passive participle with \Latin{ille}; \Latin{memorem} accusative feminine singular agreeing with \Latin{iram}.}
\Exercise{What is gained by postponing \Latin{venit}?}{Origins, exile, and destination unfold around the relative subject before the motion reaches its finite completion. The delay lets the journey feel extended without making the grammar incomplete.}
\Exercise{Give a cautious note on \Latin{et terris ... et alto}.}{Repeated \Latin{et} balances the two spheres of wandering. \Latin{alto} is a substantivized neuter adjective for the deep sea; the phrase is syntactically parallel to \Latin{terris}.}
\LongExercise{Diagram the sentence from \Latin{cano} through \Latin{ob iram}.}{The diagram should keep \Latin{cano} with its two objects; nest the relative clause under \Latin{virum}; place \Latin{venit} as its finite spine; attach destinations, appositional/predicative modifiers, participial material, and causal phrases to the correct governors.}

\begin{worksheet}{P.9}{A}{P3-A}{The elegiac couplet}{Learn the two-line contract and its central division.}{Begin before scanning the Ovid couplet.}
\Exercise{What two lines form an elegiac couplet?}{A dactylic hexameter followed by an elegiac pentameter.}
\Exercise{Describe the pentameter as two balanced halves.}{Each half contains two-and-a-half metrical units around a fixed central division; the second half closes with two dactyls and a final long or anceps syllable.}
\Exercise{What normally happens to a word at the central division?}{A word normally does not bridge it. The division should be checked in the actual line rather than inferred from punctuation alone.}
\Exercise{Why is \emph{pentameter} not permission to count five interchangeable hexameter feet?}{The name reflects the traditional total, but the line has a fixed bipartite structure unlike five freely substitutable hexameter feet. Its halves and closure govern the scan.}
\LongExercise{Draw the couplet skeleton from memory and label where semantic turn may occur.}{The drawing should distinguish the six-foot hexameter from the divided pentameter. The semantic turn often answers or narrows the first line but must be demonstrated from each couplet's grammar and thought.}
\end{worksheet}

\begin{worksheet}{P.10}{B}{P3-B}{Ovid addresses his book}{Construe person, address, parenthesis, and impersonal syntax.}{Use the verified \textit{Tristia} couplet in the packet.}
\PassageExercise{Parve---nec invideo---sine me, liber, ibis in urbem: ei mihi, quod domino non licet ire tuo!}{Parse \Latin{parve, liber, ibis, mihi, domino tuo, ire}.}{\Latin{parve} and \Latin{liber} are vocative masculine singular; \Latin{ibis} is future active indicative second singular; \Latin{mihi} is dative in the exclamation; \Latin{domino tuo} is dative of the person to whom the action is permitted; \Latin{ire} is the complementary infinitive with impersonal \Latin{licet}.}
\Exercise{What does \Latin{nec invideo} do syntactically and rhetorically?}{It is a parenthetic first-person comment, ``nor do I begrudge it.'' It interrupts the address to qualify the speaker's response to the book's freedom.}
\Exercise{What relation does \Latin{quod} introduce here?}{A causal explanation of the lament: because the book's master is not permitted to go.}
\Exercise{Give a neutral prose order without erasing the address.}{Model P: \Latin{Parve liber, sine me in urbem ibis---nec invideo. Ei mihi, quod domino tuo ire non licet!}}
\LongExercise{Explain the person-system of the couplet.}{The speaker addresses the book as second person, inserts his own first-person response in \Latin{invideo/mihi/me}, and refers to himself again as the book's third-person \Latin{domino tuo}. The shifts dramatize separation.}
\end{worksheet}

\begin{worksheet}{P.11}{C}{P3-C}{The turn in the couplet}{Join metrical division to a change in thought without confusing them.}{Use after the syntax of P3-B is secure.}
