\begin{courseunit}{II.1}
\chapter*{How to select and score a variant}
\addcontentsline{toc}{chapter}{How to select and score a variant}

Each lettered form within a unit is interchangeable in skill and approximate difficulty.  Use
one for guided practice, a second for the gate, and reserve the others for remediation or a
later retention check.  Unless an item says otherwise, parse every bolded or requested form
fully and translate the entire Latin sentence.  A model translation may be worded differently
without being wrong, but its agency, negation, time, and clause relation must match.

For a six-item sheet, mastery is five correct; for a seven-item sheet, six correct.  In either
case there may be no foundational error in the named focus.
An answer is not correct merely because its English sounds plausible: the model identifies the
Latin decision that must be visible.  Copy every corrected Latin sentence once in full before
attempting another variant.

\chapter{Pronouns: parallel forms}

\begin{worksheet}{II.1}{A}{V2.1A}{Pronouns A}{relative agreement, internal case, questions, indefinites}
{Complete without a paradigm; show the case-giver for every relative.}
\Exercise{Write the relative pronoun in all five singular cases, masculine, feminine, and
neuter.}{Nom. \Latin{qui, quæ, quod}; gen. \Latin{cuius} in all genders; dat. \Latin{cui};
acc. \Latin{quem, quam, quod}; abl. \Latin{quo, qua, quo}.}
\Exercise{Supply the relative: \Latin{sacrificium \Blank{} offerimus; hostia \Blank{}
offertur; Patri \Blank{} gratias agimus}.}{\Latin{quod} (neuter accusative object of
\Latin{offerimus}); \Latin{quæ} (feminine nominative subject of \Latin{offertur});
\Latin{cui} (masculine dative with \Latin{gratias agimus}).}
\Exercise{Parse \Latin{quid} and \Latin{quæ}: \Latin{Quid retribuam pro omnibus quæ
retribuit mihi?}}{\Latin{Quid}: interrogative pronoun, accusative neuter singular, object of
\Latin{retribuam}.  \Latin{Quæ}: relative, accusative neuter plural, referring to substantival
\Latin{omnibus}, object of \Latin{retribuit}.}
\Exercise{Combine with one relative clause: \Latin{Laudamus Agnum. Agnus peccata mundi
tollit.}}{\Latin{Laudamus Agnum, qui peccata mundi tollit.}  \Latin{Qui} is nominative because
it is subject of \Latin{tollit}, though its antecedent is accusative.}
\PassageExercise{Qui sedes ad dexteram Patris, miserere nobis.}{Parse \Latin{qui}; identify its
antecedent and translate.}{\Latin{Qui}: nominative masculine singular, subject of
\Latin{sedes}, referring to the Christ addressed as implicit \Latin{tu}.  ``You who sit at the
Father's right hand, have mercy on us.''}
\LongExercise{Write in Latin: ``If anyone receives what we offer, let him give thanks to
God.''}{\Latin{Si quis quod offerimus accipit, gratias Deo agat.}  After \Latin{si}, indefinite
\Latin{quis} normally lacks \Latin{ali-}; \Latin{quod} is a headless relative.}
\end{worksheet}

\begin{worksheet}{II.2}{B}{V2.1B}{Pronouns B}{relative agreement, internal case, questions, indefinites}
{Complete without a paradigm; show the case-giver for every relative.}
\Exercise{Write the complete plural paradigm of \Latin{qui, quæ, quod}.}{Nom.
\Latin{qui, quæ, quæ}; gen. \Latin{quorum, quarum, quorum}; dat. \Latin{quibus} all genders;
acc. \Latin{quos, quas, quæ}; abl. \Latin{quibus} all genders.}
\Exercise{Supply the relative: \Latin{dona \Blank{} benedicis; discipuli \Blank{} credunt;
Ecclesia \Blank{} servimus}.}{\Latin{quæ} (neuter accusative plural); \Latin{qui} (masculine
nominative plural); \Latin{cui} (feminine dative singular with \Latin{servimus}).}
\Exercise{Distinguish the forms: \Latin{Quis venit? Quod donum portat? Quid portat?}}{
\Latin{Quis} is substantival nominative, ``who?''; \Latin{quod} is adjectival accusative neuter
singular agreeing with \Latin{donum}, ``which gift?''; \Latin{quid} is substantival accusative,
``what?''}
\Exercise{Replace the repeated noun: \Latin{Suscipe oblationem. Oblationem offerimus pro
Ecclesia.}}{\Latin{Suscipe oblationem quam pro Ecclesia offerimus.}  \Latin{Quam} is feminine
singular from its antecedent and accusative as object of \Latin{offerimus}.}
\PassageExercise{Nemo venit ad Patrem, nisi per me.}{Parse \Latin{nemo}; explain
\Latin{nisi}; translate.}{\Latin{Nemo} is nominative singular subject of \Latin{venit};
\Latin{nisi} makes an exception to the universal negative.  ``No one comes to the Father
except through me.''}
\LongExercise{Write in Latin: ``Whoever hears these words does what God commands.''}{
\Latin{Quicumque hæc verba audit, facit quod Deus præcipit.}  In \Latin{quicumque}, only
\Latin{qui} declines; \Latin{quod} is headless, ``what.''}
\end{worksheet}

\begin{worksheet}{II.3}{C}{V2.1C}{Pronouns C}{relative agreement, internal case, questions, indefinites}
{Complete without a paradigm; show the case-giver for every relative.}
\Exercise{Give genitive, dative, accusative, and ablative singular of \Latin{quis? quid?}.}{
Gen. \Latin{cuius}; dat. \Latin{cui}; acc. masculine/feminine \Latin{quem}, neuter
\Latin{quid}; abl. \Latin{quo, qua, quo}.  The plural follows the relative paradigm.}
\Exercise{Supply the relative: \Latin{calix ex \Blank{} bibimus; verba \Blank{} audivisti;
sacerdotes \Blank{} dona tradidimus}.}{\Latin{quo} (ablative after \Latin{ex}); \Latin{quæ}
(neuter accusative plural); \Latin{quibus} (masculine dative plural with \Latin{tradidimus}).}
\Exercise{Explain the forms in \Latin{si quis, ne quid, aliquis venit, quidam sacerdos}.}{
After \Latin{si} and \Latin{ne}, indefinite \Latin{quis/quid} normally loses \Latin{ali-}.
\Latin{Aliquis} is ``someone'' without such a trigger; \Latin{quidam} modifying
\Latin{sacerdos} is ``a certain.''}
\Exercise{Make the antecedent headless: \Latin{Accipimus id quod Deus dat.}}{
\Latin{Accipimus quod Deus dat.}  Neuter \Latin{quod} alone means ``what/the thing that'' and
is object of both the relative clause and the main verb conceptually.}
\PassageExercise{Quod ore sumpsimus, Domine, pura mente capiamus.}{Parse \Latin{quod} and
translate without treating it as a conjunction.}{\Latin{Quod} is accusative neuter singular,
object of \Latin{sumpsimus}; its headless relative clause is object of \Latin{capiamus}.
``Lord, may we take in with a pure mind what we have received by mouth.''}
\LongExercise{Write in Latin: ``Each person offers something; no one offers everything.''}{
\Latin{Quisque aliquid offert; nemo omnia offert.}  \Latin{Quisque} and \Latin{nemo} are
singular subjects; \Latin{aliquid} and \Latin{omnia} are accusative objects.}
\end{worksheet}

\begin{worksheet}{II.4}{D}{V2.1D}{Pronouns D}{relative agreement, internal case, questions, indefinites}
{Complete without a paradigm; show the case-giver for every relative.}
\Exercise{Decline \Latin{uter, utra, utrum} in the singular.}{Nom. \Latin{uter, utra, utrum};
gen. \Latin{utrius}; dat. \Latin{utri}; acc. \Latin{utrum, utram, utrum}; abl.
\Latin{utro, utra, utro}.}
\Exercise{Supply the relative: \Latin{Mater \Blank{} Filium videt; Filius \Blank{} Mater
videt; discipulus a \Blank{} panis accipitur}.}{\Latin{quæ} (Mother is subject);
\Latin{quem} (Son is object); \Latin{quo} (masculine ablative after \Latin{a}).}
\Exercise{Parse \Latin{neminem, nemini, nihil, nullius hominis, nullius rei} and
give the base meaning.}{\Latin{Neminem}: accusative of \Latin{nemo};
\Latin{nemini}: dative of \Latin{nemo}; \Latin{nihil}: indeclinable neuter used
in nominative/accusative; \Latin{nullius hominis}: genitive commonly supplying
an oblique sense for \Latin{nemo}; \Latin{nullius rei}: genitive phrase for
\English{of nothing}. Bare \Latin{nullius} does not by itself supply the missing

\Latin{Hostiam quam offerimus suscipe.}  The antecedent is accusative, but \Latin{quam} is

\PassageExercise{Omnis qui se exaltat humiliabitur, et qui se humiliat exaltabitur.}{Parse both
instances of \Latin{qui} and translate.}{Both are nominative masculine singular, subject of
their indicative verbs and referring to substantival \Latin{omnis} understood in each half.
``Everyone who exalts himself will be humbled, and he who humbles himself will be exalted.''}
\LongExercise{Write in Latin: ``Which of the two gifts will you receive, and to whom will you
give it?''}{\Latin{Utrum donum accipies, et cui illud dabis?}  \Latin{Utrum} is interrogative
adjective; \Latin{cui} is dative interrogative pronoun.}
\end{worksheet}

\Latin{Scimus Christum Dominum esse et venturum esse.}  \Latin{Venturum} agrees with

\begin{worksheet}{II.53}{A}{V2.14A}{Compression A}{fronting, hyperbaton, ellipsis, attraction, collect

\Exercise{Reconnect \Latin{sanctas in manus suas accepit}.}{\Latin{sanctas} and \Latin{suas}

pedagogical expansion may use \Latin{[est]}, but the Latin is a verbless nominal
acclamation and does not uniquely require an omitted copula.}
\Exercise{Unpack \Latin{in qua mensura mensi fueritis}.}{``In the measure in which you have

apposition to \Latin{Iesum Christum}; \Latin{supplices}: implicit \Latin{nos};
