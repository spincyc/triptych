\newcommand{\MemoryReconstructionSheet}[1]{%

    Reconstruct before consulting any model.  Account for every supplied form,
    then annotate agreement, government, and clause structure.}%
  \textbf{Prompt.} \MemorySetField{#1}{reconstruction}
  \ifmemorysolutions
    \par\medskip\textbf{Model and points to check.}
    \MemorySetField{#1}{reconstructionmodel}
  \else
    \memory@fourlines
    \par\medskip\textbf{Morphology and syntax notes:}\memory@fourlines
  \fi
}

\newcommand{\MemoryCompositionSheet}[1]{%
  \memory@sheetstart{#1-COMP}{Cumulative composition}{%
    Plan the propositions first; draft, check every form and construction,
    revise, and only then compare the model.  The model is one defensible
    project composition, not a sentence to imitate mechanically.}%
  \textbf{Prompt.} \MemorySetField{#1}{composition}
  \ifmemorysolutions
    \par\medskip\textbf{Model and points to check.}
    \MemorySetField{#1}{compositionmodel}
  \else
    \par\medskip\textbf{Proposition outline:}\memory@twolines
    \par\medskip\textbf{Draft and revision:}\memory@fourlines
    \par\vspace{1.1\baselineskip}\hrulefill
    \par\vspace{1.1\baselineskip}\hrulefill
  \fi
}

\newcommand{\memory@packetheading}[2]{%
  \clearpage
  \chapter{#1: \MemorySetField{#1}{title} --- #2}
  \textbf{Packet memory key:} #1.  Work by study event, not by a calendar.
  A dependable return means that citation forms, core senses, morphological
  facts, constructions, and the packet's grammar explanations can be produced
  on fresh material without first looking at the model.
  \firstmemorysheettrue
}

\newcommand{\RenderCompleteMemoryLexicon}{%
  \chapter{Controlled Reading Lexicon}
  This cumulative lexicon contains \arabic{MemoryLexemeCount} stable entries
  used by the packet memory sets.  Citation form, concise project gloss,
  morphology or principal parts, and the principal construction cue are
  printed here so assigned packet work does not depend on an outside
  dictionary.  The strand is a pedagogical allocation, not a claim that an
  occurrence has been verified in a named author.  Broader dictionaries are
  appropriate later for extension research, semantic range, and source-specific
  idiom after an unaided hypothesis has been formed.
