\section{Sources and further reading}

\subsection{Primary and official sources}

\begin{itemize}[leftmargin=1.5em,itemsep=0.45em]
\item \emph{Missale Romanum ex decreto Sacrosancti Concilii Tridentini restitutum, Summorum Pontificum cura recognitum}, editio typica, 1962. Calendar and \emph{Proprium de tempore}; digitized facsimile used for collation: \sourceurl{https://media.churchmusicassociation.org/pdf/missale62.pdf}{Church Music Association of America mirror}.
\item John XXIII, \emph{Rubricarum instructum}, 25 July 1960, \emph{Acta Apostolicae Sedis} 52 (1960), 593--595; \sourceurl{https://www.vatican.va/content/john-xxiii/la/motu_proprio/documents/hf_j-xxiii_motu-proprio_19600725_rubricarum-instructum.html}{Holy See Latin text}.
\item Sacred Congregation of Rites, \emph{Codex rubricarum}, \emph{Acta Apostolicae Sedis} 52 (1960), 597--729, especially General Rubrics nn. 1--57 and the table of precedence; the Mass rubrics at 643--685.
\item Sacred Congregation of Rites, \emph{Instructio de calendariis particularibus et Officiorum ac Missarum propriis ad normam et mentem codicis rubricarum revisendis}, \emph{Acta Apostolicae Sedis} 53 (1961), 168--180; \sourceurl{https://www.vatican.va/archive/aas/documents/AAS-53-1961-ocr.pdf}{official AAS volume}.
\item Desclée \& Co., \emph{Proper Feasts Kept in the Dioceses of the United States of America}, imprimatur 28 September 1961, appended to a chant edition; \sourceurl{https://s9.imslp.org/files/imglnks/usimg/a/ad/IMSLP52084-PMLP29410-LU13_Supplement_for_certain_religious_congregations.pdf}{digitized composite PDF}, U.S. section beginning at PDF p. 39. Used only for calendar headings and jurisdiction labels, not reproduced prayer texts.
\item Congregation for the Doctrine of the Faith, decree \emph{Cum sanctissima}, 22 February 2020, effective 19 March 2020; \sourceurl{https://www.vatican.va/roman_curia/congregations/cfaith/documents/rc_con_cfaith_doc_20200222_decreto-cum-sanctissima_en.html}{official English text and annex}.
\item Congregation for the Doctrine of the Faith, \sourceurl{https://www.vatican.va/roman_curia/congregations/cfaith/documents/rc_con_cfaith_doc_20200222_nota-decreto-cumsanctissima_en.html}{official presentation note for \emph{Cum sanctissima}}, 22 February 2020.
\end{itemize}

\subsection{Earlier and successor reforms}

\begin{itemize}[leftmargin=1.5em,itemsep=0.45em]
\item Pius V, apostolic constitutions \emph{Quod a nobis} (9 July 1568) and \emph{Quo primum} (14 July 1570), promulgating the reformed Roman Breviary and Missal.
\item Pius X, apostolic constitution \emph{Divino afflatu}, 1 November 1911, \emph{Acta Apostolicae Sedis} 3 (1911), 633--638, with the accompanying rubrical changes.
\item Sacred Congregation of Rites, general decree \emph{Cum hac nostra aetate}, 23 March 1955, \emph{Acta Apostolicae Sedis} 47 (1955), 218--224; and general decree \emph{Maxima redemptionis nostrae mysteria}, 16 November 1955, \emph{Acta Apostolicae Sedis} 47 (1955), 838--847.
\item Paul VI, apostolic letter \emph{Mysterii Paschalis}, 14 February 1969, and the \emph{Calendarium Romanum} (1969), for the successor postconciliar calendar. They are comparative boundary sources, not inputs to the 1962 tables.
\end{itemize}

\subsection{Historical synthesis}

\begin{itemize}[leftmargin=1.5em,itemsep=0.45em]
\item Thomas J. Talley, \emph{The Origins of the Liturgical Year}, 2nd ed. (Collegeville, MN: Liturgical Press, 1991). Used for the long formation of the weekly and annual cycles, not as a substitute for edition-specific calendar evidence.
\end{itemize}
