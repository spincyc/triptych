\section{References}

\subsection{Controlling sources}

{\footnotesize
\begin{description}[style=multiline,leftmargin=0.28\linewidth,labelwidth=0.25\linewidth,itemsep=0.2em,topsep=0.2em,parsep=0pt]
\item[Promulgation] John XXIII, apostolic letter issued \latin{motu proprio}, \latin{Rubricarum instructum}, 25 July 1960, \latin{Acta Apostolicae Sedis} 52 (1960), 593--595; see the \href{https://www.vatican.va/content/john-xxiii/la/motu_proprio/documents/hf_j-xxiii_motu-proprio_19600725_rubricarum-instructum.html}{official Vatican Latin text}.
\item[Rubrical code] \latin{Codex rubricarum Breviarii ac Missalis Romani}, with the decree and associated material in \latin{Acta Apostolicae Sedis} 52 (1960), 596--740; see the \href{https://www.vatican.va/archive/aas/documents/AAS-52-1960-ocr.pdf}{official AAS volume}.  This source controls the classes, precedence table, occurrence, transfer, calendars, and the general structure incorporated into the Missal.
\item[Typical Missal] \latin{Missale Romanum ex decreto Sacrosancti Concilii Tridentini restitutum, Summorum Pontificum cura recognitum}, editio typica, Typis Polyglottis Vaticanis, 1962.  The checked public facsimile is hosted by the Church Music Association of America: \url{https://media.churchmusicassociation.org/pdf/missale62.pdf}.  Its \latin{Rubricae generales Missalis Romani}, \latin{Ritus servandus in celebratione Missae}, \latin{Ordo Missae}, Canon, temporal and sanctoral formularies, Commons, votives, Requiems, and ritual formularies control assembly at the altar.
\end{description}
}
