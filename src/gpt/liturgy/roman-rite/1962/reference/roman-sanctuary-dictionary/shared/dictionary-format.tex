% A single atlas cell. #1 stable ID; #2 English; #3 Latin; #4 art;
% #5 one-line purpose; #6 status/handling line; #7 optional symbolism.
\newcommand{\DictionaryCell}[7]{%
  \DictionaryEntryAnchor{#1}%
  \begin{minipage}[t]{\linewidth}
    \centering #4\par
    \vspace{2pt}
    \raggedright
    \noindent{\fontsize{8.8}{9.8}\selectfont\bfseries #2\hfill
      \DictionaryNoteKey{#1}\par}
    {\fontsize{7.8}{8.8}\selectfont\itshape #3\par}
    \vspace{1pt}
    {\fontsize{7.8}{9.2}\selectfont #5\par}
    \if\relax\detokenize{#6}\relax\else
      {\fontsize{6.9}{8.2}\selectfont\bfseries #6\par}%
    \fi
    \if\relax\detokenize{#7}\relax\else
      {\fontsize{7.1}{8.4}\selectfont
        \(\diamond\)\,\textit{#7}\par}%
    \fi
  \end{minipage}}

% Compact review questions and uncertainty statements sit between plate rows
% or directly below the affected cell. They are deliberately unsuitable for
% publication prose and disappear when the review switch is false.
\newcommand{\DictionaryReviewerQuestion}[2]{%
  \ifDictionaryReviewEdition
    \par\vspace{3pt}
    \noindent
    \begin{tcolorbox}[
      colback=white,
      frame hidden,
      boxrule=0pt,
      sharp corners,
      borderline west={1.2pt}{0pt}{black},
      left=5pt,right=1pt,top=2pt,bottom=2pt,
      before skip=0pt,after skip=0pt]
      {\fontsize{7.2}{8.5}\selectfont
       \textbf{REVIEWER QUESTION \texttt{#1}.}\enspace #2}
    \end{tcolorbox}
  \fi}

\newcommand{\DictionaryTerminalMatter}{%
  \appendix
  \section{Pronunciation Reference}
  Commonly spoken Latin headwords will be selected only after the inventory
  and terminology audit are complete.
  \section{English--Latin and Latin--English Indexes}
  Index entries will be generated from the canonical inventory.
  \section{Visual Index}
  Plate and figure references will be generated from the plate
  manifest.
  \section{Variant and Terminology Notes}
  Source-qualified notes will distinguish universal Roman use, authorized or
  regional variants, religious-community use, and historical examples.
  \section{Scope}
  The declared completeness boundary and rights state remain explicit.
  \section{Numbered Source Notes and References}
  Entry-keyed source notes and the reference list will be generated
  from canonical bindings after claim-level verification.
}
