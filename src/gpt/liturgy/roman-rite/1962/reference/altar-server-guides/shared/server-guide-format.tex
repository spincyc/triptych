% The training series keeps Libertinus as its readable working face.  A

% Keep publication qualifications off the detachable cards.  This uncut,
% adult-facing terminal sheet follows the last complete front/back pair, so it
% cannot disturb card positions or duplex parity.  The fixed page-relative
% measure also works for the zero-margin Low Mass flash-card document.
\newcommand{\CardCompanionTerminalApparatus}{%
  \clearpage
  \thispagestyle{empty}%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=north,text width=6.8in,align=left,inner sep=0pt]
      at ([yshift=-0.72in]current page.north) {%
        \begingroup
        \RaggedRight
        {\Large\MissalDisplay{About This Card Companion}\par}
        \vspace{0.7em}
        {\small
        \textbf{Training boundary.} These cards are rehearsal prompts for the
        ordinary 1962 Roman Rite route named on the deck. They are not an
        official liturgical book, parish directive, certificate of competence,
        or substitute for the full guide and its branch qualifications. The
        priest and master of ceremonies govern the actual function.\par
        \vspace{0.65em}
        \textbf{Safe use.} Rehearse under a responsible adult. That adult
        controls flames, hot charcoal, unfamiliar objects, and local hazards.
        If a cue is missed, stop safely, secure any object, and look to the
        priest or master of ceremonies; never improvise a rushed crossing.\par
        \vspace{0.65em}
        \textbf{Text and scope.} Latin responses are collated to the 1962
        Vatican typical Missal. Learner pronunciation and memory helps are
        editorial aids, not chant notation or an English ritual translation.
        Special rites and local ceremonial variations require their own
        instructions.\par
        \vspace{0.65em}
        \textbf{Rights.} Project-created teaching matter falls within the
        repository license boundary. Received liturgical text, official
        sources, fonts, and third-party material retain their own status.
        Installation does not authorize public distribution.\par
        \vspace{0.8em}
        \textit{This sheet is not a card and is not part of the cut grid. Keep
        it with the deck for adult reference.}\par}%
        \endgroup};
  \end{tikzpicture}%
  \null}

% Some longtable releases append \vss before splitting a page box. Its
% infinite shrink component triggers pdfTeX's split-box diagnostic. A
% stretch-only tail has the same bottom-fill purpose without invalid shrink.
%
% The tails are corrected where they are found and nowhere else. The count is
% not fixed at two: a release that appends \vss once, or that has already
% dropped it, is as valid an input as the two-tail release this was written
% against, and demanding two of them fails the build on a toolchain whose
% longtable needs no correction at all. TeX Live 2026 as Arch packages it has
% two; the pinned deployment container has none, and refusing to typeset there
% was a defect in this patch rather than in that container.
\makeatletter
\ifpatchable{\LT@output}{\vss}{%
  \patchcmd{\LT@output}{\vss}{\vfil}{}{}%
  \ifpatchable{\LT@output}{\vss}{%
    \patchcmd{\LT@output}{\vss}{\vfil}{}{}%
  }{}%
}{}%
\makeatother
