\section{Walk the Sanctuary Together}
\ASGOrientationKey
\noindent\textbf{Read one moment at a time.} A station capsule names a person
or object omitted from the close view. It does not turn a sacred minister
into a lay server. Six altar candles identify the sung model; the Low-Mass
position views use two. The torch and incense stations are separate.

\ifMissaCantataGuide
\SungMovementPanel{1. Prayers at the foot (MC-006--010)}
{\MCFootPositions}
{The entrance objects are placed and the priest begins.}
{MC, A1 and A2; free hands.}
{Arrival line to the marked foot-prayer places. Kneel together; clear the
centre when the priest ascends.}
{MC goes to the book side; A1 and A2 return to base.}
{T1--T4 and optional CB/L at base; TH/BB at the safe station on the incense branch.}

\SungMovementPanel{2. Go to the Gospel, then remain still (MC-022--026)}
{\MCGospelPositions}
{MC's Gospel-preparation signal after the interlection chants.}
{A1/A2: candles; MC: book service; TH: closed thurible if used.}
{Credence to centre stop A, then the Gospel-side altar book. Form the
candle pair only at the station; MC attends beside the book.}
{Remain through the Gospel. Afterwards candles return to their stands;
TH returns to the safe station.}
{BB has returned the boat; T1--T4, optional CB/L stay at base. The priest sings at the altar Missal.}

\SungMovementPanel{3. Present wine, then water (MC-030--031)}
{\MCCruetPositions}
{The Offertory begins and MC clears the approach.}
{A1: wine; A2: water.}
{Base to credence, then together to the Epistle handoff point. Present
wine first; recover it before presenting water.}
{Receive the same cruets back; bow and return both to their marked places.}
{MC watches the next incense or Lavabo cue. TH/BB prepare only on the incense branch; others at base.}

\SungMovementPanel{4. Keep the Lavabo lane clear (MC-032--034)}
{\SungLavaboPositions}
{With incense: the priest has been incensed. Without it: he reaches the Lavabo place.}
{A1: towel; A2: water and bowl.}
{Use the inner credence lane, bow together, wash and dry the fingers, bow
again, and return by that lane.}
{Every article is reset; both acolytes rejoin their base places.}
{On the incense route TH uses the outer lane and waits if the pair has not returned. BB's boat stays safe.}

\SungMovementPanel{5. Settle before the elevations (MC-037--042)}
{\SungCanonPositions}
{End of the Preface; then the Sanctus and the MC's warning cue.}
{T1--T4: four adult-lit torches; A1: bell; TH: thurible if used.}
{Torchbearers go via the centre to four marked places and kneel. A1
kneels separately at the bell; TH takes the Epistle incense station.}
{Hold still through both elevations. Ordinary torches return at MC-041;
the Communion or fast-day branch waits until MC-048.}
{Other servers kneel at base. Non-torch servers rise at the Lord's Prayer;
extended torchbearers remain kneeling.}

\SungMovementPanel{6. Two ablutions, two assigned pourers (MC-048--049)}
{\MCAblutionPositions}
{The priest presents the chalice for the first ablution.}
{A1: wine; A2: water.}
{Credence to the Epistle handoff. A1 first pours wine into the chalice.
Then A1 pours wine over the fingers; after he stops, A2 pours water.}
{Both cruets return to the credence; the book route becomes clear.}
{MC waits for the book-transfer cue. Extended torchbearers return after
the first ablution; the second does not wait for their round trip.}

\SungMovementPanel{7. Return the altar book (MC-050)}
{\MCBookReturnPositions}
{Both ablutions are complete and the centre lane is clear.}
{MC: Missal and stand in both hands.}
{Gospel end down to centre stop A; make the taught reverence, then
continue to the Epistle end. Keep the book supported throughout.}
{Place the book at the Introit angle and return to the book-side station.}
{A1/A2 have reset the cruets. Other servers remain at their named places;
no one takes a second copy of this job.}

\SungMovementPanel{8. Form, reverence, then depart (MC-055--057)}
{\SungDeparturePositions}
{The Last Gospel ends, or its appointed omission cue.}
{MC: biretta; A1/A2: candles; TH/BB and CB: their exit objects if assigned.}
{Go to the stationary reverence line. Object bearers use their taught
reverence; then turn and take the same single-file order as the entrance.}
{MC gives the biretta to the priest; every object returns to its named
sacristy stand or adult receiver.}
{T1--T4 and optional L are empty-handed. The adult handles flames and charcoal.}
\fi

\ifSolemnMassGuide
\SungMovementPanel{1. Sacred ministers answer at the foot (SO-006--009)}
{\SolemnFootPositions}
{The entrance objects and birettas have been placed.}
{All lay servers; free hands. C, D and SD are the sacred ministers.}
{From the arrival line to the marked base places; kneel on both knees
while the sacred ministers give their private dialogue.}
{Remain silent and follow only the assigned gestures; clear the ascent.}
{Every lay role has a base place. A1/A2 candles and the optional cross
are already on their stands.}

\SungMovementPanel{2. Give and recover the Epistle book (SO-016--017)}
{\SolemnBookPositions}
{The last Collect begins.}
{MC: Epistle book; SD receives and chants.}
{Credence to SD, then accompany him through centre stop A to the
Epistle station. Remain beside him; recover the book on his return.}
{The book is back at the credence; MC takes his marked bench place.}
{Other lay servers sit at the named bench for the Epistle and chants.
All rise at the Gospel-preparation cue.}

\SungMovementPanel{3. Travel single file; form at the Gospel (SO-018--024)}
{\SolemnGospelRoutePositions}
{D rises with the Gospel book after his blessing; MC signals.}
{MC, TH, A1, A2, SD, D in that order; D carries the book.}
{Meet at the centre, make the common reverence, and travel one behind
another to the Gospel station. There A1/A2 separate to flank SD's book.}
{After the Gospel the candles return to their stands. TH remains for
the celebrant's Gospel incensation before returning the thurible.}
{BB has reset the boat and stays at base with TB1--TB4 and optional CB.
Use the illustrated Gospel close view for D, SD and MC's facing.}

\SungMovementPanel{4. Keep the sacred vessels with the ministers (SO-030--034)}
{\SolemnOffertoryPositions}
{The Offertory Oremus and MC's credence cue.}
{MC: humeral veil; A1: both cruets; A2: chalice veil.}
{MC vests SD with the humeral veil and steps aside. A2 receives the
chalice veil at the credence. A1 follows SD to the altar with both cruets.}
{A1 receives both cruets back from SD and resets them. A2 has folded
and placed the veil at its named credence place.}
{TH/BB prepare incense. All keep SD's central paten route open; lay
servers do not take the chalice or paten.}

\SungMovementPanel{5. Lavabo and incense use separate lanes (SO-037--040)}
{\SungLavaboPositions}
{The celebrant has been incensed; the rest of the incense route continues.}
{A1: towel; A2: water in the right hand, basin in the left.}
{Use the inner credence lane. Bow, wash and dry the fingers, bow again,
and return by that same lane while D and TH continue outward.}
{Reset the articles and rejoin the server line before TH reaches the
acolytes. If delayed, hold clear while TH waits.}
{BB remains at base with the boat parked. MC keeps the centre open.}

\SungMovementPanel{6. Four torches, one bell place (SO-041--049)}
{\SungCanonPositions}
{The adult lights the torches during the Preface; TH brings the group back.}
{TB1--TB4: torches; A1: bell; TH: thurible. T1--T4 in the picture are TB1--TB4.}
{Form for the common reverence, separate to the four marked places,
and kneel. TH kneels at the Epistle station; A1 stays at the bell place.}
{TH returns independently after the chalice elevation. Ordinary torches
also return then; Communion or fast-day torches remain until SO-057.}
{MC, A2, BB and optional CB kneel at base. BB has returned the boat
after the unblessed incense preparation. D and SD keep their own offices.}

\SungMovementPanel{7. SD pours; A1 supplies (SO-057--060)}
{\SolemnAblutionPositions}
{First ablution after Communion.}
{A1: both cruets; SD: pouring; A2: folded chalice veil afterwards.}
{A1 brings the cruets to the local handoff point and supplies SD.
SD pours first wine into the chalice, then wine and water over C's fingers.}
{A1 resets both cruets. A2 places the folded veil at the prepared Gospel-side
point and returns to base; SD returns the covered chalice to the credence.}
{MC clears the ministers' route. Extended torchbearers return after the
first ablution. The lay servers do not take over either minister's job.}

\SungMovementPanel{8. Return every object to its owner (SO-067--070)}
{\SungDeparturePositions}
{The normal Last Gospel ends, or the appointed omission cue.}
{MC: three birettas; BB: boat; A1/A2: candles; CB: cross if used. TH is empty-handed.}
{Form at the final reverence line, face the altar, make the assigned
reverence, then turn into the same single-file order as the entrance.}
{MC gives D his own and C's birettas, and gives SD his. In the sacristy
replace the cross and candles; BB gives the boat to the adult and TH checks
the thurible already at its safe station.}
{TB1--TB4 are empty-handed. No one departs or unvests before the
concluding reverence and the accounting for people and objects.}
