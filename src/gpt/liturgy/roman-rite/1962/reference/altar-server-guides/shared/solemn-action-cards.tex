\begin{flashgrid}
\actionflashback{SO-A02}{Three blessed stations; one unblessed}{%
\textbf{1. First incense:} boat to its marked station; after the celebrant is
incensed, closed thurible to its safe station.\par
\textbf{2. Gospel:} boat to its marked station and Bb to base; after the
celebrant is incensed, closed thurible to its safe station.\par
\textbf{3. Offertory:} boat at its marked station; after the people, closed
thurible to the metal-safe sacristy station.\par
\textbf{4. Elevations:} boat to its marked station and Bb to base; after the
chalice, closed thurible to the metal-safe station.}
{Trainer reference: Solemn Mass; Th/Bb/MC; SO-010--011, 018--025, 035--040,
044--048, 067.}
&
\actionflashback{SO-A01}{Six checks; one clear exit}{%
\textbf{1. Setup:} check every object and route.\par
\textbf{2. Entrance:} receive the birettas.\par
\textbf{3. Lessons:} cue the books and seats.\par
\textbf{4. Gospel return:} separate the book lane from the candle lane.\par
\textbf{5. Elevations:} load Th's open thurible with unblessed incense.\par
\textbf{6. Ending:} call the ending and form the exit. Give the deacon his and
the celebrant's birettas, and the subdeacon his; keep the middle clear.}
{Trainer reference: Solemn Mass; MC; SO-001, 007, 013--017, 024, 044, 067.}\\[4pt]
\actionflashback{SO-A04}{Carry, return, rejoin A1}{%
\textbf{1. Candles:} serve the entrance and Gospel. At the exit, pick up after
the Last Gospel, or after the blessing at the third Mass of Christmas.\par
\textbf{2. Veil:} carry it at the Offertory and ablutions as assigned.\par
\textbf{3. Lavabo:} carry the water and basin.\par
\textbf{4. Pax and reset:} complete the pax duty and credence reset.\par
\textbf{5. Finish:} return each object before rejoining A1; place the exit
candle in its stand.}
{Trainer reference: Solemn Mass; A2; SO-003, 008, 020--024, 030, 033,
037--039, 053, 059--060, 066--067, 070.}
&
\actionflashback{SO-A03}{Serve each object; finish empty-handed}{%
\textbf{1. Candles:} serve the entrance and Gospel. At the exit, pick up after
the Last Gospel, or after the blessing at the third Mass of Christmas.\par
\textbf{2. Offertory:} serve the cruets and Lavabo towel.\par
\textbf{3. Canon:} ring the bells at the appointed cues.\par
\textbf{4. Communion:} receive the veil; if the faithful receive, use the bell
and then the plate.\par
\textbf{5. Ablutions and exit:} serve the cruets; return the exit candle to
its stand.}
{Trainer reference: Solemn Mass; A1; SO-003, 008, 020--024, 030, 032, 039,
043, 045--047, 050--051, 055--058, 066--067, 070.}\\[4pt]
\actionflashback{SO-A06}{Four lanes; every object disposed}{%
\textbf{1. Veil and cruets:} ready them before their calls.\par
\textbf{2. Covered paten:} keep its lane clear through the handoff.\par
\textbf{3. Incense:} complete the Offertory route; after the people, close the
thurible.\par
\textbf{4. Lavabo:} begin while the priest is incensed, use the inner lane,
and return before A1 and A2 are incensed.\par
\textbf{5. Finish:} dispose every object and take the Preface places.}
{Trainer reference: Solemn Mass; MC/Th/Bb/A1/A2; SO-029--040.}
&
\actionflashback{SO-A05}{Serve, form, flank, return separately}{%
\textbf{1.} Bb serves the incense, then returns to station or base---never to
the Gospel procession.\par
\textbf{2.} MC forms the Gospel group.\par
\textbf{3.} A1 and A2 flank the book while Th serves its incense.\par
\textbf{4.} Candles and MC's book use separate return lanes.\par
\textbf{5.} Th receives the thurible. After the chanted Gospel, do not add
\CanonLatin{Laus tibi, Christe}.}
{Trainer reference: Solemn Mass; Bb/MC/Th/A1/A2; SO-019, 021--025.}\\
\end{flashgrid}

\actionflashback{SO-A08}{Take only today's marked ending}{%
\textbf{1. Lent:} keep the appointed kneeling and Prayer-over-the-People
branch.\par
\textbf{2. Easter octave:} use the double \CanonLatin{alleluia}.\par
\textbf{3. \CanonLatin{Benedicamus}:} stop this route and use the parish's
separate procession sheet (not included).\par
\textbf{4. Blessing:} use the normal blessing; at the third Mass of Christmas,
omit the Last Gospel.\par
\textbf{5. Exit:} Th is empty-handed; Bb has the boat; optional Cb stands
between the candle acolytes; four torches are empty-handed; then MC and the
sacred ministers. Free hands use the reservation branch; object-bearers bow
the head.}
{Trainer reference: Solemn Mass; all roles; SO-061--070.}
&
\actionflashback{SO-A07}{Light four; ring; take the marked branch}{%
\textbf{1.} The adult lights and gives all four torches.\par
\textbf{2.} A1 gives one ring at the \CanonLatin{Sanctus}, one warning before
the Consecration, and three distinct strokes at each elevation.\par
\textbf{3.} After the chalice elevation, Th returns alone.\par
\textbf{4. Ordinary branch:} on a non-fast day without faithful Communion,
the torches leave then.\par
\textbf{5. Extended branch:} with faithful Communion, or on a fast day, remain
through the first wine. The adult receives and extinguishes all four.}
{Trainer reference: Solemn Mass; Tb1--Tb4/Th/A1/adult; SO-041, 043, 045--048,
057.}\\[4pt]
\actionflashback{SO-A10}{Mark it; call it; have everyone repeat}{%
\textbf{1.} Mark the opening route, ferial and proper-text posture cues, and
Creed.\par
\textbf{2.} Mark faithful Communion, the torch branch, and cross-bearer.\par
\textbf{3.} Mark the ending and final reverence.\par
\textbf{4.} Call every choice; every server repeats it before lineup.\par
\textbf{5. Prescribed branch:} it is not a free choice. If
\CanonLatin{Benedicamus} is appointed, stop this route and use the parish's
separate procession sheet (not included).}
{Trainer reference: Solemn Mass; MC; SO-005; compare SO-063.}
&
\actionflashback{SO-A09}{Seven scenes; three facts for your role}{%
\textbf{1.} Ready and enter.\par
\textbf{2.} First incense and lessons.\par
\textbf{3.} Gospel and Creed.\par
\textbf{4.} Offertory.\par
\textbf{5.} Canon and elevations.\par
\textbf{6.} Pax, Communion, and ablutions.\par
\textbf{7.} Ending and recession.\par
\textbf{Trainer check:} use the assigned role sheet. Require the exact first
cue, object or empty hands, and named finish or next place; repeat a missed
scene after two others.}
{Trainer reference: Solemn Mass; all roles; scenes SO-001--009, 010--017,
018--029, 030--040, 041--049, 050--060, 061--070.}\\[4pt]
\actionflashback{SO-A12}{Name it; present it; stop safely}{%
\textbf{1. Before moving:} name the giver, receiver, grip, and final place.\par
\textbf{2. Handoff:} present the object; do not push it.\par
\textbf{3. Missed cue:} stop, secure the object, face the altar, and look to
MC. Never repair a miss with a rushed crossing.\par
\textbf{4. Finish:} return every object to its named place. The adult
extinguishes flames and handles hot charcoal.}
{Trainer reference: Solemn Mass; every role with partner/MC; role-sheet
four-pass recovery rule and SO-070 safe finish.}
&
\actionflashback{SO-A11}{Receive, answer, pass, clear, reset}{%
\textbf{1. Veil:} A1 receives and folds it.\par
\textbf{2. Pax:} A1 receives from the subdeacon, answers, and passes to A2;
A2 answers and ends the lay chain. MC keeps the route clear.\par
\textbf{3. Communion branch:} if the faithful receive, A1 uses the bell and
then the plate.\par
\textbf{4. Ablutions:} A1 serves both cruets; A2 moves the veil.\par
\textbf{5. Reset:} both clear and reset the credence. Keep every giver and
receiver named.}
{Trainer reference: Solemn Mass; A1/A2/MC; SO-050--051, 053, 055--060.}\\
\end{flashgrid}
