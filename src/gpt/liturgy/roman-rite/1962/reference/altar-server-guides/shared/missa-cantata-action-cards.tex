\begin{flashgrid}
\flashback{MC-A02}{Take, form, flank, return}{%
\textbf{1.} Take the candles together.\par
\textbf{2.} Form without crossing TH.\par
\textbf{3.} Stand level at the Gospel.\par
\textbf{4.} Return and replace the candles together.}
{Trainer reference: Missa Cantata; A1/A2; MC-023--026.} &
\flashback{MC-A01}{Check, cue, return, account}{%
\textbf{1.} Check every object and route; verify the adult candle handoff and
all entrance pickups.\par
\textbf{2.} Place the priest's biretta and cue the quiet response team.\par
\textbf{3.} For the exit, retrieve the biretta; after the final reverence give
it back to the priest and account for the team and objects.}
{Trainer reference: Missa Cantata; MC; MC-001, 003--010, 055--057.}\\[4pt]
\flashback{MC-A04}{Return now, or remain to first wine}{%
\textbf{1. Ordinary branch:} without faithful Communion on a non-fast day,
return after the chalice elevation.\par
\textbf{2. Extended branch:} with faithful Communion, or on a fast day, remain
through the first wine ablution.\par
\textbf{3.} Return in pairs; the adult extinguishes the torches.}
{Trainer reference: Missa Cantata; T1--T4; MC-037, 041, 048.} &
\flashback{MC-A03}{Both objects finish at the safe station}{%
\textbf{1. Altar:} BB returns the boat; after the priest is incensed, TH closes
the thurible. Both finish at the safe station.\par
\textbf{2. Gospel:} BB returns the boat before the procession; TH returns the
closed thurible after the Gospel. Both go to the safe station.\par
\textbf{3. Offertory:} after the people, TH closes the thurible; TH and BB
return both objects to the safe station.\par
\textbf{4. Elevations:} BB returns the boat before the Consecration; after the
chalice, TH closes and returns the thurible to the safe station.}
{Trainer reference: Missa Cantata; TH/BB; incense branch; MC-011, 022, 032, 040.}\\[4pt]
\flashback{MC-A06}{Quiet team; public singers}{%
\textbf{1. Quiet answers:} MC, A1, and A2 answer as the ministers' response
team in this model.\par
\textbf{2. Public sung answers:} choir or people sing.\par
\textbf{3. Assignment branch:} another server joins only when separately
assigned.}
{Trainer reference: Missa Cantata; quiet team and public singers;
MC-008, 017--018.} &
\flashback{MC-A05}{A1: wine and towel; A2: water and basin}{%
\textbf{1. Offertory:} A1 brings wine and towel; A2 brings water and basin.\par
\textbf{2. Lavabo cue:} with incense, begin after the priest is incensed;
without incense, MC cues his arrival.\par
\textbf{3. First ablution:} A1 pours wine.\par
\textbf{4. Second ablution:} A1 pours wine while A2 pours water over the
priest's fingers.}
{Trainer reference: Missa Cantata; A1/A2; MC-030--033, 048--049.}\\
\end{flashgrid}

\begin{flashgrid}
\flashback{MC-A08}{Stop guessing; hold; watch MC}{%
\textbf{1.} Stop instead of guessing.\par
\textbf{2.} Keep any object safe and still.\par
\textbf{3.} Watch MC and rejoin at the next cue you know.\par
\textbf{4.} Never invent a route.}
{Trainer reference: Missa Cantata; all roles; recovery; MC-003.} &
\flashback{MC-A07}{Go directly to \textit{Ecce Agnus Dei}}{%
\textbf{1.} Continue directly to \textit{Ecce Agnus Dei}.\par
\textbf{2.} Do not insert a second \textit{Confiteor}.\par
\textbf{3.} Do not insert \textit{Misereatur} or \textit{Indulgentiam}; none is
in the printed 1962 Communion rite.}
{Trainer reference: Missa Cantata; all roles; MC-046.}\\[4pt]
\flashback{MC-A10}{L goes and returns empty-handed}{%
\textbf{1.} Go empty-handed; the marked book is already on the lectern.\par
\textbf{2.} Chant there, leave the book there, and return empty-handed.\par
\textbf{3.} After the concluding sacristy reverence, the assigned adult
recovers the book; L confirms that it is accounted for.}
{Trainer reference: Missa Cantata; L branch; MC-001, 020, 057.} &
\flashback{MC-A09}{Free hands genuflect; candles stay upright}{%
\textbf{1.} MC, TH with the closed thurible, and every other free-handed server
face the altar.\par
\textbf{2.} They genuflect with the priest.\par
\textbf{3. Candle branch:} A1 and A2 remain upright with their candles.}
{Trainer reference: Missa Cantata; prescribed Gospel condition; MC-025.}\\[4pt]
\flashback{MC-A12}{Keep the blessing; omit the Last Gospel}{%
\textbf{1.} Complete the blessing.\par
\textbf{2.} Omit the Last Gospel.\par
\textbf{3.} Go directly to forming the recession.}
{Trainer reference: Missa Cantata; third Mass of Christmas; MC-053--055.} &
\flashback{MC-A11}{Stop this route; use the procession sheet}{%
\textbf{1.} Give the response \CanonLatin{Deo gratias}.\par
\textbf{2.} Stop the ordinary route: there is no blessing and no Last Gospel.\par
\textbf{3.} Follow the separately rehearsed procession sheet.}
{Trainer reference: Missa Cantata; procession-linked ending; MC-052.}\\
\end{flashgrid}
