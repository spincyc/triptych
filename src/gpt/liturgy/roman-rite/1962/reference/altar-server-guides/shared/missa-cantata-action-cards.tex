% Duplex action sheet 1, backs mirrored within each row.
\begin{flashgrid}
\flashback{MC-A02}{Take, form, flank, return}{Take candles together; form
without crossing TH; stand level at the Gospel; return and replace together.}
{Missa Cantata; A1/A2; MC-023--026.} &
\flashback{MC-A01}{Check, name, account}{Check objects and routes; verify the
adult candle handoff and every entrance pickup; place the priest's biretta and
retrieve it for the exit; after the final reverence give it back to the priest;
cue the quiet response team.}
{Missa Cantata; MC; MC-001, 003--010, 055--057.}\\[4pt]
\flashback{MC-A04}{MC-041 or MC-048}{Without faithful Communion on an ordinary
day, return after the chalice elevation. With faithful Communion, or on a fast
day, remain through the first wine ablution. Return in pairs; the adult
extinguishes the torches.}
{Missa Cantata; T1--T4; MC-037, 041, 048.} &
\flashback{MC-A03}{Altar, Gospel, Offertory, elevations}{Complete each boat and
thurible handoff. At Offertory TH uses the outer lane, incenses the named server
groups and people, and returns. Incense is added again for the elevations.}
{Missa Cantata; TH/BB; incense branch; MC-011, 022, 032, 040.}\\[4pt]
\flashback{MC-A06}{MC/A1/A2 quietly; choir or people publicly}{MC, A1, and A2
make the quiet minister's answers in this model. Choir or people sing the
public response; servers join only if assigned.}
{Missa Cantata; two voice tracks; MC-008, 017--018.} &
\flashback{MC-A05}{A1 wine and towel; A2 water and basin}{At the first
ablution A1 pours wine. At the second, A1 pours wine while A2 pours water over
the priest's fingers. With incense, the Lavabo begins after the priest is
incensed; without it, MC cues his arrival.}
{Missa Cantata; A1/A2; MC-030--033, 048--049.}\\
\end{flashgrid}

% Duplex action sheet 2, backs mirrored within each row.
\begin{flashgrid}
\flashback{MC-A08}{Stop guessing; hold; watch MC}{Keep the object safe and
still, watch the MC, and rejoin at the next known cue. Never invent a route.}
{Missa Cantata; all roles; recovery; MC-003.} &
\flashback{MC-A07}{Directly to \textit{Ecce Agnus Dei}}{There is no second
\textit{Confiteor}, \textit{Misereatur}, or \textit{Indulgentiam} in the
printed 1962 Communion rite.}
{Missa Cantata; all roles; MC-046.}\\[4pt]
\flashback{MC-A10}{L goes and returns empty-handed}{The marked book is already
on the lectern. L chants there and leaves it there. After the concluding
sacristy reverence, the assigned adult recovers it and L confirms it is
accounted for.}
{Missa Cantata; L branch; MC-001, 020, 057.} &
\flashback{MC-A09}{Free hands and TH move; candles stay upright}{MC, TH with
the closed thurible, and every other free-handed server face the altar and
genuflect with the priest. A1/A2 remain upright with their candles.}
{Missa Cantata; prescribed Gospel condition; MC-025.}\\[4pt]
\flashback{MC-A12}{No Last Gospel}{The blessing remains. At its completion,
omit the Last Gospel and go directly to forming the recession at MC-055.}
{Missa Cantata; third Mass of Christmas; MC-053--055.} &
\flashback{MC-A11}{Use the separate procession sheet}{After
\textit{Benedicamus Domino} and \textit{Deo gratias}, the ordinary route stops:
there is no blessing and no Last Gospel.}
{Missa Cantata; procession-linked ending; MC-052.}\\
\end{flashgrid}
