\section*{Scriptural Date and Location}

\begin{dossiertable}
\dossierentry{Offertory}{\label{chronology-offertory}}{Ps.~39:2--4 (modern 40)}{Davidic title; Israel's assembly}
{\chronodate{offertory}{Undated in the traditional chronology corpus.}}
{Modern: thanksgiving--lament before Israel's assembly (USCCB NABRE Ps.~40); first audience, composition place, and Davidic stage unrecovered (Clementine; Challoner; Catholic Encyclopedia, ``Psalms'').}
\dossierseparator
\dossierentry{Introit}{\label{chronology-introit}}{Ps.~85:1--4 (modern 86)}{Davidic prayer addressed to God}
{\chronodate{introit}{Undated in the traditional chronology corpus.}}
{Tradition: David, praying to God. Modern: Israelite lament; first audience, place, life-stage, and date unknown (Clementine; Challoner; USCCB NABRE Ps.~86; chronology record).}
\dossierseparator
\dossierentry{Gradual}{\label{chronology-gradual}}{Ps.~91:2--3 (modern 92)}{Sabbath canticle; no human author}
{\chronodate{gradual}{Undated in the traditional chronology corpus.}}
{Modern: a protected speaker's praise in Israel's Sabbath worship (USCCB NABRE Ps.~92); first congregation, composition place, and authorial stage unrecovered (Clementine; Catholic Encyclopedia, ``Psalms'').}
\dossierseparator
\dossierentry{Alleluia}{\label{chronology-alleluia}}{Ps.~94:3 (modern 95), adapted with Ps.~46:3}{Israel's corporate worship and kingship}
{\chronodate{alleluia}{Undated in the traditional chronology corpus.}}
{Modern: distinct corporate worship and kingship hymns in Israel (USCCB NABRE Pss.~95, 47); first congregations, places, and stages uncertain (Clementine: Ps.~94 Davidic, Ps.~46 Core's sons; Catholic Encyclopedia, ``Psalms'').}
\dossierseparator
\dossierevententry{Gospel}{\label{chronology-gospel}}{Luke~7:11--16}{Event: Naim, Galilee; tradition: Achaia or Boeotia; modern city uncertain}
{\chronodate{gospel}{Disputed: \chronology{composition.gospel-of-luke}{composition}{About the year 70}; \chronology{composition.gospel-of-luke}{composition}{before the end of the Roman imprisonment, when the Acts was finished}.}}
{Event: Jesus raises the youth at Naim after Capharnaum; undated (USCCB NABRE Luke~7; Gadenz, 141--143; chronology record).}
{Tradition: Luke, Antiochene physician and Paul's companion, in Achaia/Boeotia. Modern: after Jerusalem's fall, a non-eyewitness apostolic Gospel for Theophilus and Gentiles outside Palestine; city uncertain (Catholic Encyclopedia; Luke~1:1--4; USCCB NABRE Luke introduction; Gadenz, 16--19).}
\dossierseparator
\dossierevententry{Communion}{\label{chronology-communion}}{John~6:52 (modern 6:51b)}{Event: Capharnaum; tradition: Ephesus; modern city uncertain}
{\chronodate{communion}{Disputed: \chronology{composition.gospel-of-john}{composition}{from the year 90 to 100 (approximately)}; \chronology{composition.gospel-of-john}{composition}{96 or one of the succeeding years}.}}
{Event: near Passover, from the Sea of Galilee to Capharnaum; undated (USCCB NABRE John~6; chronology record).}
{Tradition: aged apostle John at Ephesus for later Asian Christians. Modern: an edited Gospel amid synagogue conflict, probably in the 90s but in an unknown city; apostolic witness to the incarnate Christ (Catholic Encyclopedia; USCCB NABRE John introduction).}
\dossierseparator
\dossierentry{Epistle}{\label{chronology-epistle}}{Gal.~5:25--6:10}{Galatian churches; north or south disputed}
{\chronodate{epistle}{Disputed: \chronology{composition.epistle-to-the-galatians}{composition}{A.D. 49-50}; \chronology{composition.epistle-to-the-galatians}{composition}{about A.D. 53, 54}; \chronology{composition.epistle-to-the-galatians}{composition}{(1 and 2 Corinthians; Galatians), 56}; \chronology{composition.epistle-to-the-galatians}{composition}{A.D. 57 or 58}.}}
{Paul, during post-Paschal missionary apostleship, addresses Gentile Galatians. Modern: north/Ancyra, Ephesus, and the third journey favored; south and an earlier mission remain; place/date disputed (USCCB NABRE Galatians introduction; Catholic Encyclopedia; chronology record).}
\end{dossiertable}
