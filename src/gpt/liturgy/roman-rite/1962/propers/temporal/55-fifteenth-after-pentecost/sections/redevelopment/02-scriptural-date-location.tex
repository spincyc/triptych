\section*{Scriptural Date and Location}

\begin{dossiertable}
\dossierentry{Offertory}{\label{chronology-offertory}}{Ps.~39:2--4 (modern 40)}{Davidic title; Israel's assembly}
{\chronodate{offertory}{\chronologyannotation{offertory}}}
{Modern: thanksgiving--lament before Israel's assembly; first audience, composition place, and Davidic stage uncertain (Clementine; Challoner; USCCB NABRE Ps.~40; Catholic Encyclopedia, ``Psalms'').}
\dossierseparator
\dossierentry{Introit}{\label{chronology-introit}}{Ps.~85:1--4 (modern 86)}{Davidic prayer addressed to God}
{\chronodate{introit}{\chronologyannotation{introit}}}
{Tradition: David praying to God. Modern: an Israelite lament; first audience, place, and Davidic life-stage uncertain (Clementine; Challoner; USCCB NABRE Ps.~86; Catholic Encyclopedia, ``Psalms'').}
\dossierseparator
\dossierentry{Gradual}{\label{chronology-gradual}}{Ps.~91:2--3 (modern 92)}{Sabbath canticle; no human author}
{\chronodate{gradual}{\chronologyannotation{gradual}}}
{Modern: a protected speaker's praise in Israel's Sabbath worship; first congregation, composition place, and authorial stage uncertain (Clementine; USCCB NABRE Ps.~92; Catholic Encyclopedia, ``Psalms'').}
\dossierseparator
\dossierentry{Alleluia}{\label{chronology-alleluia}}{Ps.~94:3 (modern 95), adapted with Ps.~46:3}{Israel's corporate worship and kingship}
{\chronodate{alleluia}{\chronologyannotation{alleluia}}}
{Modern: distinct corporate-worship and kingship hymns; first congregations, places, and stages uncertain (Clementine: Ps.~94 Davidic, Ps.~46 Core's sons; USCCB NABRE Pss.~95, 47; Catholic Encyclopedia, ``Psalms'').}
\dossierseparator
\dossierevententry{Gospel}{\label{chronology-gospel}}{Luke~7:11--16}{Event: Naim, Galilee; tradition: Achaia or Boeotia; modern city uncertain}
{\chronodate{gospel}{\chronologyannotation{gospel}}}
{Event: Christ raises the widow's son at Naim during the third journey, after returning from Capharnaum (Catholic Encyclopedia, ``Jesus Christ''; USCCB NABRE Luke~7; Gadenz, 141--143).}
{Tradition: Luke, Paul's companion, in Achaia/Boeotia; the Pontifical Biblical Commission places composition before Acts and the close of Paul's Roman imprisonment. Modern: an apostolic Gospel after Jerusalem's fall, for Theophilus and Gentiles outside Palestine; city uncertain (Catholic Encyclopedia; Luke~1:1--4; USCCB NABRE Luke introduction; Gadenz, 16--19).}
\dossierseparator
\dossierevententry{Communion}{\label{chronology-communion}}{John~6:52 (modern 6:51b)}{Event: Capharnaum; tradition: Ephesus; modern city uncertain}
{\chronodate{communion}{\chronologyannotation{communion}}}
{Event: near Passover in the fifth journey, Christ delivers the Bread of Life discourse at Capharnaum (Catholic Encyclopedia, ``Jesus Christ''; USCCB NABRE John~6).}
{Tradition: aged apostle John at Ephesus for Asian Christians. Modern: an edited late-first-century Gospel amid synagogue conflict; city uncertain (Catholic Encyclopedia; USCCB NABRE John introduction).}
\dossierseparator
\dossierentry{Epistle}{\label{chronology-epistle}}{Gal.~5:25--6:10}{Galatian churches; north or south disputed}
{\chronodate{epistle}{\chronologyannotation{epistle}}}
{Paul addresses Gentile Galatians during his missionary apostleship. Baronius, through a Lapide, supplies the preferred year; the Catholic Encyclopedia preserves four alternatives. Northern or southern addressees and the place remain disputed (a Lapide, \emph{Prooemium} and \emph{Argumentum}; USCCB NABRE Galatians introduction; Catholic Encyclopedia).}
\end{dossiertable}
