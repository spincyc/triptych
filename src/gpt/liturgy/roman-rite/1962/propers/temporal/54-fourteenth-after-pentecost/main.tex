\weektitle{Fourteenth Sunday after Pentecost}{DOMINICA DECIMA QUARTA post Pentecosten}{1962 Roman Missal propers: undivided service, Spirit-led freedom, filial providence, Eucharistic purification, and the kingdom sought first}

\begin{properrecords}
\properrecord{Introit}{\latin{Protéctor noster, áspice, Deus} (Ps.~83~(84):10--11a, 2--3a)}{Protector, Christ, courts}{The assembly asks God to behold the face of his Christ and declares one day in God's courts better than thousands.}
\properrecord{Collect}{\latin{Custódi, Dómine, quǽsumus}}{Mercy directs mortal life}{Because unaided mortality slips, the Church asks perpetual favor, withdrawal from harms, and direction toward saving things.}
\properrecord{Epistle}{Gal.~5:16--24}{Spirit against disordered flesh}{Paul contrasts works issuing from disordered desire with the Spirit's one fruit and the crucifixion of passions in those who belong to Christ.}
\properrecord{Gradual}{\latin{Bonum est confidere in Dómino} (Ps.~117~(118):8--9)}{Final trust in God}{Confidence rests in the Lord; Augustine places human and angelic helpers under the God who gives them power (Ps.~117, §4).}
\properrecord{Alleluia}{\latin{Veníte, exsultémus Dómino} (Ps.~94~(95):1)}{Joyful approach}{The call to praise moves the assembly toward the Gospel and altar under the psalm's larger summons to obedient hearing.}
\properrecord{Gospel}{Matt.~6:24--33; Credo}{One master; the kingdom first}{Christ rejects divided service and anxious mastery, then grounds confidence in the Father's knowledge and orders necessities beneath the kingdom.}
\properrecord{Offertory}{\latin{Immíttet Angelus Dómini} (Ps.~33~(34):8--9a)}{Protection and tasting}{The Lord's angel encamps around those who fear him; the invitation to taste his goodness accompanies the offering.}
\properrecord{Secret}{\latin{hæc hóstia salutáris}}{Victim, purification, propitiation}{The Church asks that the saving victim purify sins and stand as propitiation before divine power.}
\properrecord{Communion}{\latin{Primum quǽrite regnum Dei} (Matt.~6:33)}{Priority repeated}{Reception is governed by the Gospel's command: seek the kingdom first; necessities are added, not enthroned.}
\properrecord{Postcommunion}{\latin{Puríficent semper et múniant}}{Purified, strengthened, led}{The sacraments are asked always to purify and strengthen the recipients and lead them to perpetual salvation's effect.}
\end{properrecords}
\begin{senserecords}
\senserecord{Literal}{Pilgrims desire God's courts; Paul describes moral conflict and Spirit-given fruit; Christ orders food and clothing beneath the Father's kingdom; the Church asks mercy, purification, and perseverance.}
\senserecord{Allegorical}{Augustine asks that Christ be made known to all in the Introit's anointed face (Ps.~83, §13), and hears Christ's Body and Blood in the Offertory's tasting (Ps.~33, second exposition, §11).}
\senserecord{Moral}{Refuse divided mastery, walk by the Spirit, accept creaturely help without making it ultimate, work without anxious servitude, and seek God's reign and justice before every subordinate good.}
\senserecord{Anagogical}{Augustine reads the better day as everlasting life (Ps.~83, §14); the Postcommunion asks the sacraments to lead the recipients to perpetual salvation.}
\end{senserecords}
