\section*{The Propers: Interpretive Possibilities}

\noindent\emph{The following are exploratory editorial or AI proposals. They
are not attributed teaching, historical intent, or evidence about the
formulary's compiler. Each states its anchors, mechanism, fruit, nearest checked
analogue, and controlling limit.}

\begin{proposal}{Ministry begins where the traveller leaves}
\pfield{Anchors}{Epistle, Gospel, Postcommunion.}
\pfield{Mechanism}{Paul is made a fit minister; the Samaritan entrusts the
wounded man to an innkeeper for continuing care; the Postcommunion asks holy
participation to vivify and to grant expiation and protection to its
recipients.}
\pfield{Fruit}{Mercy needs durable ministries, not only a first responder. The
Church receives care from Christ in order to continue it responsibly.}
\pfield{Precedent result}{\textbf{Near analogue located.} Ambrose 7.82--84,
Augustine \work{Questions on the Gospels} 2.19, and Bede identify the inn and
its keeper ecclesially; none joins Paul's sufficiency or this Postcommunion.}
\pfield{Limit}{No minister or institution may identify itself simply with the
Samaritan; ecclesial allegory remains subject to the literal command of mercy.}
\end{proposal}

\begin{proposal}{Promise is destination, covenant ground, and guaranty}
\pfield{Anchors}{Collect, Offertory, Gospel.}
\pfield{Mechanism}{The Church runs toward God's promises; Moses invokes the
oath to the patriarchs; the Samaritan promises to return and repay further
expense.}
\pfield{Fruit}{Christian hope may be read not as escape from wounded history
but as fidelity that sustains costly care within it.}
\pfield{Precedent result}{\textbf{Near analogue located.} Origen,
\work{Homily on Luke} 34, reports an elder's eschatological reading of the
return; Augustine, \work{Questions on the Gospels} 2.19, also maps the return
within his allegory. Checked witnesses do not join all three promise forms.}
\pfield{Limit}{The Samaritan's fictional undertaking is not a covenant equal
to God's oath, and the proposal proves no compositional design.}
\end{proposal}

\begin{proposal}{Mercy's longer clock}
\pfield{Anchors}{Gradual, Alleluia, Gospel.}
\pfield{Mechanism}{Praise is ``always,'' prayer continues day and night, care
passes into the next day and awaits return.}
\pfield{Fruit}{Compassion becomes a temporal discipline: attention must survive
the first emotion, the first expense, and the first liturgical moment.}
\pfield{Precedent result}{\textbf{Not located in the checked corpus.} Individual
witnesses interpret each temporal phrase, but no checked source joins this
three-part duration.}
\pfield{Limit}{The words arise in different literary units; they do not create a
schedule, and urgency in the Introit remains as important as duration.}
\end{proposal}

\begin{proposal}{A bounded question receives an open-ended answer}
\pfield{Anchors}{Gospel and Secret.}
\pfield{Mechanism}{\latin{Quis est meus proximus?} seeks the limit of the
person owed love; \latin{quodcúmque supererogáveris} opens expenditure beyond
the initial sum; \latin{indulgéntiam largiéndo} asks pardon to be lavished.}
\pfield{Fruit}{Mercy may resist the self-protecting move from ``whom must I
love?'' to the smallest defensible obligation.}
\pfield{Precedent result}{\textbf{Near analogue located.} Gospel commentators
stress universal neighbour-love and the innkeeper's further expense, but no
checked witness joins the Secret's gerund.}
\pfield{Limit}{Christian generosity requires prudence, justice, vocation, and
finite stewardship; \work{Fratelli tutti} 79--82 does not erase these goods.}
\end{proposal}

\begin{proposal}{Compassion and appeasement name two directions of mercy}
\pfield{Anchors}{Gospel, Offertory, Secret.}
\pfield{Mechanism}{The Samaritan is moved toward the sufferer; Moses speaks
before God for the guilty; the Secret asks favorable regard and pardon at the
altar.}
\pfield{Fruit}{Mercy can be both approach to the wounded and intercession for
the culpable, holding care and judgment in one prayer.}
\pfield{Precedent result}{\textbf{Near analogue located.} Patristic witnesses
read Christ in both Moses and the Samaritan, but no checked source constructs
this directional pairing with the Secret.}
\pfield{Limit}{God undergoes no emotional change: Augustine, \work{City of
God} 15.25, governs the language of anger and appeasement.}
\end{proposal}

\begin{proposal}{The medicine is neither only natural nor only sacramental}
\pfield{Anchors}{Gospel, Communion, Postcommunion.}
\pfield{Mechanism}{Oil and wine treat wounds; Psalm 103 praises bread, wine,
and oil as providential gifts; holy participation gives life and protection.}
\pfield{Fruit}{Bodily works of mercy and sacramental communion can be held
together without turning either into a decorative metaphor.}
\pfield{Precedent result}{\textbf{Near analogue located.} Ambrose and Augustine
connect the Samaritan's remedies to ecclesial healing; Augustine and
Cassiodorus receive Psalm 103 sacramentally. The exact three-element synthesis
is editorial.}
\pfield{Limit}{The psalm's literal created goods and the parable's practical
care must not be dissolved into Eucharistic allegory.}
\end{proposal}
