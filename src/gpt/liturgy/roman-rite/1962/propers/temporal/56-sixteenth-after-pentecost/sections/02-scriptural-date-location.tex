\section*{Scriptural Date and Location}
\begin{dossiertable}
\dossierentry{Offertory}{}{Ps.~39 (40):14--15, selected}{Davidic title; an endangered petitioner.}{\chronodate{offertory}{\chronologyannotation{offertory}}}{Corbett's ``usual chronology'' supplies the Davidic regnal frame (\emph{Catholic Encyclopedia}, IV, p.~642). Rescue and obedience return to threatened life; writing-place and enemy are unnamed. USCCB's composition bound is shared by the Psalter, not an individual date. \sourcecite{Ps.~39:1--18; USCCB, Psalms introduction.}}
\dossierseparator
\dossierentry{Communion}{}{Ps.~70 (71):16--18, selected}{An aging speaker; Davidic and captivity settings remain distinct.}{\chronodate{communion}{\chronologyannotation{communion}}}{Youth, failing strength and future hearers frame Bellarmine's Davidic reading; writing-place is unestablished. The composition bound is general; the alternate A.M. figure concerns the superscription, not the preferred historical setting. \sourcecite{Ps.~70:1--24; Bellarmine, pp.~212--213; chronology sources in References.}}
\dossierseparator
\dossierentry{Introit}{}{Ps.~85 (86):3,5; v.~1}{Davidic title; a needy servant threatened by attackers.}{\chronodate{introit}{\chronologyannotation{introit}}}{The title names David, without a particular crisis or writing-place. Prayer opens toward the nations' worship; individual composition remains insecure within the common critical bound. \sourcecite{Ps.~85:1--17; USCCB, Psalms introduction.}}
\dossierseparator
\dossierentry{Alleluia}{}{Ps.~97 (98):1, opening}{Israel's salvation before the nations and creation.}{\chronodate{alleluia}{\chronologyannotation{alleluia}}}{The Douay title names David, without a particular rescue or writing-place. The composition bound is common; the Nativity dates concern prophecy, with Gerard reporting one as others' opinion. \sourcecite{Ps.~97:1--9; USCCB, Psalms introduction; Gerard, ``General Chronology,'' ``Christian era,'' first paragraph; ``Chronology of the Life of Jesus Christ.''}}
\dossierseparator
\dossierentry{Gradual}{}{Ps.~101 (102):16--17}{Sion/Jerusalem restored; peoples gathered for worship.}{\chronodate{gradual}{\chronologyannotation{gradual}}}{The afflicted speaker contrasts his frailty with God's permanence. Jerusalem is the subject; writing-place, restoration date and individual composition date are unestablished. \sourcecite{Ps.~101:1--29; USCCB, Psalms introduction.}}
\dossierseparator
\dossierevententry{Gospel}{}{Luke~14:1--11}{St Luke's Gospel; an unnamed leading Pharisee's house.}{\chronodate{gospel}{\chronologyannotation{gospel}}}{Event: Sabbath meal and cure in the journey-to-Jerusalem context (Luke~13:22; 14:1--11); town and host unnamed. The wedding is the seating parable's setting.}{Composition alternatives follow the \textit{Catholic Encyclopedia} and Biblical Commission; writing-place, first-audience geography and a full modern authorship account remain unestablished. USCCB notes son/ox beside the ass/ox variant retained by the Missal.}
\dossierseparator
\dossierentry{Epistle}{}{Eph.~3:13--21}{St Paul, imprisoned for his Gentile mission; saints sharing one inheritance.}{\chronodate{epistle}{\chronologyannotation{epistle}}}{Affliction and the Gentile mission frame the prayer (Eph.~2--4). The Encyclopedia supplies the traditional dates; prison city, destination geography and a full modern authorship account remain unestablished. USCCB treats wording and interpretation.}
\end{dossiertable}
