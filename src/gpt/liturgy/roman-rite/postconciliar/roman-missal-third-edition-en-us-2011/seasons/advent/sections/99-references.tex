\section{Books to keep beside the handbook}

\subsection{The liturgy}

\begin{itemize}
\item \emph{The Roman Missal}, third edition, English translation for the United States (2011). Proper of Time, the Sundays of Advent, Prefaces I and II of Advent, and the four Masses of the Nativity. Read the collects, prayers over the offerings and prayers after Communion for the four Sundays in the \href{https://www.usccb.org/prayer-and-worship/liturgical-year-and-calendar/advent/commentary-on-advent-proper-prayers}{U.S. bishops' collection of Advent prayers with commentary}.
\item \emph{Lectionary for Mass for Use in the Dioceses of the United States of America}, second typical edition, nos.~1--12, 175--201, and the Christmas Masses, nos.~13--16. The \href{https://www.liturgyoffice.org.uk/Resources/GIRM/Documents/Lectionary.pdf}{General Introduction}, nos.~65--69, 93--94, explains the order of readings. The \href{https://bible.usccb.org/readings/calendar}{bishops' calendar of readings} gives the text at Mass for an actual day.
\item \href{https://www.liturgyoffice.org.uk/Resources/GIRM/Documents/GNLY.pdf}{Universal Norms on the Liturgical Year and the Calendar}, nos.~3--5, 39--42, 45--47, and the table of precedence at no.~59. The \href{https://www.usccb.org/prayer-worship/liturgical-year}{U.S. liturgical calendar} gives territorial observances. For the responsorial psalm, Gloria, Creed and the choice of texts at a memorial, see the \href{https://www.usccb.org/prayer-and-worship/the-mass/general-instruction-of-the-roman-missal}{General Instruction of the Roman Missal}, nos.~53, 61, 68, 355, 357.
\item \emph{The Liturgy of the Hours}, English edition for the United States, volume I, Proper of Seasons, Advent. See the \href{https://www.liturgyoffice.org.uk/Resources/Rites/GILH.pdf}{General Instruction on the Liturgy of the Hours}, nos.~37--56, 59, 64--65, and the \href{https://www.usccb.org/prayer-worship/liturgical-year/advent}{U.S. bishops' introduction to Advent}, with the O Antiphons. See also the Congregation for Divine Worship, \href{https://www.usccb.org/resources/Homiletic-Directory-CDWDS.pdf}{\emph{Homiletic Directory}}, nos.~78--96.
\item For the comparison, \href{https://media.churchmusicassociation.org/pdf/missale62.pdf}{\emph{Missale Romanum}}, Vatican typical edition, 1962, Proper of Time, printed pp.~1--16. See the two \href{https://mystago.gy/library/traditional-latin-mass.html#seasonal-handbooks}{Triptych handbooks for the 1962 books} for the older Mass and Office and their own daily course.
\end{itemize}

\subsection{Prayer and Christian life}

\begin{itemize}
\item Second Vatican Council, \href{https://www.vatican.va/archive/hist_councils/ii_vatican_council/documents/vat-ii_const_19631204_sacrosanctum-concilium_en.html}{\emph{Sacrosanctum Concilium}}, nos.~7, 51, 83--84, 100, 102. Christ in his praying Church, the riches of Scripture at Mass, common prayer and the mystery unfolded in the year.
\item St Paul VI, \href{https://www.vatican.va/content/paul-vi/en/apost_exhortations/documents/hf_p-vi_exh_19740202_marialis-cultus.html}{\emph{Marialis cultus}}, nos.~3--4, 41. Mary in the Advent liturgy and the prayer of the Angelus. Pope Francis, \href{https://www.vatican.va/content/francesco/en/apost_letters/documents/papa-francesco-lettera-ap_20191201_admirabile-signum.html}{\emph{Admirabile signum}}, nos.~5--8, on coming to the crib.
\item \emph{Catechism of the Catholic Church}, no.~524 on Israel's expectation, and \href{https://www.vatican.va/content/catechism/en/part_two/section_two/chapter_two/article_4/vii_the_acts_of_the_penitent.html}{nos.~1451--1460} on confession and reparation. \href{https://www.vatican.va/archive/cod-iuris-canonici/eng/documents/cic_lib4-cann1244-1253_en.html}{1983 Code of Canon Law}, canons 1249--1253, and the \href{https://www.usccb.org/prayer-and-worship/liturgical-year-and-calendar/lent/us-bishops-pastoral-statement-on-penance-and-abstinence}{U.S. bishops' pastoral statement}, especially nos.~7, 22--25.
\end{itemize}
