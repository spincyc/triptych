\section*{The Appointed Texts: Rights-Limited Study Sheet}

\begin{studybox}{Text boundary}
The controlling U.S. Missal and Lectionary English is protected and is not
reproduced here. This sheet gives the complete ordered inventory, lawful
Latin incipits for the orations and antiphons, public-domain scriptural loci,
and concise descriptions sufficient to identify every branch. It is a study
aid, not a substitute for the approved liturgical books.
\end{studybox}

\begin{variabletextstable}
\variabletextgroup{Introductory Rites}
\variabletextrow{Entrance Antiphon}{\emph{Cf.} Ps.~67:6--7, 36 (modern 68);
\latin{Deus in loco sancto suo}; God gives the forsaken a dwelling and
strengthens his people.}
\variabletextrow{Collect}{\latin{Protector in te sperantium, Deus}; God is
asked to protect, govern, and guide his people so that passing goods are used
without losing those that endure.}
\variabletextgroup{Liturgy of the Word}
\variabletextrow{First Reading}{1~Kgs.~3:5, 7--12; Solomon asks for a listening
heart capable of discerning good and evil for the people entrusted to him.}
\variabletextrow{Responsorial Psalm}{Ps.~119:57, 72, 76--77, 127--130, with
response from v.~97a; divine instruction is preferred to wealth and asked to
give understanding.}
\variabletextrow{Second Reading}{Rom.~8:28--30; within suffering and
Spirit-aided prayer, divine purpose tends toward conformity to the Son.}
\variabletextrow{Gospel Acclamation}{\emph{Cf.} Matt.~11:25; the Father
reveals the mysteries of the Kingdom to the little ones.}
\variabletextrow{Gospel}{Long branch: Matt.~13:44--52, treasure, pearl, net,
sorting, and the instructed householder. Short branch: Matt.~13:44--46,
treasure and pearl only.}
\variabletextgroup{Liturgy of the Eucharist}
\variabletextrow{Prayer over Offerings}{\latin{Suscipe, quaesumus, Domine,
munera}; gifts received from divine bounty are returned, with a petition that
the mysteries sanctify present life.}
\variabletextrow{Communion Antiphon A}{Ps.~102:2 (modern 103); the soul blesses
the Lord and refuses to forget his benefits.}
\variabletextrow{Communion Antiphon B}{Matt.~5:7--8; the merciful and the
clean of heart are named with their distinct promises.}
\variabletextrow{Prayer after Communion}{\latin{Sumpsimus, Domine, divinum
sacramentum}; the sacrament is confessed as the perpetual memorial of the
Son's Passion and its saving fruit is sought.}
\end{variabletextstable}
