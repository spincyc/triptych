\subsection{Ancient witnesses}

\begin{itemize}
\item Clement of Rome. \emph{First Epistle to the Corinthians}, 5, especially
  5.4. Translated by John Keith in \emph{Ante-Nicene Fathers}, vol.~9, edited
  by Allan Menzies (New York: Christian Literature Company, 1896).
  Public-domain English translation at
  \sourceurl{https://en.wikisource.org/wiki/Ante-Nicene_Fathers/Volume_IX/The_Epistles_of_Clement/The_First_Epistle_of_Clement_to_the_Corinthians/Chapter_5}{Wikisource, chapter 5},
  accessed 15 July 2026. The unusual Wikisource volume path reproduces its
  hosted organization; only the cited chapter was used.

\item Ignatius of Antioch. \emph{Letter to the Romans}, 4.3, middle recension.
  In \emph{Ante-Nicene Fathers}, vol.~1. Public-domain patristic translation,
  revised web presentation at
  \sourceurl{https://www.newadvent.org/fathers/0107.htm}{New Advent}, accessed
  15 July 2026. Only a short phrase was quoted.

\item Irenaeus of Lyons. \emph{Against Heresies} 3.1.1 and 3.3.2--3. In
  \emph{Ante-Nicene Fathers}, vol.~1. Public-domain English translation at
  \sourceurl{https://ccel.org/ccel/irenaeus/against_heresies_iii/anf01.ix.iv.iv.html}{CCEL, 3.3},
  accessed 15 July 2026.

\item Eusebius of Caesarea. \emph{Ecclesiastical History} 2.25.5--8
  (Dionysius and Gaius); 3.1.2 (Origen on the deaths); 3.30.2 (Clement on
  Peter's wife); 3.39.15 (Papias on Mark). Translation by Arthur Cushman
  McGiffert, in \emph{Nicene and Post-Nicene Fathers}, second series, vol.~1
  (1890), public domain. Online text:
  \sourceurl{https://www.newadvent.org/fathers/250102.htm}{book 2} and
  \sourceurl{https://www.newadvent.org/fathers/250103.htm}{book 3}, accessed
  15 July 2026.

\item Tertullian. \emph{The Prescription Against Heretics}, 36; and
  \emph{Scorpiace}, 15. In \emph{Ante-Nicene Fathers}, vol.~3 (1885),
  public-domain English translations. Online text:
  \sourceurl{https://www.newadvent.org/fathers/0311.htm}{Prescription} and
  \sourceurl{https://www.newadvent.org/fathers/0318.htm}{Scorpiace}, accessed
  15 July 2026.

\item \emph{Acts of Peter}, especially 35--39. Translation by M.~R. James,
  \emph{The Apocryphal New Testament} (Oxford: Clarendon Press, 1924), public
  domain in the United States. Online text at
  \sourceurl{https://www.earlychristianwritings.com/text/actspeter.html}{Early Christian Writings},
  accessed 15 July 2026. The work is late-second-century apocrypha used for
  reception history, not a contemporary source.

\item Tacitus. \emph{Annals} 15.44. Translation by Alfred John Church and
  William Jackson Brodribb (1876), public domain. Online at
  \sourceurl{https://en.wikisource.org/wiki/The_Annals_(Tacitus)/Book_15}{Wikisource, book 15},
  accessed 15 July 2026. Tacitus supplies the Neronian persecution context and
  does not name either apostle.
\end{itemize}

\subsection{Magisterial, liturgical, and institutional sources}

\begin{itemize}
\item Second Vatican Council. Dogmatic Constitution \emph{Lumen gentium}
  (21 November 1964), especially 18--23. Vatican English text:

  \sourceurl{https://www.vatican.va/content/catechism/en.html}{Catechism index},
  accessed 15 July 2026.

\item Benedict XVI\@. General audiences on Peter, 17 and 24 May and 7 June 2006;
  homily for Saints Peter and Paul, 29 June 2008. Vatican English texts:
  \sourceurl{https://www.vatican.va/content/benedict-xvi/en/audiences/2006/documents/hf_ben-xvi_aud_20060517.html}{17 May},
  \sourceurl{https://www.vatican.va/content/benedict-xvi/en/audiences/2006/documents/hf_ben-xvi_aud_20060524.html}{24 May},

\item \emph{Depositio martyrum}, in the Chronography of 354, June entry,
Mommsen, ed., MGH \emph{Chronica minora} 1 (Berlin, 1892), pp.~71--72,
\sourceurl{https://www.tertullian.org/fathers/chronography_of_354_12_depositions_martyrs.htm}{Latin

  \sourceurl{https://www.basilicasanpietro.va/en/san-pietro/the-necropolis}{Necropolis},
  accessed 15 July 2026. These establish the shrine's official identification
  and presentation; archaeological inferences are separately graded.
\end{itemize}
