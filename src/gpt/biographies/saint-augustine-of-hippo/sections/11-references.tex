\section{References}\label{sec:aug-references}

\subsection{Ancient and official sources}

\begin{itemize}
\item Augustine. \emph{Confessionum libri XIII}. Ed. Pius Knöll, CSEL 33
(Vienna, 1896). Latin text and scan through the
\sourceurl{https://bkv.unifr.ch/en/works/cpl-251/versions/aug-conf-csel}{Bibliothek
der Kirchenväter}; accessed 2026-07-15. The old critical text is public domain;
this study paraphrases and gives loci rather than reproducing a modern
translation.
\item Augustine. \emph{Confessions}; \emph{De beata vita};
\emph{De magistro}; \emph{De doctrina christiana}; \emph{De Trinitate};
\emph{De civitate Dei}; \emph{Retractationes}; anti-Manichaean,
anti-Donatist, and anti-Pelagian works; sermons and letters. Edition
  orientation and volume map in the
\sourceurl{https://csel.at/en/publikationen/csel/}{CSEL catalog}; accessed
  2026-07-15. Exact loci used are mapped in the source audit. Individual
  critical editions and modern translations beyond the checked
  \emph{Confessions} base were not collated or quoted.
\item Possidius. \emph{Sancti Augustini Vita}. Trans. Herbert T. Weiskotten,
\emph{The Life of Saint Augustine} (Princeton, 1919),
\sourceurl{https://www.tertullian.org/fathers/possidius_life_of_augustine_02_text.htm}{English
and Latin text}; translator's
\sourceurl{https://www.tertullian.org/fathers/possidius_life_of_augustine_01_intro.htm}{introduction};
accessed 2026-07-15. Public-domain edition; used primarily by paraphrase.
\item Prosper of Aquitaine. \emph{Epitoma chronicon}, in Theodor Mommsen,
ed., \emph{Chronica minora}, MGH Auctores antiquissimi 9 (Berlin, 1892),
p.~473,
\sourceurl{https://www.dmgh.de/mgh_auct_ant_9/index.htm\#page/473/mode/1up}{digital
MGH}; accessed 2026-07-17. The entry directly supplies the 28 August death date.
\item \emph{Gesta collationis Carthaginiensis}, Paris, BnF lat.~1546,
fols.~1v--143v,
\sourceurl{https://doi.org/10.11588/diglit.13514}{Bibliotheca Laureshamensis
facsimile}; and Clemens Weidmann, ed., \emph{Collatio Carthaginensis anni 411},
CSEL 104 (Berlin/Boston, 2018),
\sourceurl{https://csel.at/en/news/csel-104-collatio-carthaginensis-anni-411/}{edition
record}; accessed 2026-07-17. Used for the representatives and dossier
architecture, not as neutral or complete proceedings.
\item \emph{Codex Theodosianus} 16.5.38 (12 February 405), Mommsen--Meyer
text,
\sourceurl{https://www.ancienttexts.org/library/latinlibrary/theod16.html}{online
Latin}; accessed 2026-07-17. Used to establish the imperial legal setting.
\item Augustine. Letter 211.5--16,
\sourceurl{https://www.augustinus.it/latino/lettere/lettera_217_testo.htm}{Latin
text}; with the
\sourceurl{https://earlymedievalmonasticism.org/texts/Augustine-Regulae.html}{Rule
text-and-manuscript catalogue}; accessed 2026-07-17. Used for the women's
community and the plurality of transmitted Rule text families.
\item Paul the Deacon. \emph{Historia Langobardorum} 6.48, ed. Georg Waitz,
MGH \emph{Scriptores rerum Langobardicarum} (Hanover, 1878), p.~234,
\sourceurl{https://saraceni.uni-koeln.de/wiki/Historia_Langobardorum_0010}{Latin,
edition locus, and research translation}; accessed 2026-07-17. Used only for
the late-eighth-century Sardinia--Pavia translation report.
\item Benedict XVI\@. General Audience, 9 January 2008,
\sourceurl{https://www.vatican.va/content/benedict-xvi/en/audiences/2008/documents/hf_ben-xvi_aud_20080109.html}{“Saint Augustine of Hippo”};
accessed 2026-07-15. Official ecclesial reception; Vatican English remains
third-party text and is paraphrased.
\end{itemize}
