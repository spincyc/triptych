\subsection{Detailed chronology}\label{sec:aug-detailed-chronology}

\begin{biotimeline}
\biotimelinerow{13 Nov.~354}{Thagaste}{Birth to Patricius and Monnica. Birthday: \emph{De beata vita} 1.6; year: ancient chronological convergence.}{Attested / reconstructed}
\biotimelinerow{c.~365}{Madaura}{Advanced grammar and literary schooling begins.}{Approximate}
\biotimelinerow{c.~370--371}{Thagaste}{Education pauses for lack of funds; pear theft belongs to Augustine's retrospective account of this interval.}{Approximate / A}
\biotimelinerow{371--372}{Carthage}{Studies rhetoric; begins long partnership; Adeodatus born around 372.}{Approximate / A}
\biotimelinerow{373}{Carthage}{Cicero's \emph{Hortensius} awakens the desire for wisdom; Augustine joins Manichaean hearers.}{Approximate / A}
\biotimelinerow{374--383}{Thagaste / Carthage}{Teaches grammar and rhetoric; remains in Manichaean networks; death of close friend; meets Faustus.}{Sequence attested}
\biotimelinerow{383}{Rome}{Secret departure from Carthage; illness and teaching in Rome.}{Attested in A}
\biotimelinerow{384}{Milan}{Appointed imperial professor of rhetoric through Symmachus; begins hearing Ambrose.}{Well attested}
\biotimelinerow{385--386}{Milan}{Marriage negotiations; unnamed partner returns to Africa; Platonist reading and Pauline study.}{Sequence attested}
\biotimelinerow{late summer 386}{Milan}{Garden conversion with Alypius as narrated in \emph{Confessions} 8.12.}{Season probable; exact day unresolved}
\biotimelinerow{386--387}{Cassiciacum / Milan}{Philosophical-retreat dialogues; catechumenate.}{Well attested}
\biotimelinerow{Easter Vigil 387}{Milan}{Baptized by Ambrose with Adeodatus and Alypius.}{Event attested; exact civil day not supplied here}
\biotimelinerow{387}{Ostia}{Contemplative conversation and death of Monnica.}{Attested in A}
\biotimelinerow{After 387; date unresolved}{Rome}{Delayed before African return; anti-Manichaean writing.}{Relative sequence only}
\biotimelinerow{after c.~389; year unknown}{Thagaste / Africa}{Lay common life; Augustine says Adeodatus was taken from life not long after the dialogue associated with him.}{Relative sequence only}
\biotimelinerow{391}{Hippo Regius}{Ordained presbyter by Valerius; monastery and preaching ministry established.}{Well attested A/N}
\biotimelinerow{392}{Hippo}{Public debate with Fortunatus.}{Dated record}
\biotimelinerow{395/396}{Hippo}{Consecrated coadjutor and succeeds Valerius.}{Competing year placement}
\biotimelinerow{After becoming bishop}{Hippo}{Composes \emph{Confessions}; \emph{Retractationes} 2.6 supplies the relative corpus position.}{Exact span not controlled here}
\biotimelinerow{Across many episcopal years}{Hippo}{Long composition of \emph{The Trinity}.}{Exact span not controlled here}
\biotimelinerow{By Letter 93}{Roman Africa}{Augustine defends corrective legal pressure against Donatists.}{A; conventional 407/408 date not edition-checked}
\biotimelinerow{June 411}{Carthage}{Imperially convened conference; the unique surviving \emph{Gesta} manuscript names Augustine among seven Catholic representatives and preserves an incomplete three-session dossier.}{A/N adversarial record; not neutral dialogue}
\biotimelinerow{After 410}{Africa / Mediterranean}{Augustine's anti-Pelagian writings develop; opponent, council, and imperial dossiers remain uncollated.}{A-side sequence only}
\biotimelinerow{After Rome's sack; before late self-review}{Hippo}{Composes \emph{City of God}.}{Relative A chronology}
\biotimelinerow{426}{Hippo}{Public designation of Heraclius; Letter 213 records proceedings.}{Dated record}
\biotimelinerow{Late episcopate}{Hippo}{Begins \emph{Retractationes}.}{Exact start year not controlled here}
\biotimelinerow{429}{Roman Africa}{Vandals cross into Africa; refugee and military crisis.}{External chronology}
\biotimelinerow{430}{Hippo}{Siege; Augustine becomes ill in its third month.}{Possidius N}
\biotimelinerow{28 Aug.~430}{Hippo}{Death and burial during the siege are described by Possidius; Prosper's \emph{Chronicle} supplies \emph{V kalendas Septembris}.}{N death and burial; near-contemporary E exact day}
\end{biotimeline}
