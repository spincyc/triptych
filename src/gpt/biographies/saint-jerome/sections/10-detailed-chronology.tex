\section{Detailed chronology}\label{sec:jerome-detailed-chronology}

\begin{biotimeline}
\biotimelinerow{331 or c.~347}{Stridon}{Birth to a Christian provincial family; Jerome names his father Eusebius and the Dalmatian-Pannonian border town in \emph{De viris illustribus} 135.}{Competing ancient/modern chronologies}
\biotimelinerow{350s/360s}{Rome}{Grammar and rhetoric under Aelius Donatus; book collecting, classical formation, and visits to martyrs' memorials.}{Relative sequence strong}
\biotimelinerow{c.~358--366}{Rome}{Baptism, often associated with Liberius's episcopate.}{Range depends on birth chronology}
\biotimelinerow{late 360s--early 370s}{Trier / Aquileia}{Travel, encounter with monastic ideals, and common ascetic circle with Rufinus, Bonosus, Heliodorus, and others.}{Approximate}
\biotimelinerow{373/374}{Antioch}{Arrival in the East; illness and loss of companions.}{Approximate A}
\biotimelinerow{c.~375}{Antioch / Chalcis}{Fever and Ciceronian dream as later narrated in Letter 22.30.}{Event A; exact date reconstructed}
\biotimelinerow{c.~375--377}{Chalcis region}{Ascetic withdrawal, correspondence, Greek/Syriac setting, and beginning Hebrew study.}{Networked, not isolated}
\biotimelinerow{c.~378/379}{Antioch}{Ordained presbyter by Paulinus during the schism.}{Probable year}
\biotimelinerow{c.~379--382}{Constantinople}{Studies with Gregory of Nazianzus; translates Greek Christian works; council-era presence.}{Role at council not precisely documented}
\biotimelinerow{382}{Rome}{Returns with eastern bishops; enters Damasus's close scholarly service.}{Well attested}
\biotimelinerow{382--384}{Rome}{Revises four Latin Gospels against Greek manuscripts.}{Secure project; not whole NT}
\biotimelinerow{384}{Rome}{Letter 22 to Eustochium; Roman ascetic and scriptural school flourishes.}{Dated A}
\biotimelinerow{11 Dec.~384}{Rome}{Death of Damasus removes principal papal patron.}{External chronology}
\biotimelinerow{Aug.~385}{Ostia / eastern voyage}{Leaves Rome amid controversy; Letter 45 and \emph{Against Rufinus} 3.22 defend the departure.}{Event secure; legal characterization disputed}
\biotimelinerow{385--386}{Palestine / Egypt}{Paula and Eustochium travel; biblical pilgrimage and Egyptian monastic visits.}{Principally Letter 108 A}
\biotimelinerow{386}{Bethlehem}{Men's and women's communities and pilgrim hospitality established through Paula's resources.}{Strong A/M; roles distinguished}
\biotimelinerow{late 380s}{Bethlehem}{Greek-based Psalm work and first commentaries; deeper Hebrew study and textual collection.}{Project sequence approximate}
\biotimelinerow{c.~390--405}{Bethlehem}{Hebrew-based translations of most Old Testament books, with distinct methods and prefaces.}{Book dates vary}
\biotimelinerow{392/393}{Bethlehem}{Completes \emph{De viris illustribus}, ending with his own work list.}{Strong date range}
\biotimelinerow{393}{Bethlehem}{Writes \emph{Against Jovinian}; Roman allies contest its treatment of marriage.}{Dated controversy}
\biotimelinerow{393--397}{Palestine}{Conflict with John of Jerusalem and Epiphanius over Origen and jurisdiction.}{Documented, partisan dossiers}
\biotimelinerow{c.~394--405}{Hippo / Bethlehem by correspondence}{Early Augustine exchange over Galatians and biblical translation; letters circulate unevenly.}{Dual numbering; dates refined by letter}
\biotimelinerow{398--402}{Rome / Aquileia / Bethlehem}{Rufinus controversy over Origen's works, translation, and personal authority.}{Well documented; mutually hostile}
\biotimelinerow{26 Jan.~404}{Bethlehem}{Paula dies; Jerome composes Letter 108 to Eustochium.}{Strong A date}
\biotimelinerow{406}{Bethlehem}{Writes \emph{Against Vigilantius} on relics, vigils, alms, and asceticism.}{Dated polemic}
\biotimelinerow{410--411}{Rome / Bethlehem}{Sack of Rome, refugee crisis, Marcella's death, and Jerome's Letter 127.}{Sequence strong; letter is memorial}
\biotimelinerow{c.~415--417}{Bethlehem}{Writes dialogue against Pelagian positions; alliance with Augustine deepens.}{Dating varies}
\biotimelinerow{416--417}{Bethlehem / Rome}{Augustine, Innocent, and Jerome report serious violence and burned monastic buildings; Innocent also records the absence of named charges.}{Destruction converges; casualties, perpetrator, command, adjudication, and aftermath unresolved}
\biotimelinerow{late 418 / early 419}{Bethlehem}{Eustochium dies; younger Paula and community continue.}{Critical reconstruction; exact date open}
\biotimelinerow{30 Sept.~419 or 420}{Bethlehem}{Jerome dies; 420 is official Catholic reception, while 419 persists in critical chronology.}{Year disputed; day received}
\biotimelinerow{1546}{Trent}{Council receives the old Latin Vulgate as authentic for ecclesial public use.}{M; not whole-Bible authorship ruling}
\biotimelinerow{1979}{Rome}{John Paul II promulgates the \emph{Nova Vulgata}.}{M; revised typical Latin edition}
\biotimelinerow{2020}{Rome}{Francis issues \emph{Scripturae Sacrae affectus} for the received 1600th anniversary.}{M reception}
\end{biotimeline}
