\section{Sources and Further Study}

The sources below are the texts actually used. Primary and official sources control status, corpus, and ecclesial interpretation; documentary scholarship supplies chronology and comparison. All mutable official links and the DDF index were rechecked through 16 July 2026.

\subsection{Public Revelation and doctrinal controls}

\begin{enumerate}
  \item Sacred Scripture: especially Exod 20:7--11; Deut 5:12--15; Jer 18:7--10; Jonah; Job; Lk 1:46--55; 13:1--5; Jn 9:1--3; 11:35; 1 Tim 2:5--6; 2 Cor 5:18--21; Col 1:19--20; and Jas 5:12.
  \item Second Vatican Council, \emph{Dei Verbum}, 2--4, on Revelation's completion in Christ: \sourceurl{https://www.vatican.va/archive/hist_councils/ii_vatican_council/documents/vat-ii_const_19651118_dei-verbum_en.html}{official text}.
  \item Second Vatican Council, \emph{Lumen gentium}, 52--69, especially 56 and 60--62 on Mary's graced cooperation and Christ's unique mediation: \sourceurl{https://www.vatican.va/archive/hist_councils/ii_vatican_council/documents/vat-ii_const_19641121_lumen-gentium_en.html}{official text}.
  \item \emph{Catechism of the Catholic Church}, 65--67, 970, 2168--2195, and 2683--2684: \sourceurl{https://www.vatican.va/content/catechism/en.html}{official edition}.
  \item Dicastery for the Doctrine of the Faith, \emph{Norms for Proceeding in the Discernment of Alleged Supernatural Phenomena} (17 May 2024; effective 19 May), including the presentation's explicit citation of the 1851 La Salette formula: \sourceurl{https://www.vatican.va/roman_curia/congregations/cfaith/documents/rc_ddf_doc_20240517_norme-fenomeni-soprannaturali_en.html}{official text}.
  \item DDF, \emph{Mater Populi Fidelis}: Doctrinal Note on Some Marian Titles Regarding Mary's Cooperation in the Work of Salvation (4 November 2025), especially the safeguards concerning Christ's unique mediation and Marian intercession: \sourceurl{https://www.vatican.va/roman_curia/congregations/cfaith/documents/rc_ddf_doc_20251104_mater-populi-fidelis_en.html}{official text}.
  \item DDF complete document index, searched for later La Salette action through 16 July 2026: \sourceurl{https://www.vatican.va/roman_curia/congregations/cfaith/doc_doc_index.htm}{official index}.
  \item Code of Canon Law (1983), canon 6 \S1, especially 3$^\circ$ on earlier Apostolic See penal laws: \sourceurl{https://www.vatican.va/archive/cod-iuris-canonici/eng/documents/cic_lib1-cann1-6_en.html}{official text}.
\end{enumerate}

\subsection{The event, early documents, and Grenoble acts}

\begin{enumerate}
  \item Jean Stern, M.S., ed., \emph{La Salette: Documents authentiques---dossier chronologique intégral}, vol. 1, \emph{Septembre 1846--début mars 1847} (Paris: Desclée de Brouwer, 1980); vol. 2, \emph{Fin mars 1847--avril 1849} (Paris and Corps: Cerf / Association des pèlerins, 1984); vol. 3, \emph{Confirmation de la grâce de La Salette par l'Église: 1er mai 1849--4 novembre 1854} (Paris and Corps: Cerf / Association des pèlerins, 1991). Bibliographic controls: \sourceurl{https://search.worldcat.org/title/7732514}{WorldCat record}; \sourceurl{https://patrimoine.bm-grenoble.fr/PATRIMOINE/doc/SYRACUSE/372183/la-salette-documents-authentiques-dossier-chronologique-integral-3-confirmation-de-la-grace-de-la-sa}{volume 3 record}.
  \item Missionaries of Our Lady of La Salette, ``Apparition,'' received English public narrative, physical description, message, and 1851 formula: \sourceurl{https://www.lasalette.info/mission/apparition}{official institute page}.
  \item Shrine of Our Lady of La Salette, ``L'histoire,'' current French custodial narrative and pastoral presentation: \sourceurl{https://lasalette.cef.fr/lhistoire/}{official shrine page}.
  \item Philibert de Bruillard, mandement of 19 September 1851, especially Articles 1--4 and 7, reproduced in Stern, vol. 3, and in Maximin Giraud's scanned \emph{Ma profession de foi}, printed pp. 70--72 (PDF pp. 71--73): \sourceurl{https://melanie-de-la-salette.fr/livres/maximin-ma-profession-de-foi/Ma_profession_de_foi_sur_l_apparition_de.pdf}{public-domain scan}. The exact final clause is independently quoted by the DDF's 2024 presentation.
  \item De Bruillard, \emph{Lettre pastorale et Mandement} (1 May 1852), on the shrine, cornerstone, missionaries, Eucharist, and ministry of reconciliation: \sourceurl{https://lasalette.info/fr/l-informations/205-1-may-1852}{Missionaries' transcription}.
  \item Jacques-Marie-Achille Ginoulhiac, \emph{Instruction pastorale et Mandement de Mgr l'évêque de Grenoble, portant condamnation du livre intitulé: Affaire de La Salette, Mémoire au Pape, etc.} (Grenoble: J. Baratier, 4 November 1854), forty-seven-page primary edition: \sourceurl{https://mediatheques.montpellier3m.fr/BIBNUMERIQUE/doc/CAMO/10090/instruction-pastorale-et-mandement-de-mgr-l-eveque-de-grenoble-ginoulhiac-portant-condamnation-du-li}{digital catalog and scan record}; also Stern, vol. 3.
  \item Marie Des Brûlais, \emph{Suite de l'Écho de la Sainte Montagne} (Nantes: Imprimerie Charpentier, December 1855; reprinted 1904), printed p. 97 and its footnote for the children's completed mission, and printed p. 106 for the ``Souvenir'' of Ginoulhiac's 19 September allocution sending pilgrims as missionaries: \sourceurl{https://gallica.bnf.fr/ark:/12148/bpt6k65339891}{BnF Gallica scan}, PDF pp. 109 and 118. This near-contemporary report controls the pastoral substance, not every polished form of the later maxim.
  \item Missionaries of Our Lady of La Salette, ``L'Histoire de la Congrégation,'' for the Pra relation, subsistence crisis, inquiry conferences, pilgrimage, 1851--1854 acts, institute, basilica, witnesses' deaths, and reception chronology: \sourceurl{https://www.lasalette.info/fr/mission/l-histoire-de-la-congregation}{official institute chronology}.
\end{enumerate}

\subsection{The secret texts and Roman discipline}

\begin{enumerate}
  \item René Laurentin and Michel Corteville, \emph{Découverte du secret de La Salette: Au-delà des polémiques, la vérité sur l'apparition et ses voyants} (Paris: Fayard, 2002), 263 pp., especially manuscript facsimiles and pp. 46--49, 253--278. Bibliographic control: \sourceurl{https://openlibrary.org/works/OL36975746W/Decouverte_du_secret_de_La_Salette}{catalog record}. This is scholarship, not a DDF edition.
  \item Public facsimile-based transcription of the Maximin and Mélanie 1851 manuscripts, checked against the Laurentin--Corteville publication: \sourceurl{https://patristica.net/La-Salette\%26f\%26e}{French/English transcription and images}. The site's theological claims do not control this study.
  \item Mélanie Calvat, \emph{L'Apparition de la Très Sainte Vierge sur la montagne de La Salette le samedi 19 septembre 1846}, manuscript dated Castellammare, 21 November 1878; Lecce publication with local permission dated 15 November 1879: \sourceurl{https://melanie-de-la-salette.fr/livres/brochure-1879/brochure.pdf}{public-domain booklet scan}. Used as the primary expanded text, not as an approved message.
  \item Missionaries of Our Lady of La Salette, René Butler, M.S., ``Call to Integrity'' (17 October 2017), expressly distinguishing the unapproved secrets, the rediscovered 1851 texts, and the much longer 1879 version: \sourceurl{https://www.lasalette.info/mission/our-rule-of-life/tag/reflection?start=250}{official institute reflection}.
  \item Supreme Sacred Congregation of the Holy Office, \emph{Decretum circa vulgo dictum ``Secret de La Salette''}, 21 December 1915, AAS 7 (1915), 594: \sourceurl{https://www.vatican.va/archive/aas/documents/AAS-07-1915-ocr.pdf}{official AAS volume}.
  \item Holy Office declaration of 12 April 1916, AAS 8 (1916), 175, and Congregation of the Index decree of 5 June 1916, pp. 178--179, on \emph{La Leçon de l'Hôpital Notre-Dame d'Ypres}: \sourceurl{https://www.vatican.va/archive/aas/documents/AAS-08-1916-ocr.pdf}{official AAS volume}.
  \item Holy Office decree of 9--10 May 1923, AAS 15 (1923), 287--288, proscribing the named 1922 Société Saint-Augustin booklet and printing ``19 septembre 1845'' [\emph{sic}]: \sourceurl{https://www.vatican.va/archive/aas/documents/AAS-15-1923-ocr.pdf}{official AAS volume}.
  \item Congregation for the Doctrine of the Faith, notification on the Index (14 June 1966), AAS 58 (1966), 445: \sourceurl{https://www.vatican.va/roman_curia/congregations/cfaith/documents/rc_con_cfaith_doc_19660614_de-indicis-libr-prohib_en.html}{official English text}; and interpretive decree (15 November 1966): \sourceurl{https://www.vatican.va/roman_curia/congregations/cfaith/documents/rc_con_cfaith_doc_19661115_decretum_fr.html}{official French text}.
\end{enumerate}

\end{enumerate}

\subsection{Historical interpretation}

\begin{enumerate}
  \item Jean Stern, \emph{Le curé d'Ars et le message authentique de La Salette: La préhistoire des pseudo-secrets}, preface by Guy de Kerimel (Paris: L'Harmattan, 2018): \sourceurl{https://api.pageplace.de/preview/DT0400.9782140089206_A49331451/preview-9782140089206_A49331451.pdf}{publisher preview}.
  \item Sandra L. Zimdars-Swartz, \emph{Encountering Mary: From La Salette to Medjugorje} (Princeton University Press, 1991; later reprints). Used for nineteenth-century visionary culture and reception, not ecclesial status.
  \item François Angelier and Claude Langlois, eds., \emph{La Salette: Apocalypse, pèlerinage et littérature (1856--1996)} (Grenoble: Jérôme Millon, 2000). Used as multidisciplinary historical context, not a competent act.
  \item Laurent Gruaz, ``Maximin Giraud, le berger de La Salette: de l'Apparition de la Vierge aux soldats du Pape,'' \emph{Chrétiens et sociétés} 17 (2011), 151--172, DOI 10.4000/chretienssocietes.2827. Used for Maximin's documented 1865 service.
\end{enumerate}
