% Deterministic semantic layer for the Ark/New-Ark atlas plates.
% The cartographic graphite plate supplies geography and relief only.  Every
% place, route, status, number, label, and typological relation remains TeX.
%
% The source raster is one 3800x2436 composite.  Both commands below place it
% at 31.2x20 cm and clip to one full-width panel at approximately 309 dpi.  The
% projection rectangles and formulas therefore remain the exact v3 Web
% Mercator registration while each panel becomes a full-width atlas plate.

% Formulaic EPSG:3857 projection into the source-first regional panel.
% Longitude is linear.  For latitude phi, M(phi)=ln(tan(45 degrees+phi/2));
% north-up y is bottom+height*[M(phi)-M(south)]/[M(north)-M(south)].
\newcommand{\ArkRegionalPoint}[3]{%
  \pgfmathsetmacro{\arkProjectedX}{%
    .344842 + 7.141970*((#2)-31.0)/(36.5-31.0)}%
  \pgfmathsetmacro{\arkMercatorPoint}{ln(tan(45+(#3)/2))}%
  \pgfmathsetmacro{\arkMercatorSouth}{ln(tan(45+27.4/2))}%
  \pgfmathsetmacro{\arkMercatorNorth}{ln(tan(45+33.5/2))}%
  \pgfmathsetmacro{\arkProjectedY}{%
    .402299 + 9.195402*(\arkMercatorPoint-\arkMercatorSouth)/%
    (\arkMercatorNorth-\arkMercatorSouth)}%
  \coordinate (#1) at (\arkProjectedX,\arkProjectedY);%
}

% Formulaic EPSG:3857 projection into the source-first Levant panel.
\newcommand{\ArkLevantPoint}[3]{%
  \pgfmathsetmacro{\arkProjectedX}{%
    7.865684 + 7.389474*((#2)-33.1)/(36.2-33.1)}%
  \pgfmathsetmacro{\arkMercatorPoint}{ln(tan(45+(#3)/2))}%
  \pgfmathsetmacro{\arkMercatorSouth}{ln(tan(45+31.2/2))}%
  \pgfmathsetmacro{\arkMercatorNorth}{ln(tan(45+33.5/2))}%
  \pgfmathsetmacro{\arkProjectedY}{%
    1.754867 + 6.490266*(\arkMercatorPoint-\arkMercatorSouth)/%
    (\arkMercatorNorth-\arkMercatorSouth)}%
  \coordinate (#1) at (\arkProjectedX,\arkProjectedY);%
}
